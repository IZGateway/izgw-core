1   package gov.cdc.izgateway.soap.fault;
2   
3   import gov.cdc.izgateway.common.HasDestinationId;
4   import gov.cdc.izgateway.model.RetryStrategy;
5   import lombok.Getter;
6   
7   public class UnknownDestinationFault extends Fault implements HasDestinationId {
8       private static final long serialVersionUID = 1L;
9       public static final String FAULT_NAME = "UnknownDestinationFault";
10      private static final MessageSupport[] MESSAGE_TEMPLATES = {
11          new MessageSupport(FAULT_NAME, "40", "Missing HubRequestHeader", null,
12              "The Request to IZ Gateway is missing the required <iishub:HubRequestHeader> or <iishub:DestinationId> element in the <soap:Header>.", RetryStrategy.CORRECT_MESSAGE),
13          new MessageSupport(FAULT_NAME, "41", "IIS Not Registered", null,
14              "The Destination IIS is not registered for use in IZ Gateway for the requested operation. If this IIS should be registered, "
15              + "contact IZ Gateway support.", RetryStrategy.CONTACT_SUPPORT),
16          new MessageSupport(FAULT_NAME, "42", "Invalid Destination ID", null,
17              "The Destination IIS is not valid for use in IZ Gateway.", RetryStrategy.CORRECT_MESSAGE)
18      };
19      @Getter 
20      private final String destinationId;
21      public UnknownDestinationFault(int index, String destId, String message, Throwable cause) {
22          super(MESSAGE_TEMPLATES[index], message, cause);
23      	this.destinationId = destId;
24      }
25      
26      public static UnknownDestinationFault missingDestination(String message) {
27          return new UnknownDestinationFault(0, "_null_", message, null);
28      }
29      
30      public static UnknownDestinationFault unknownDestination(String destId, String message, Throwable cause) {
31          return new UnknownDestinationFault(1, destId, message, cause);
32      }
33  
34      public static UnknownDestinationFault invalidDestination(String destId, String message) {
35          return new UnknownDestinationFault(2, destId == null ? "_null_" : destId, message, null);
36      }
37      
38      public static UnknownDestinationFault devAction(String destId) {
39      	return unknownDestination(destId, "Simulated UnknownDestinationFault", new Exception("There is no reason for this"));
40      }
41      	 
42  }
