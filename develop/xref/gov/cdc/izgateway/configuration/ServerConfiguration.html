3   import java.net.URI;
4   import java.net.URL;

6   import org.springframework.beans.factory.InitializingBean;
7   import org.springframework.boot.context.properties.ConfigurationProperties;
8   import org.springframework.context.annotation.Configuration;
9   
10  import io.swagger.v3.oas.annotations.media.Schema;
11  import lombok.Data;
12  
13  @Configuration
14  @ConfigurationProperties(prefix = "server")
15  @Data
16  public class ServerConfiguration implements InitializingBean {
17  	@Schema(description = "The server hostname (which should be found in a a certificate having a private key in the keystore)")
18  	private String hostname = "dev.izgateway.org";
19  
20  	@Schema(description = "The server port (typically 443 for https and 9081 for http)")
21  	private int port = 8080;
22  
23  	@Schema(description = "The server protocol (typically https, but http may be used for some testing)")
24  	private String protocol = "http";
25  
26  	@Schema(description = "The baseUrl to use for the server. This is computed after other properties are set, which will trash any configured value")
27  	private URL baseUrl;
28  
29  	@Schema(
30  		description = "The mode of operation for the server (production or development).  Production servers will NOT log any PHI.  Development servers"
31  				+ "may log it for diagnostic testing. The onboarding server typically operates in development mode.", 
32  		pattern="prod|dev"
33  	)
34  	private String mode = "prod";
35  
36  	@Override
37  	public void afterPropertiesSet() throws Exception {
38  		baseUrl = new URI(protocol, null, hostname, port, "/", null, null).toURL();
39  	}
40  }
