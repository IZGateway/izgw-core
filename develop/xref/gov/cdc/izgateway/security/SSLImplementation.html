3   import javax.net.ssl.TrustManager;
4   import javax.net.ssl.X509ExtendedTrustManager;
5   
6   import org.apache.tomcat.util.net.SSLHostConfigCertificate;
7   import org.apache.tomcat.util.net.SSLUtil;
8   import org.apache.tomcat.util.net.jsse.JSSEImplementation;
9   import org.apache.tomcat.util.net.jsse.JSSEUtil;
10  
11  import gov.cdc.izgateway.security.ocsp.RevocationTrustManager;
12  
13  public class SSLImplementation extends JSSEImplementation {
14  	/**
15  	 * This class overrides the JSSEUtil.getTrustManagers response to return a RevocationTrustManager.
16  	 * @author boonek
17  	 *
18  	 */
19  	private static class JSSEUtilWithOCSP extends JSSEUtil {
20  		public JSSEUtilWithOCSP(SSLHostConfigCertificate certificate) {
21  			super(certificate);
22  		}
23  
24  		@Override
25  		public TrustManager[] getTrustManagers() throws Exception {
26  			TrustManager[] tmgrs = super.getTrustManagers();
27  			for (int i = 0; i < tmgrs.length; i++) {
28  				if (tmgrs[i] instanceof X509ExtendedTrustManager x509tmgr) {
29  					tmgrs[i] = new RevocationTrustManager(x509tmgr);
30  				}
31  			}
32  			return tmgrs;
33  		}
34  	}
35  	
36      @Override
37      public SSLUtil getSSLUtil(SSLHostConfigCertificate certificate) {
38          return new JSSEUtilWithOCSP(certificate);
39      }
40  
41  }
