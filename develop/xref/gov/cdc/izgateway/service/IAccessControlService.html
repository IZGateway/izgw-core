18  public interface IAccessControlService {

35  	void refresh();

40  	/**

43  	 * The value is an object describing the group.
44  	 * @return Get mapping of users to groups
45       */
46  	Map<String, Object> getGroups();
47  
48  	/**

50  	 * @param user	The user
51  	 * @param group	The group
52  	 * @return	true if the user is in the group
53  	 */
54  	boolean isUserInGroup(String user, String group);
55  	
56  	/**

58  	 * @param user	The user (certificate common name)
59  	 * @param role	The role
60  	 * @return	true if user is permitted in the given role
61  	 */
62  	boolean isUserInRole(String user, String role);
63  
64  	/**

66  	 * @param user	The user
67  	 * @return	true if the user has been Denied access
68  	 */
69  	boolean isUserDenied(String user);
70  
71  	/**
72  	 * @return the list of event (submission) types
73  	 */
74  	Set<String> getEventTypes();
75  
76  	/**
77  	 * Get the roles allows to access a given method and path
78  	 * @param method	The HTTP method
79  	 * @param path	The URL path
80  	 * @return	The list of roles permitted to access the path and method
81  	 */
82  	List<String> getAllowedRoles(RequestMethod method, String path);
83  
84  	/**
85  	 * Determine if a user can access a given path and method
86  	 * @param user	The user
87  	 * @param method	The HTTP method
88  	 * @param path	The URL path
89  	 * @return true if the user can access the specified method and path
90  	 */
91  	Boolean checkAccess(String user, String method, String path);
92  
93  	/**
94  	 * Determines whether a user is a member of a given group.
95  	 * @param user	The user
96  	 * @param group	The group
97  	 * @return	True if use is a member of the group
98  	 */
99  	boolean canAccessDestination(String user, String group);
100 
101 	/**
102 	 * Removes a user from the denylist.  Only used for deny listing at this point
103 	 * INSIDE IZGW, and not suitable for general user/group membership management.
104 	 * @param user	The user
105 	 * @return	The access control entry that was removed, or null if none existed.
106 	 */
107 	Object removeUserFromDenyList(String user);
108 
109 	/**
110 	 * Add a user to the DenyList. Only used for denying at this point INSIDE IZGW,
111 	 * and not suitable for general user/group membership management
112 	 * @param user	The user
113 	 * @param reason The reason for adding the user to the deny list
114 	 * @return	The new access control entry that was added.
115 	 */
116 	Object addUserToDenyList(String user, String reason);
117 
118 	/**
119 	 * Get the list of deny listed users.
120 	 * @return The set of deny listed users.
121 	 */
122 	Set<String> getDenyList();
123 
124 	/**
125 	 * Check access to a destination, throwing an exception if access is denied.
126 	 * @param destId The destination ID.
127 	 * @throws SecurityFault If access is denied
128 	 */

130 		// Default implementation does nothing.
131 	}

138 	 */
139 	default IFileType getFileType(String reportType) {
140 		return null;
141 	}
