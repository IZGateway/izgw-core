6   import com.fasterxml.jackson.annotation.JsonFormat;
7   import com.fasterxml.jackson.annotation.JsonFormat.Shape;
8   import com.fasterxml.jackson.annotation.JsonIgnore;
9   import com.fasterxml.jackson.annotation.JsonProperty;
10  
11  import gov.cdc.izgateway.common.Constants;
12  import io.swagger.v3.oas.annotations.media.Schema;
13  import lombok.Data;
14  import lombok.EqualsAndHashCode;
15  import lombok.NoArgsConstructor;
16  
17  /**
18   * An endpoint describes the inbound or outbound connection to
19   * IZ Gateway during a transaction.  It is abstract to ensure
20   * that it is not used on its own.
21   */
22  @Schema(description = "Information common to all endpoints")
23  @Data
24  @EqualsAndHashCode(callSuper=true)
25  @NoArgsConstructor
26  public abstract class EndPointInfo extends HostInfo implements Serializable {
27  	private static final long serialVersionUID = 1L;
28  
29      @Schema(description="The common name on the certificate associated with the requester")
30      @JsonProperty
31      protected String commonName;
32  
33      @Schema(description="The cipher suite used by the endpoint.")
34      @JsonProperty
35      protected String cipherSuite;
36  
37      @Schema(description="The organization associated with the with the endpoint.")
38      @JsonProperty
39      protected String organization;
40  
41      @Schema(description="The serial number associated with the with certificate on the endpoint.")
42      @JsonProperty
43      protected String serialNumber;
44  
45      protected String serialNumberHex;
46  
47      @JsonProperty
48      @JsonFormat(shape=Shape.STRING, pattern = Constants.TIMESTAMP_FORMAT)
49      @Schema(description="The starting date associated with the with certificate on the endpoint.")
50      protected Date validFrom;
51  
52      @JsonProperty
53      @Schema(description="The expiration date associated with the with certificate on the endpoint.")
54      @JsonFormat(shape=Shape.STRING, pattern = Constants.TIMESTAMP_FORMAT)
55      protected Date validTo;
56  
57      @Schema(description="The identifier of the endpoint.")
58      @JsonProperty
59      protected String id;
60  
61      /**
62       * Copy constructor
63       * @param that The endpoint to copy
64       */
65      protected EndPointInfo(EndPointInfo that) {
66      	super(that.ipAddress, that.host);
67          this.commonName = that.getCommonName();
68          this.organization = that.organization;
69          this.serialNumber = that.serialNumber;
70          this.serialNumberHex = that.serialNumberHex;
71          this.validFrom = that.validFrom;
72          this.validTo = that.validTo;
73          this.id = that.id;
74          this.cipherSuite = that.cipherSuite;
75      }
76  
77      @Schema(description="The FIPS state code associated with the jurisdiction associated with the endpoint.")
78      @JsonProperty
79      public String getFips() {
80          return id == null ? null : id.toUpperCase();
81      }
82      
83      /**
84       * This getter is necessary to ensure that the serial number is reported
85       * @return the serialNumber in decimal
86       */
87      @Schema(description="The serial number in decimal associated with the with certificate on the endpoint.")
88      @JsonProperty
89      public String getSerialNumber() {
90          return serialNumber;
91      }
92  
93      /**
94       * @return the serialNumber in hexadecimal
95       */
96      @Schema(description="The serial number in hex associated with the with certificate on the endpoint.")
97      @JsonProperty
98      public String getSerialNumberHex() {
99          return serialNumberHex;
100     }
101 
102     @Schema(description="The organization name in the certificate associated with the endpoint.")
103     @JsonIgnore 
104     public String getName() {
105 		return StringUtils.isBlank(organization) ? commonName : organization;
106     }
107 }
