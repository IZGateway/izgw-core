97  	protected static String convertSize(long sizeBytes) {
98      	if (sizeBytes <= 0) {
99      		return "0b";
100     	}
101     	String[] sizeName = {"b", "Kb", "Mb", "Gb"};
102     	int i = (int)Math.floor(Math.log(sizeBytes)/Math.log(1024));
103 		double p = Math.pow(1024, i);
104 		return String.format("%0.2f %s", sizeBytes / p, sizeName[i]);
105     }

107     @Override
108     public final void invoke(Request req, Response resp) throws IOException, ServletException {
109         fixHeaders(req);
110 
111         // If no event was created for this session, go ahead and create a new one.
112         TransactionData t = createTransactionData(req);
113 
114         SourceInfo source = setSourceInfoValues(req, t);
115 
116         // Set RequestContext values.
117         IzgPrincipal p = getPrincipal(req);
118         RequestContext.setPrincipal(p);
119         RequestContext.setTransactionData(t);
120         RequestContext.setHttpHeaders(getHeaders(req));
121         RequestContext.setResponse(resp);
122         
123         // Update the principal in source
124         t.getSource().setPrincipal(p);
125         // Set MDC Values must occur after principal is set in source
126         setMdcValues(req, t.getEventId(), source.getCommonName());
127         
128         if (!isLogged(req.getRequestURI())) {
129             RequestContext.disableTransactionDataLogging();
130         }
131 
132         try {
133             handleSpecificInvoke(req, resp, source);
134         } catch (Exception ex) {
135             log.error(Markers2.append(ex), "Uncaught Exception during invocation", ex);
136         } catch (Error err) {  // NOSONAR OK to Catch Error here
137             log.error(Markers2.append(err), "Error during invocation", err);
138         } finally {
139             // Log first, then clean up MDC!
140             if (t != null && !RequestContext.isLoggingDisabled()) {
141                 t.logIt();
142             }
143             RequestContext.clear();
144             clearMdcValues();
145             clearContext();
146         }
147     }

166     protected static Map<String, List<String>> getHeaders(Response resp) {
167     	Map<String, List<String>> headers = new TreeMap<>();
168     	for (String name: resp.getHeaderNames()) {
169 			List<String> l = new ArrayList<>();
170 			for (String v: resp.getHeaders(name)) {
171     			l.add(v);
172     		}
173     		headers.put(name, l);
174     	}
175     	return headers;
176 	}

178 	protected static Map<String, List<String>> getHeaders(Request req) {
179     	Map<String, List<String>> headers = new TreeMap<>();
180     	for (Enumeration<String> h = req.getHeaderNames(); h.hasMoreElements(); ) {
181     		String name = h.nextElement();
182 			List<String> l = new ArrayList<>();
183     		for (Enumeration<String> v = req.getHeaders(name); v.hasMoreElements(); ) {
184     			l.add(v.nextElement());
185     		}
186     		headers.put(name, l);
187     	}
188     	return headers;
189 	}

192      * This method is to deal with invalid headers, which IZ Gateway 1.X versions simply ignored.
193      * @param req The request to check and adjust if needed.
194      */

196 	protected void fixHeaders(Request req) {
197 		fixHeader(req, HttpHeaders.ACCEPT, t -> Strings.CS.contains(t, "/"), "*/*"); 
198 		fixHeader(req, HttpHeaders.ACCEPT_CHARSET, t-> "UTF-8".equalsIgnoreCase(t) || checkCharset(t), "utf-8"); 
199 	}

201 	/**
202 	 * Verify that a header on the request is valid, and replace it with a default value if not.
203 	 * @param req	The request to check
204 	 * @param header	The name of the header to verify
205 	 * @param test	The test to perform for validation
206 	 * @param replacement	The replacement text to substitute or null to simply remove the header.
207 	 */
208 	protected void fixHeader(Request req, String header, Predicate<String> test, String replacement) {
209 		Enumeration<String> headers = req.getHeaders(header);
210 		while (headers.hasMoreElements()) {
211 			String value = headers.nextElement();
212 			if (!test.test(value)) {
213 				// A bogus header value was found.
214 				req.getCoyoteRequest().getMimeHeaders().removeHeader(header);
215 				if (replacement != null && !replacement.isEmpty()) {
216 					req.getCoyoteRequest().getMimeHeaders().addValue(header).setString(replacement);
217 				}
218 				return;
219 			}
220 		}
221 	}

237 	protected void setMdcValues(Request req, String eventId, String commonName) {
238 		// Put variables into thread local storage so that threaded events can get to them
239         MDC.put(EventId.EVENTID_KEY, eventId);
240         MDC.put(COMMON_NAME, commonName);
241         MDC.put(SESSION_ID, req.getSession().getId());
242         MDC.put(REQUEST_URI, req.getRequestURI());
243         MDC.put(METHOD, req.getMethod());
244         MDC.put(IP_ADDRESS, req.getRemoteAddr());
245 	}
246 	
247 	protected void clearMdcValues() {
248         // Remove Added MDC Keys
249         MDC.remove(EventId.EVENTID_KEY);
250         MDC.remove(COMMON_NAME);
251         MDC.remove(SESSION_ID);
252         MDC.remove(REQUEST_URI);
253         MDC.remove(METHOD);
254         MDC.remove(IP_ADDRESS);
255     }

257     protected SourceInfo setSourceInfoValues(Request req, TransactionData t) {
258 
259         SourceInfo source = t.getSource();
260         source.setCipherSuite((String) req.getAttribute(Globals.CIPHER_SUITE_ATTR));
261         source.setHost(req.getRemoteHost());
262         source.setIpAddress(
263         	StringUtils.substringBefore(
264     			StringUtils.defaultIfEmpty(
265     				req.getHeader(X_FORWARDED_FOR), 
266     				req.getRemoteAddr()
267     			),
268     			","
269         	).trim()
270         );
271         return source;
272     }

274 	public Event getEvent(HttpSession sess) {
275         String sessionId = sess.getId();
276         LoggingValveEvent event = map.get(sessionId);
277 
278         if (event != null && --event.refs == 0) {
279             map.remove(sessionId);
280         }
281 
282         // Eventually, events will age out
283         if (map.size() > 20) {
284             // Purge the cache of values that are older than MAX_AGE
285             long time = System.currentTimeMillis() - MAX_AGE;
286             Iterator<Entry<String, LoggingValveEvent>> it = map.entrySet().iterator();
287             while (it.hasNext()) {
288                 Entry<String, LoggingValveEvent> e = it.next();
289                 if (e.getValue().getDate().getTime() > time) {
290                     break;
291                 }
292                 it.remove();
293             }
294         }
295 
296         return event;
297     }

299     @Override
300     public String createEvent(SSLSession sess) {
301         LoggingValveEvent event = map.get(EventCreator.toHex(sess.getId()));
302 
303         if (event != null) {
304             event.date = new Date(); // reset the date
305             event.refs++;   // Bump the ref count
306         } else {
307             event = new LoggingValveEvent(TransactionData.getNextEventId(), new Date());
308             map.put(EventCreator.toHex(sess.getId()), event);
309         }
310 
311         return event.getId();
312     }
