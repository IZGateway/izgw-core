16  /**
17   * DateUtil arose around frustration every time a date needed to be parsed with varying potential precision or formats.
18   * It won't parse all known date formats, but it will generally get anything that would be expected, and has some useful
19   * shortcuts.
20   * 
21   * NOW = the current date and time.
22   * TODAY = Today at midnight.
23   * TODAY 09:00 AM or 9:00 AM today mean today at 09:00 AM.
24   * 
25   * Other keywords work including YESTERDAY, TOMORROW, and the Day of the Week (which always corresponds to today or a day in the future).
26   * 
27   */
28  public class DateUtil {

31      private static final String[] AMPM = { "AM", "PM" };

33      	Month, Day, Digits, Punct, AmPm, Unknown, Empty, Year, Hour, Minute, Second, Millis, TZ

66  	    	cleanupHours();
67  	    	Calendar cal = new GregorianCalendar();
68  	    	setDefaultDateValues(cal);
69  	    	
70  	    	if (tz != null) {
71  	    		cal.setTimeZone(tz);
72  	    	}
73  
74  	    	cal.set(Calendar.YEAR, year);
75  	    	cal.set(Calendar.MONTH, month);
76  	    	cal.set(Calendar.DAY_OF_MONTH, day);
77  	    	cal.set(Calendar.HOUR_OF_DAY, hour >= 0 ? hour : 0);
78  			cal.set(Calendar.MINUTE, minutes >= 0 ? minutes : 0);
79  	    	cal.set(Calendar.SECOND, seconds >= 0 ? seconds : 0);
80  	    	cal.set(Calendar.MILLISECOND, nanos >= 0 ? nanos / 1000 : 0);
81  
82  	    	if (dow != 0 && dow != cal.get(Calendar.DAY_OF_WEEK)) {
83  	    		// Day of Week doesn't match
84  	    	}
85  	    	return cal;
86  		}

88  			// Assume current date time for any missing date components

91  	    		if (month == -1) {
92  	    			month = cal.get(Calendar.MONTH);
93  	    			if (day == -1) {
94  	    				day = cal.get(Calendar.DATE);
95  	    			}

99  		private void cleanupHours() {
100 			if (hour > 0 && hour < 12 && ampm == 1) {
101 	    		hour += 12;
102 	    	} else if (hour == 12 && ampm == 0) {
103 	    		hour = 0;
104 	    	} else if (hour > 12 && ampm == 1) {
105 	    		// We have an AMPM Indicator, and an hour greater than 12
106 	    	}
107 		}

109 	public static Date parseDate(String date) throws ParseException {
110     	// Simple forms for computers. 
111     	Date d = simpleParse(date);
112     	if (d != null) {
113     		return d;
114     	}
115     	
116     	// Case doesn't matter
117     	date = date.toUpperCase();
118     	List<String> parts = new ArrayList<>();
119     	List<PartType> partTypes = new ArrayList<>();
120     	List<Integer> offsets = new ArrayList<>();
121     	
122     	parseDatePass1(date, parts, partTypes, offsets);
123     	ParseState state = new ParseState();
124 
125     	// Second scan, determine content of alpha and digit strings
126     	for (int i = 0; i < parts.size(); i++) {
127         	int offset = offsets.get(i);
128     		try {
129 	    		PartType type = partTypes.get(i);
130 	    		String s = parts.get(i).toUpperCase();
131 	    		if (type == PartType.Unknown) {
132 	    			parseUnknownDateParts(state, offset, s);
133 	    		} else if (type == PartType.Digits) {
134 	    			type = parseDateDigits(parts, partTypes, state, i, s);
135 					partTypes.set(i, type);
136 	    		}
137     		} catch (Exception e) {
138     			throw new ParseException(e.getMessage(), offset);
139     		}
140     	}
141     	int[] counts = new int[PartType.values().length];
142     	checkForDuplication(partTypes, counts);
143     		
144     	return state.getCalendar().getTime();
145     }

147 	private static void parseUnknownDateParts(ParseState state, int offset, String s) throws ParseException {
148 		int pos;
149 		if ((pos = matchesAny(s, MONTHS, 3)) >= 0) {
150 			state.month = pos;
151 		} else if ((pos = matchesAny(s, AMPM, 1)) >= 0) {
152 			state.ampm = pos;
153 		} else if (isTimeZone(s)) {
154 			state.tz = TimeZone.getTimeZone(s);
155 		} else {
156 			Calendar cal = parseDateName(state, offset, s);
157 			if (cal != null) {
158 				state.setFromCalendar(cal);
159 			}
160 		}
161 	}

172 	private static PartType parseDateDigits(List<String> parts, List<PartType> partTypes, ParseState state, int i,
173 			String s) {
174 		PartType type;
175 		type = getDigitPartType(parts, partTypes, i);
176 		switch (type) {
177 		case Year:
178 			if (s.length() < 3) {
179 				state.year = 2000 + Integer.parseInt(s);
180 			} else {
181 				state.year = Integer.parseInt(s);
182 			}
183 			break;
184 		case Month:
185 			state.month = Integer.parseInt(s);
186 			break;
187 		case Day:
188 			state.day = Integer.parseInt(s);
189 			break;
190 		case Hour:
191 			state.hour = Integer.parseInt(s);
192 			break;
193 		case Minute:
194 			state.minutes = Integer.parseInt(s);
195 			break;
196 		case Second:
197 			state.seconds = Integer.parseInt(s);
198 			break;
199 		case Millis:
200 			state.nanos = Integer.parseInt((s + "000000").substring(0, 6));
201 			break;
202 		case TZ:
203 			parseTimeZone(parts, state, i, s);
204 			break;
205 		default:
206 			type = parseYear(state, s, type); 
207 		}
208 		return type;
209 	}

211 	private static PartType parseYear(ParseState state, String s, PartType type) {
212 		int value = Integer.parseInt(s);
213 		// If still unknown but greater than 60, likely a year
214 		if (value > 60) {
215 			type = PartType.Year;
216 			if (s.length() < 3) {
217 				state.year = 2000 + value;
218 			} else {
219 				state.year = value;
220 			}
221 		}
222 		return type;
223 	}

225 	private static void parseTimeZone(List<String> parts, ParseState state, int i, String s) {
226 		if (i + 1 < parts.size()) {
227 			if (":".equals(parts.get(i+1))) {
228 				String nextPart = i + 2 < parts.size() ? parts.get(i+2) : null;
229 				if (nextPart != null) {
230 					s += ":" + nextPart;
231 				}
232 				s = parts.get(i - 1) + s;
233 				state.tz = TimeZone.getTimeZone("GMT" + s);
234 			}
235 		} else if (s.length() > 2) {
236 			s = s.substring(0, 2) + ":" + s.substring(2);
237 			state.tz = TimeZone.getTimeZone("GMT" + s);
238 		}
239 	}

241 	private static Calendar parseDateName(ParseState state, int offset, String s) throws ParseException {
242 		int pos;
243 		Calendar cal = Calendar.getInstance();
244 		switch (s) {
245 		case "Z":
246 			state.tz = TimeZone.getTimeZone("GMT");
247 			cal = null;
248 			break;
249 		case "NOW":
250 			break;
251 		case "TODAY":
252 			// If the hour hasn't been set, set time to midnight, otherwise use existing time that was set
253 			if (state.hour < 0) setToMidnight(cal);  
254 			break;
255 		case "YESTERDAY":
256 			cal.add(Calendar.DAY_OF_MONTH, -1);
257 			if (state.hour < 0) setToMidnight(cal);
258 			break;
259 		case "TOMORROW":
260 			cal.add(Calendar.DAY_OF_MONTH, 1);
261 			if (state.hour < 0) setToMidnight(cal);
262 			break;
263 		default:
264 			// MONDAY, TUESDAY, et cetera, in this context means "this Monday", the nearest day (including today).
265 			if ((pos = matchesAny(s, DAYS, 3)) >= 0) {
266 				if (state.hour < 0) setToMidnight(cal);
267 				state.dow = pos + 1;
268 				int add = state.dow - (cal.get(Calendar.DAY_OF_WEEK) + 1);
269 				if (add < 0) {
270 					add += DAYS.length;
271 				}
272 				cal.add(Calendar.DAY_OF_MONTH, add);
273 				break;
274 			} 
275 			throw new ParseException("Unrecognized date part: " + s, offset);
276 		}
277 		return cal;
278 	}

280 	private static boolean isTimeZone(String s) {
281 		// Respond from most to least common cases.
282 		if (ZoneId.getAvailableZoneIds().contains(s)) {
283 			return true;
284 		}
285 		if ("Z".equals(s) || s.startsWith("+") || s.startsWith("-")) {
286 			return true;
287 		}
288 		return s.startsWith("UT") || s.startsWith("GMT") || s.startsWith("UTC");
289 	}
290 	private static void setToMidnight(Calendar cal) {
291 		cal.set(Calendar.HOUR, 0);
292 		cal.set(Calendar.MINUTE, 0);
293 		cal.set(Calendar.SECOND, 0);
294 		cal.set(Calendar.MILLISECOND, 0);
295 	}

303     public static class SimpleParseState {
304 		String front = null;
305 		String back = "";
306 		String zone = "";
307     }
308     private static Date simpleParse(String date) throws ParseException {
309 		String s = date.trim().replaceAll("[-:Tt]","");
310 		if (!s.matches("\\d+(\\.\\d+)?(Z|[+-][01]\\d([0-4]\\d)?)?")) {
311 			return null;
312 		}
313 		SimpleParseState state = new SimpleParseState();
314 		// It's almost all digits
315 		// Could be one of two forms YYYYMMDDHHMMss.SSS[+-]ZZZZ, or Unix Epoch
316 		if (StringUtils.containsAny(s, '+', '-', '.', 'Z')) {
317 			parseSimpleLongForm(s, state);
318 		} else {
319 			state.front = s;
320 			// Looks like an epoch date 
321 			long l = Long.parseLong(s);
322 			if (l > 30000000l && l < 399999999999l) {
323 				return new Date(l);
324 			}
325 		}
326 		if (state.front.length() > SIMPLE_FORMAT.length()) {
327 			// Should bail on this.
328 		}
329 		String format = SIMPLE_FORMAT.substring(0, Math.min(state.front.length(), SIMPLE_FORMAT.length()));
330 		if (state.back.length() > 0) {
331 			format += "." + StringUtils.repeat('S', state.back.length());
332 		}
333 		if (state.zone != null && state.zone.length() != 0) {
334 			format += "X";
335 		}
336 		FastDateFormat fdf = FastDateFormat.getInstance(format);
337 		return fdf.parse(s);
338     }

340 	private static void parseSimpleLongForm(String s, SimpleParseState state) {
341 		// Guaranteed to be of the first form.
342 		state.front = StringUtils.substringBefore(s, ".");
343 		state.back = StringUtils.substringAfter(s, ".");
344 		state.zone = null;
345 		String h = state.back.length() == 0 ? state.front : state.back;
346 		
347 		for (int i = 0; i < h.length(); i++) {
348 			char c = h.charAt(i); 
349 			if (c == '-' || c == '+' || c == 'Z') {
350 				state.zone = h.substring(i);
351 				h = h.substring(0, i);
352 				break;
353 			}
354 		}
355 		if (state.back.length() == 0) {
356 			state.front = h;
357 		} else {
358 			state.back = h;
359 		}
360 	}

362 	private static PartType getDigitPartType(List<String> parts, List<PartType> partTypes, int i) {
363 		String s;
364 		
365 		if (i == 0) {
366 			if (i + 1 < parts.size()) {
367 				s = parts.get(i + 1);
368 				return parseDelimited(s);
369 			}
370 			return PartType.Unknown;
371 		}
372 		
373 		s = parts.get(i-1);
374 		// preceded by a colon it could be minutes or seconds
375 		if ("+-".contains(s)) {
376 			return PartType.TZ;
377 		} else if (":".equals(s)) {
378 			return parseMinutes(parts, i);
379 		} else if ("/\\".contains(s)) {
380 			return parseSlashedDate(parts, i);
381 		} else if ("-".contains(s)) {
382 			return parseHyphenatedDate(parts, partTypes, i);
383 		} else if (i + 1 < parts.size()) {
384 			return parseByFollowingDelimiter(parts, i);
385 		} 
386 		return PartType.Unknown;
387 	}

389 	private static PartType parseByFollowingDelimiter(List<String> parts, int i) {
390 		String s;
391 		s = parts.get(i+1);
392 		if (":".equals(s)) {
393 			// follows by a : but not preceded by it, must be Hour
394 			return PartType.Hour;
395 		} else if ("/\\".contains(s)) {
396 			// Followed by a / but not preceded by it
397 			if (Locale.getDefault().getCountry().equalsIgnoreCase("US")) {
398 				return PartType.Month;
399 			}
400 			return PartType.Day;
401 		}
402 		return PartType.Unknown;
403 	}

405 	private static PartType parseHyphenatedDate(List<String> parts, List<PartType> partTypes, int i) {
406 		String s;
407 		// preceded by a dash, it could be in YYYY-MM-DD form, so month or day, 
408 		// or it could be DD-MMM-YYYY or DD-MMM-YY form, so YEAR
409 		// Or ich, DD-MM-YYYY form
410 		if (i + 1 < parts.size() && "-".equals(parts.get(i+1))) {
411 			// Preceded and followed by a dash, gonna be Month
412 			return PartType.Month;
413 		}
414 		// Only preceded by a dash.  YYYY[-MM], or YYYY-MM[-DD] or DD-MM[-YYYY] or ich, DD-MM[-YY]
415 		s = parts.get(i);
416 		if (s.length() > 2 && s.length() < 4) {
417 			return PartType.Year;
418 		}
419 		// Only preceded by a dash.  YYYY[-MM], or YYYY-MM[-DD] or ich, DD-MM[-YY]
420 		if (i - 2 > 0) {
421 			switch (partTypes.get(i-2)) {
422 			case Year:
423 				return PartType.Month;
424 			case Month:
425 				return PartType.Day;
426 			default:
427 				break;
428 			}
429 		}
430 		return PartType.Unknown;
431 	}

433 	private static PartType parseSlashedDate(List<String> parts, int i) {
434 		// preceded by a slash it could be year, day or month
435 		if (i + 1 < parts.size() && parts.get(i+1).equals("/")) {
436 			// mm/dd/yyyy or dd/mm/yyyy
437 			if (Locale.getDefault().getCountry().equalsIgnoreCase("US")) {
438 				return PartType.Day;
439 			}
440 			return PartType.Month;
441 		}
442 		return PartType.Unknown;
443 	}

445 	private static PartType parseMinutes(List<String> parts, int i) {
446 		// if followed by a colon, it is definitely minutes.
447 		if (i + 1 < parts.size() && parts.get(i+1).equals(":")) {
448 			return PartType.Minute;
449 		}
450 		// If there are no other colons preceding, then it's likely hh:mm form.
451 		for (int j = i - 2; j > 0; j--) {
452 			if (":".equals(parts.get(j))) {
453 				return PartType.Second;	// There were two such animals, it's seconds
454 			}
455 		}
456 		return PartType.Minute;	// Assume hh:mm form
457 	}

459 	private static PartType parseDelimited(String s) {
460 		switch (s) {
461 		case ":":
462 			return PartType.Hour;
463 		case "/", "\\":
464 			return Locale.getDefault().getCountry().equals("US") ? PartType.Month : PartType.Day;
465 		case "-":
466 			if (s.length() >= 3 && s.length() <= 4) {
467 				return PartType.Year;
468 			}
469 			return PartType.Unknown;
470 		default:
471 			return PartType.Unknown;
472 		}
473 	}

475 	private static int matchesAny(String s, String[] list, int minLength) {
476 		for (int pos = 0; pos < list.length; pos++) {
477 			String m = list[pos];
478 			if (m.startsWith(s) && s.length() >= minLength) {
479 				return pos;
480 			}
481 		}
482 		return -1;
483 	}

485 	private static void parseDatePass1(String date, List<String> parts, List<PartType> partTypes, List<Integer> offsets) {
486     	PartType type = PartType.Empty;
487     	StringBuilder b = new StringBuilder();
488     	int offset = 0;
489 		// First scan, break into strings of digits, alpha, and punctuation
490     	for (int i = 0; i < date.length(); i++) {
491     		char c = date.charAt(i);
492     		
493     		if (Character.isDigit(c)) {
494     			switch (type) {
495     			case Unknown:
496     				parts.add(b.toString());
497     				offsets.add(offset);
498     				offset = i;
499     				b.setLength(0);
500     				// Fall through
501     			case Empty:
502     				type = PartType.Digits;
503     				partTypes.add(type);
504     				// Fall through
505     			case Digits:
506     				b.append(c);
507     				break;
508     			default:
509     				break;
510     			}
511     		} else if (Character.isAlphabetic(c)) {
512     			switch (type) {
513     			case Digits:
514     				parts.add(b.toString());
515     				offsets.add(offset);
516     				offset = i;
517     				b.setLength(0);
518     				// Fall through
519     			case Empty:
520     				type = PartType.Unknown;
521     				partTypes.add(type);
522     				// Fall through
523     			case Unknown:
524     				b.append(c);
525     				break;
526     			default:
527     				break;
528     			}
529     			
530     		} else {
531     			if (type != PartType.Empty) {
532     				parts.add(b.toString());
533     				offsets.add(offset);
534     				offset = i;
535     				b.setLength(0);
536     			}
537 				partTypes.add(PartType.Punct);
538 				parts.add(b.append(c).toString());
539 				offsets.add(offset);
540 				offset = i;
541 				b.setLength(0);
542 				type = PartType.Empty;
543     		}
544     	}
545     	if (b.length() != 0) {
546     		offsets.add(offset);
547     		parts.add(b.toString());
548     	}
549 	}
