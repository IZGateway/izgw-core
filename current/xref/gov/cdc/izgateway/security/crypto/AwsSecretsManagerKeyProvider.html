12  /**
13   * AWS Secrets Manager implementation of the KeyProvider interface.
14   * <p>
15   * This provider retrieves encryption keys from AWS Secrets Manager service.
16   * The secret is expected to be stored as a UTF-8 encoded string representing
17   * a 32-byte encryption key suitable for AES-256 encryption.
18   * </p>
19   * <p>
20   * Configuration is handled through environment variables:
21   * <ul>
22   *   <li>{@code DB_ENCRYPTION_KEYNAME} - The name of the secret in AWS Secrets Manager</li>
23   *   <li>{@code AWS_REGION} - The AWS region where the secret is stored (defaults to us-east-1)</li>
24   * </ul>
25   * </p>
26   *
27   * @author CDC IZ Gateway Team
28   * @since 1.0
29   */
30  class AwsSecretsManagerKeyProvider extends KeyProviderBase implements KeyProvider {
31      private static final String DB_ENCRYPTION_KEYNAME = "DB_ENCRYPTION_KEYNAME";
32  
33      /**
34       * Constructor that loads the current and previous keys into the key history using AWS Secret tags.
35       */
36      AwsSecretsManagerKeyProvider() {
37          String secretName = getEncryptionKeySecretName();
38          if (StringUtils.isEmpty(secretName)) {
39              return;
40          }
41          try (SecretsManagerClient client = SecretsManagerClient.builder().build()) {
42              // Get secret description (includes tags and version ids)
43              var describeRequest = software.amazon.awssdk.services.secretsmanager.model.DescribeSecretRequest.builder()
44                      .secretId(secretName).build();
45              var describeResponse = client.describeSecret(describeRequest);
46              Map<String, List<String>> versionStages = describeResponse.versionIdsToStages();
47              String currentVersionId = null;
48              String previousVersionId = null;
49              for (Map.Entry<String, List<String>> entry : versionStages.entrySet()) {
50                  if (entry.getValue().contains("AWSCURRENT")) {
51                      currentVersionId = entry.getKey();
52                  } else if (entry.getValue().contains("AWSPREVIOUS")) {
53                  	// We must load the previous key in case CC is not yet finished rolling out the new key
54                      previousVersionId = entry.getKey();
55                  }
56              }
57              if (currentVersionId != null) {
58                  byte[] currentKey = loadKeyVersion(client, secretName, currentVersionId);
59                  if (currentKey != null) addKeyToHistory(currentKey);
60              }
61              if (previousVersionId != null) {
62                  byte[] previousKey = loadKeyVersion(client, secretName, previousVersionId);
63                  if (previousKey != null) addKeyToHistory(previousKey);
64              }
65          } catch (SdkClientException e) {
66  			throw new ServiceConfigurationError("Failed to load encryption key from AWS Secrets Manager", e);
67  		}
68      }
69  
70      private byte[] loadKeyVersion(SecretsManagerClient client, String secretName, String versionId) {
71          try {
72          	GetSecretValueRequest req = GetSecretValueRequest.builder().secretId(secretName).versionId(versionId).build();
73          	GetSecretValueResponse resp = client.getSecretValue(req);
74              String secret = resp.secretString();
75  			return checkSecret(secret);
76          } catch (Exception ignored) {}
77          return null;
78      }
79  
80      /**
81       * Loads the encryption key from AWS Secrets Manager.
82       *
83       * @return a 32-byte array containing the encryption key
84       * @throws IllegalArgumentException if the secret name environment variable is not set,
85       *                                  the secret value is empty, or the key length is not 32 bytes
86       * @throws CryptoException if there's an error communicating with AWS Secrets Manager
87       */
88      @Override
89      public byte[] loadKey() throws CryptoException {
90          String secretName = getEncryptionKeySecretName();
91          if (StringUtils.isEmpty(secretName)) {
92              throw new IllegalArgumentException(DB_ENCRYPTION_KEYNAME + " environment variable is not set.");
93          }
94  
95          try (SecretsManagerClient client = SecretsManagerClient.builder().build()) {
96          	return loadKeyVersion(client, secretName, null);
97          } catch (SdkClientException e) {
98              throw new CryptoException("Failed to load encryption key from AWS Secrets Manager", e);
99          }
100     }
101 
102 	private byte[] checkSecret(String secret) {
103 		if (StringUtils.isEmpty(secret)) {
104 		    throw new IllegalArgumentException("Secret value is empty.");
105 		}
106 	    byte[] decoded = Hex.decode(secret.trim());
107 	    if (decoded.length != 32) {
108 	        throw new IllegalArgumentException("Secret key length is invalid. Expected 32 bytes, got " + decoded.length + " bytes.");
109 	    }
110         return decoded;
111 	}
112 
113     private String getEncryptionKeySecretName() {
114         return System.getenv().getOrDefault(DB_ENCRYPTION_KEYNAME, "izgw-dev-password-encryption-key");
115     }
