3   import java.util.LinkedHashSet;
4   import java.util.List;

7   class KeyProviderBase {
8       private final Set<ByteArrayWrapper> keyHistory = new LinkedHashSet<>();
9   
10      public List<byte[]> getAllKeys() {
11          synchronized (TestKeyProvider.class) {
12              return keyHistory.stream()
13                      .map(ByteArrayWrapper::getData)
14                      .toList();
15          }
16      }
17  
18      public void addKeyToHistory(byte[] keyBytes) {
19          synchronized (TestKeyProvider.class) {
20              keyHistory.add(new ByteArrayWrapper(keyBytes));
21          }
22      }
23  
24      // Check if key exists
25      public boolean keyExists(byte[] keyBytes) {
26          synchronized (CryptoSupport.class) {
27              return keyHistory.contains(new ByteArrayWrapper(keyBytes));
28          }
29      }
