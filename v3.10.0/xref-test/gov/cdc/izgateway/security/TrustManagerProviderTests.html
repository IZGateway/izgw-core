29  /**
30   * Regression coverage for IGDD-3582: {@link TrustManagerProvider} must not freeze the trust
31   * manager it hands out at startup -- a later keystore/truststore rotation (pushed via
32   * {@link ServerSslBundleReloader}) must be reflected here too, via the
33   * {@link SslBundles#addBundleUpdateHandler} registration in
34   * {@link TrustManagerProvider#initializeTrustManager()}.
35   */
36  @ExtendWith(MockitoExtension.class)
37  class TrustManagerProviderTests {
38  
39  	private static final String BUNDLE_NAME = "izgw-server";
40  
41  	@BeforeEach
42  	void registerBcFipsProviders() {
43  		CryptoSupport.initialize();
44  	}
45  
46  	private static KeyStore emptyKeyStore() throws Exception {
47  		KeyStore ks = KeyStore.getInstance(KeyStore.getDefaultType());
48  		ks.load(null, null);
49  		return ks;
50  	}
51  
52  	private static SslBundle bundleWithTrustStore(KeyStore trustStore) {
53  		return SslBundle.of(SslStoreBundle.of(null, null, trustStore));
54  	}
55  
56  	private TrustManagerProvider newBundleBackedProvider(SslBundles sslBundles) {
57  		TrustManagerProvider provider = new TrustManagerProvider(sslBundles);
58  		ReflectionTestUtils.setField(provider, "sslBundleName", BUNDLE_NAME);
59  		ReflectionTestUtils.invokeMethod(provider, "initializeTrustManager");
60  		return provider;
61  	}
62  
63  	@Test
64  	void loadsTrustManagerFromBundleOnInit() throws Exception {
65  		SslBundles sslBundles = mock(SslBundles.class);
66  		KeyStore initial = emptyKeyStore();
67  		when(sslBundles.getBundle(BUNDLE_NAME)).thenReturn(bundleWithTrustStore(initial));
68  
69  		TrustManagerProvider provider = newBundleBackedProvider(sslBundles);
70  
71  		assertSame(initial, provider.getTrustStore());
72  		assertNotNull(provider.getServerTrustManager());
73  	}
74  
75  	@Test
76  	@SuppressWarnings("unchecked")
77  	void refreshesTrustManagerWhenBundleIsUpdated() throws Exception {
78  		SslBundles sslBundles = mock(SslBundles.class);
79  		KeyStore initial = emptyKeyStore();
80  		when(sslBundles.getBundle(BUNDLE_NAME)).thenReturn(bundleWithTrustStore(initial));
81  
82  		TrustManagerProvider provider = newBundleBackedProvider(sslBundles);
83  		var initialTrustManager = provider.getServerTrustManager();
84  
85  		ArgumentCaptor<Consumer<SslBundle>> handlerCaptor = ArgumentCaptor.forClass(Consumer.class);
86  		verify(sslBundles).addBundleUpdateHandler(eq(BUNDLE_NAME), handlerCaptor.capture());
87  
88  		KeyStore rotated = emptyKeyStore();
89  		handlerCaptor.getValue().accept(bundleWithTrustStore(rotated));
90  
91  		assertSame(rotated, provider.getTrustStore());
92  		assertNotSame(initialTrustManager, provider.getServerTrustManager());
93  	}
94  
95  	@Test
96  	void doesNotRegisterAnUpdateHandlerWhenNoBundleIsConfigured() {
97  		SslBundles sslBundles = mock(SslBundles.class);
98  		TrustManagerProvider provider = new TrustManagerProvider(sslBundles);
99  		ReflectionTestUtils.setField(provider, "sslBundleName", "");
100 		ReflectionTestUtils.setField(provider, "trustStorePath", "src/test/resources/does-not-exist.bcfks");
101 		ReflectionTestUtils.setField(provider, "trustStorePassword", "whatever");
102 		ReflectionTestUtils.setField(provider, "trustStoreType", "BCFKS");
103 		ReflectionTestUtils.setField(provider, "trustStoreProvider", "BCFIPS");
104 
105 		org.junit.jupiter.api.Assertions.assertThrows(java.util.ServiceConfigurationError.class,
106 				() -> ReflectionTestUtils.invokeMethod(provider, "initializeTrustManager"));
107 
108 		verify(sslBundles, never()).addBundleUpdateHandler(anyString(), any());
109 	}
