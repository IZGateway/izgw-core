1   package gov.cdc.izgateway.logging.event;
2   
3   import java.util.concurrent.atomic.AtomicLong;
4   import java.util.function.LongUnaryOperator;
5   
6   import org.apache.commons.lang3.StringUtils;
7   
8   import gov.cdc.izgateway.utils.SystemUtils;
9   
10  @SuppressWarnings("serial")
11  public class EventId extends AtomicLong { // NOSONAR Singleton pattern intended
12  	// These values need to be initialized in this order.
13      private static final String PREFIX = computeEventIdPrefix();
14      private static final LongUnaryOperator NEXT_UPDATE_OP = value -> ((value < Long.MAX_VALUE) ? ++value : 1);
15      /**
16       * The default instance of an eventId to obtain new values from.
17       */
18      public static final EventId INSTANCE = new EventId();
19      /**
20       * The default instance of an eventId for the Status Checker main thread.
21       */
22      public static final String DEFAULT_TX_ID = INSTANCE.getNext();
23      /**
24       * The keyword used to get the eventId from the MDC
25       */
26      public static final String EVENTID_KEY = "eventId";
27  	private static boolean obfuscate = true;	// Set to false to use server ip addresses when available as first part of eventId
28  
29      private EventId() {
30      }
31      /**
32       * Compute a prefix to prepend to event identifiers to make them unique 
33       * @return	A unique prefix.
34       */
35      private static String computeEventIdPrefix() {
36  		String hostname = SystemUtils.getHostname();
37  		
38  		if (!obfuscate && hostname.contains(".")) {
39  			// Tie it back to host IP if we can, AWS Names are often in the form ip-10-9-97-2.ec2.internal
40  			String[] parts = StringUtils.substringBefore(hostname, ".").split("-");
41  			StringBuilder prefix = new StringBuilder();
42  			for (String part: parts) {
43  				if (StringUtils.isNumeric(part)) {
44  					int i = Integer.parseInt(part);
45  					prefix.append(String.format("%03d", i));   // Which would become 010009097002
46  				}
47  			}
48  			return StringUtils.stripStart(prefix.toString(), "0");
49  		}
50  		return Integer.toUnsignedString(hostname.hashCode())+"000";
51  	}
52      
53      public String getNext() {
54          return PREFIX + "." + updateAndGet(NEXT_UPDATE_OP);
55      }
56      
57      public static String getPrefix() {
58      	return PREFIX;
59      }
60  }
