161 	/**
162 	 * Send a submitSingleMessage request.  This method sends a submitSingleMessage request to a destination endpoint,
163 	 * retrying if both possible and reasonable to retry.  Certain failures such as a TLS error take longer to "fix",
164 	 * are not transient in nature, and thus are not retried.  Updates the status of an endpoint on success or failure,
165 	 * and on a failure of a non-transient sort or repeated transient failure, sets the circuit breaker.
166 	 *   
167 	 * @param dest	The destination to send it to
168 	 * @param submitSingleMessage	The message to send
169 	 * @return	The response
170 	 * @throws Fault	If an error occurred while sending or unmarshalling the response
171 	 */
172 	public SubmitSingleMessageResponse sendSubmitSingleMessage(
173 		IDestination dest,
174 		SubmitSingleMessageRequest submitSingleMessage
175 	) throws Fault {
176 
177 		IEndpointStatus status = checkDestinationStatus(dest);
178 		SubmitSingleMessageRequest toBeSent = 
179 			new SubmitSingleMessageRequest(submitSingleMessage, getSchemaToUse(dest), true);
180 		// Clear the hub header, we don't forward that.
181 		if (!dest.isHub()) {
182 			toBeSent.getHubHeader().clear();
183 		}
184 		copyCredentials(toBeSent, dest);
185 		int retryCount = 0;
186 		while (true) {
187 			try {
188 				SubmitSingleMessageResponse responseFromClient = sendMessage(SubmitSingleMessageResponse.class, dest, toBeSent);
189 				SubmitSingleMessageResponse toBeReturned = new SubmitSingleMessageResponse(responseFromClient, submitSingleMessage.getSchema(), true);
190 				toBeReturned.updateAction(true);  // Now a Hub Response
191 				RequestContext.getTransactionData().setRetries(retryCount);
192 				getStatusChecker().updateStatus(status, dest, null);
193 				return toBeReturned;
194 			} catch (Fault f) {
195 				if (!canRetry(++retryCount, f)) {
196 					RequestContext.getTransactionData().setProcessError(f);
197 					RequestContext.getTransactionData().setRetries(retryCount);
198 					getStatusChecker().updateStatus(status, dest, f);
199 					throw f;
200 				}
201 			} catch (RuntimeException e) {
202 				// A RuntimeException escaping sendMessage would bypass setProcessError, causing
203 				// logstash-logback-encoder to silently drop the transactionData log entry.
204 					RequestContext.getTransactionData().setRetries(retryCount + 1);
205 				throw e;
206 			}
207 		}
208 	}
209 
210 	private boolean canRetry(int retryCount, Fault f) {
211 		if (!f.isRetryable() || f.getCause() instanceof XMLStreamException) {
212 			// This is not a retry-able failure.
213 			return false;
214 		}
215 		
216 		// Retry only up to the max retries
217 		return retryCount <= senderConfig.getMaxRetries();
218 	}

234 	/**
235 	 * Check the status of a destination on an inbound request
236 	 * @param dest	The destination to check.
237 	 * @return	The current status of the destination
238 	 * @throws DestinationConnectionFault	If the destination is under maintenance or has had its circuit breaker thrown
239 	 * 
240 	 * NOTE: Administrative users skip maintenance and circuit breaker thrown checks  
241 	 */
242 	private IEndpointStatus checkDestinationStatus(IDestination dest) throws DestinationConnectionFault {
243 		// Check for destination under maintenance
244 		if (dest.isUnderMaintenance() && userIsNotAdmin()) {
245 			throw DestinationConnectionFault.underMaintenance(dest);
246 		}
247 		
248 		// Check the circuit breaker
249 		IEndpointStatus status = statusService.getEndpointStatus(dest);
250 		// Skip endpoints exempt from status checking (b/c they cannot reset) 
251 		String destId = dest.getDestId();
252 		if (status.isCircuitBreakerThrown() && userIsNotAdmin() && !statusChecker.isExempt(destId)) {
253 			throw DestinationConnectionFault.circuitBreakerThrown(dest, status.getDetail());
254 		}
255 		return status;
256 	}

258 	private boolean userIsNotAdmin() {
259 		return !RequestContext.getRoles().contains(Roles.ADMIN) ||			// User is not ADMIN 
260 			RequestContext.getRoles().contains(Roles.NOT_ADMIN_HEADER);		// Admin user requested to be treated as non-admin for testing in header
261 	}

273 	/**
274 	 * Sends a connectivity test request to the destination.
275 	 * 
276 	 * Connectivity tests are used to verify connectivity between IZ Gateway and the destination endpoint.  They originated
277 	 * in the CDC WSDL but IZ Gateway can also route a connectivity test from a sender to a specific destination to verify
278 	 * that it is alive.  Not all jurisdictions are able to support the connectivity test, either due to security requirements
279 	 * or technical limitations.  If you get back a response, the jurisdiction endpoint is at least alive and listening.  If
280 	 * you do not, it MAY be due to jurisdiction limitations on supporting this capability.  All new onboarded jurisdictions
281 	 * MUST support this capability.
282 	 * 
283 	 * @param dest	The destination
284 	 * @param connectivityTest	The connectivity test message
285 	 * @return	The response (which should contain the same message body back)
286 	 * @throws Fault	If an error occurred.
287 	 */
288 	public ConnectivityTestResponse sendConnectivityTest(IDestination dest, ConnectivityTestRequest connectivityTest)
289 			throws Fault {
290 		String schemaToUse = dest.is2011() ? SoapMessage.IIS2011_NS : SoapMessage.IIS2014_NS;
291 		ConnectivityTestRequest toBeSent = new ConnectivityTestRequest(connectivityTest, schemaToUse, true);
292 		toBeSent.getHubHeader().clear();
293 		ConnectivityTestResponse responseFromClient = sendMessage(ConnectivityTestResponse.class, dest, toBeSent);
294 		ConnectivityTestResponse toBeReturned = new ConnectivityTestResponse(responseFromClient, connectivityTest.getSchema(), false);
295 		toBeReturned.updateAction(connectivityTest.is2014Message());  // Update action to match request
296 		return toBeReturned;
297 	}

299 	/**
300 	 * Send a message and obtain a response of the specified type to the given destination 
301 	 * @param <T>	represents the type of the response
302 	 * @param clazz	The class of the expected response message
303 	 * @param dest	The destination to send the message to
304 	 * @param toBeSent	The message to send
305 	 * @return	The response
306 	 * @throws Fault	If a fault occurs during sending.
307 	 */
308 	public <T extends SoapMessage> T sendMessage(Class<T> clazz, IDestination dest, SoapMessage toBeSent)
309 			throws Fault {
310 		long started = 0;
311 		long readStarted = 0;
312 		HttpURLConnection con = null;
313 		OutputStream pos = null;
314 		URL location = getUrl(dest);
315 		T result = null;
316 		try { // NOSONAR try with resources not appropriate here
317 			started = System.currentTimeMillis();
318 			con = setupConnection(tlsSupport.getSNIEnabledConnection(location));
319 			MessageInfo messageInfo = new MessageInfo(toBeSent, EndpointType.CLIENT, Direction.OUTBOUND, isProduction);
320 			RequestContext.getTransactionData().getClientRequest().setWs_request_message(messageInfo);
321 			toBeSent.updateAction(dest.isHub());  // Sending to non-IZ Gateway endpoint
322 			String action = toBeSent.getWsaHeaders().getAction();
323 			con.setRequestProperty(HttpHeaders.CONTENT_TYPE, "application/soap+xml;charset=UTF-8;action=\"" + action + "\"");
324 			messageInfo.setHttpHeaders(con.getRequestProperties());
325 			if (preserveOutput) {
326 				pos = new PreservingOutputStream(con.getOutputStream(), FixedByteArrayOutputStream.DEFAULT_SIZE);
327 			} else {
328 				pos = con.getOutputStream();
329 			}
330 			con.connect();
331 			// If we got here, we are connected.
332 			RequestContext.getDestinationInfo().setConnected(true);
333 			// Now log the certificates, before doing anything else that could mess up the connection.
334 			logDestinationCertificates(con);	
335 			converter.write(toBeSent, pos);
336 			readStarted = System.currentTimeMillis();
337 			result = readResult(clazz, dest, con, started);
338 			result.respondingTo(toBeSent);
339 			// Save the response HttpHeader fields.
340 			messageInfo = new MessageInfo(result, EndpointType.CLIENT, Direction.INBOUND, isProduction);
341 			messageInfo.setHttpHeaders(con.getHeaderFields());
342 			RequestContext.getTransactionData().getClientResponse().setWs_response_message(messageInfo);
343 			return result;
344 		} catch (HttpMessageNotWritableException ex) {
345 			if (ex.getCause() instanceof CryptoException f) {
346 				throw SecurityFault.decryptionFailure(dest, f);
347 			}
348 			throw DestinationConnectionFault.writeError(dest, ex.getCause());
349 		} catch (ConnectException ex) {
350 			throw DestinationConnectionFault.connectError(dest, ex, System.currentTimeMillis() - started);
351 		} catch (SocketTimeoutException ex) {
352 			throw DestinationConnectionFault.timedOut(dest, ex, System.currentTimeMillis() - started);
353 		} catch (UnknownHostException ex) {
354 			throw DestinationConnectionFault.unknownHost(dest, ex);
355 		} catch (IOException ex) {
356 			throw DestinationConnectionFault.writeError(dest, ex);
357 		} finally {
358 			long finished = System.currentTimeMillis(); 
359 			if (result != null) {
360 				RequestContext.getTransactionData().getClientResponse().setWs_response_message(new MessageInfo(result, EndpointType.CLIENT, Direction.INBOUND, isProduction));
361 			}
362 			// Increment elapsed time here in case of retries.
363 			TransactionData tData = RequestContext.getTransactionData();
364 			tData.setElapsedTimeIIS(tData.getElapsedTimeIIS() + (finished - started));
365 			tData.setReadTimeIIS(tData.getReadTimeIIS() + (finished - readStarted));
366 		}
367 	}

369 	private HttpURLConnection setupConnection(
370 			HttpURLConnection con) throws ProtocolException {
371 		con.setRequestMethod(HttpMethod.POST.name());
372 		con.setDoInput(true);
373 		con.setDoOutput(true);
374 		con.setConnectTimeout((int) TimeUnit.SECONDS.toMillis(clientConfig.getConnectTimeout()));
375 		con.setReadTimeout((int) TimeUnit.SECONDS.toMillis(clientConfig.getReadTimeout()));
376 		con.setUseCaches(false);
377 		con.setAllowUserInteraction(false);
378 		con.setInstanceFollowRedirects(false);
379 		con.setRequestProperty(HttpHeaders.CONTENT_TYPE, clientConfig.getContentType());
380 		return con;
381 	}

383 	private <T extends SoapMessage> T readResult(Class<T> clazz, IDestination dest, HttpURLConnection con, long started)
384 			throws Fault {
385 
386 		int statusCode = -1;
387 		try {
388 			statusCode = con.getResponseCode();
389 			RequestContext.getDestinationInfo().setConnected(true);
390 		} catch (ConnectException ex) {
391 			throw DestinationConnectionFault.connectError(dest, ex, System.currentTimeMillis() - started);
392 		} catch (SocketTimeoutException ex) {
393 			throw DestinationConnectionFault.timedOut(dest, ex, System.currentTimeMillis() - started);
394 		} catch (IOException ex) {
395 			throw DestinationConnectionFault.readError(dest, ex);
396 		}
397 
398 		HttpUrlConnectionInputMessage m = null;
399 		InputStream body = null;
400 		Exception savedEx;
401 		try {
402 			SoapMessage result = null;
403 			// Mark the buffer so we can reread on error.
404 			m = new HttpUrlConnectionInputMessage(con, clientConfig.getMaxBufferSize());
405 			statusCode = m.getStatusCode();
406 			body = m.getBody();
407 			m.mark();
408 			EndPointInfo endPoint = RequestContext.getDestinationInfo();
409 			if (ACCEPTABLE_RESPONSE_CODES.contains(statusCode)) {
410 				result = converter.read(m, endPoint);
411 				if (result instanceof FaultMessage) {
412 					m.reset();
413 					// Record the IIS fault response as the client response before throwing
414 					RequestContext.getTransactionData().getClientResponse()
415 						.setWs_response_message(new MessageInfo(result, EndpointType.CLIENT, Direction.INBOUND, isProduction));
416 						// We received a fault response, so we were connected.
417 					if (endPoint instanceof DestinationInfo destinationInfo) {
418 						destinationInfo.setConnected(true);
419 					}
420 					throw HubClientFault.clientThrewFault(null, dest, statusCode, body, result, StringUtils.substringBefore(con.getURL().toString(), "?"));
421 				}
422 				return clazz.cast(result);
423 			} else {
424 				try (InputStream errStream = con.getErrorStream()) {
425 					String path = StringUtils.substringBefore(con.getURL().toString(), "?");
426 					throw processHttpError(dest, statusCode, errStream, path);
427 				}
428 			}
429 		} catch (ClassCastException ex) {
430 			savedEx = ex;
431 		} catch (IOException ex) {
432 			// There was an IO Exception reading the content
433 			// We'll call this a destination connection fault of some sort.
434 			throw DestinationConnectionFault.readError(dest, ex);
435 		} catch (HttpMessageNotReadableException ex) {
436 			if (ex.getCause() instanceof Fault f) {
437 				throw f;
438 			}
439 			savedEx = ex;
440 		}
441 		if (m != null) {
442 			m.reset();
443 		}
444 		// There can be no result here.
445 		throw HubClientFault.invalidMessage(savedEx, dest, statusCode, body);
446 	}

448 	private HubClientFault processHttpError(IDestination dest, int statusCode, InputStream err, String path) {
449 		String error = "";
450 		if (err != null) {
451 			try {
452 				error = IOUtils.toString(err, StandardCharsets.UTF_8);
453 			} catch (IOException e) {
454 				// We couldn't read the error stream, log it but otherwise
455 				// proceed normally.
456 			}
457 		}
458 		return HubClientFault.httpError(dest, statusCode, error, path);
459 	}

461 	/**
462 	 * Get the URL from the destination
463 	 * @param dest	The destination to get the URL for
464 	 * @return	A url properly configured for sending, and verified against security rules.
465 	 * @throws DestinationConnectionFault	If the URL is misconfigured.
466 	 * @throws SecurityFault	If the system and destination do not agree on the operating environment (e.g., prod, onboarding, dev) 
467 	 */
468 	URL getUrl(IDestination dest) throws DestinationConnectionFault, SecurityFault {
469 		
470 		String destUri = StringUtils.substringBefore(dest.getDestUri(), "?"); // Remove any query parameter from the path.
471 		String errorMsg = "Destination " + dest.getDestId();
472 
473 		if (dest.getDestTypeId() != SystemUtils.getDestType()) {
474 			throw SecurityFault.generalSecurity("Destination Environment Mismatch", 
475 					"Expected " + SystemUtils.getDestTypeAsString() + " for " + dest.getDestId() + " != " + dest.getDestType(), null);
476 		}
477 		
478 
479 		if (Strings.CS.startsWith(destUri, "http:")) {
480 			throw SecurityFault.generalSecurity("Destination Not Using HTTPS", destUri + " for " + dest.getDestId(), null);
481 		}
482 
483 		try {
484 			if (Strings.CS.startsWith(destUri, "https:")) {
485 				return new URI(destUri).toURL();
486 			}
487 			if (Strings.CS.startsWith(destUri, "/")) {
488 				return new URI(serverConfig.getProtocol(), null, "localhost", serverConfig.getPort(), null, null, null).resolve(destUri).toURL();
489 			}
490 		} catch (MalformedURLException | URISyntaxException | IllegalArgumentException e) {
491 			throw DestinationConnectionFault.configurationError(dest,
492 					errorMsg + " is not configured correctly", e);
493 		}
494 		throw DestinationConnectionFault.notHttps(dest,
495 				errorMsg + " is using an unknown or unsupported protocol: " + destUri, null);
496 	}

498 	URL getReportedUrl(IDestination dest, URL destUrl) throws DestinationConnectionFault {
499 		String destUri = dest.getDestUri();
500 		if (Strings.CS.startsWith(destUri, "/")) {
501 			try {
502 				return serverConfig.getBaseUrl().toURI().resolve(destUri).toURL();
503 			} catch (MalformedURLException | URISyntaxException | IllegalArgumentException e) {
504 				throw DestinationConnectionFault.configurationError(dest,
505 						"Destination " + dest.getDestId() + " is not configured correctly", e);
506 			}
507 		}
508 		return destUrl;
509 	}

512 	 * Capture information about the destination certificates for subsequent logging.
513 	 * 
514 	 * This method captures information about
515 	 * - whether or not a connection was made successfully 
516 	 * - the jurisdiction certificate used in the connection
517 	 * - the cipher suite selected for the connection.
518 	 * 
519 	 * @param con	The connection used to connect to the jurisdiction.
520 	 */
521 	private static void logDestinationCertificates(HttpURLConnection con) {
522 		DestinationInfo destination = RequestContext.getDestinationInfo();
523 		if (destination.isConnected() && con instanceof HttpsURLConnection conx) {
524 			try {
525 				X509Certificate[] certs = (X509Certificate[]) conx.getServerCertificates();
526 				destination.setCertificate(certs[0]);
527 				destination.setCipherSuite(conx.getCipherSuite());
528 			} catch (SSLPeerUnverifiedException | IllegalStateException ex) {
529 				// Ignore this.
530 			}
531 		}
