18  import java.net.URL;

23  public class RequestHelper {
24  
25  	private static final Logger LOGGER = LoggerFactory.getLogger(RequestHelper.class);
26  
27  	protected CertificateId getOcspRequestCertificateId(DigestCalculator digestCalc,
28  			X509CertificateHolder issuerCertHolder, X509Certificate cert)
29  			throws CertPathValidatorException {
30  		CertificateId ocspReqCertId;
31  
32  		try {
33  			ocspReqCertId = new CertificateId(digestCalc, issuerCertHolder, cert.getSerialNumber());
34  		} catch (OCSPException e) {
35  			throw new CertPathValidatorException("Unable to determine OCSP ID." + e.getMessage());
36  		}
37  		return ocspReqCertId;
38  	}
39  
40  	/*
41  	 * Create OCSP request with nonce extension to send to the responder url
42  	 * Certificate ID is generated for the verifying cert and send to the request
43  	 */
44  	protected OCSPReq createOCSPRequest(CertificateId ocspReqCertId, Extension nonceOcspReqExt) throws OCSPException {
45  
46  		OCSPReq request = null;
47  		try {
48  
49  			OCSPReqBuilder ocspGen = new OCSPReqBuilder();
50  			// add the certificate id to the request builder and build the request
51  			ocspGen.addRequest(ocspReqCertId);
52  			ocspGen.setRequestExtensions(new Extensions(nonceOcspReqExt));
53  
54  			request = ocspGen.build();
55  		} catch (OCSPException e) {
56  			throw new OCSPException("Error with generating OCSPRequest: " + e.getMessage());
57  		}
58  
59  		return request;
60  	}
61  
62  	/* send the query to ocsp responder server url with the ocsp request content */
63  
64  	protected OCSPResp queryOcspResponder(URL ocspResponderUrl, OCSPReq ocspReqContent,
65  			int connectTimeout, int readTimeout, Map<String, String> reqHeaders) throws IOException {
66  
67  		HttpURLConnection ocspResponderConn = ((HttpURLConnection) ocspResponderUrl.openConnection());
68  		LOGGER.debug("open connection to the ocsp server url {}", ocspResponderUrl);
69  		ocspResponderConn.setDoInput(true);
70  		ocspResponderConn.setDoOutput(true);
71  		ocspResponderConn.setUseCaches(false);
72  		ocspResponderConn.setConnectTimeout(connectTimeout);
73  		ocspResponderConn.setReadTimeout(readTimeout);
74  		ocspResponderConn.setRequestMethod(HttpPost.METHOD_NAME);
75  		byte[] ocspReq = ocspReqContent.getEncoded();
76  
77  		reqHeaders.forEach(ocspResponderConn::setRequestProperty);
78  		ocspResponderConn.setRequestProperty(HttpHeaders.CONTENT_LENGTH, Integer.toString(ocspReq.length));
79  
80  		try (OutputStream ocspResponderOutStream = ocspResponderConn.getOutputStream()) {
81  			ocspResponderOutStream.write(ocspReq);
82  			ocspResponderOutStream.flush();
83  		}
84  
85  		OCSPResp ocspResp = null;
86  
87  		try (InputStream ocspResponderInStream = ocspResponderConn.getInputStream()) {
88  			ocspResp = new OCSPResp(IOUtils.toByteArray(ocspResponderInStream));
89  		}
90  
91  		return ocspResp;
92  	}

94  }
