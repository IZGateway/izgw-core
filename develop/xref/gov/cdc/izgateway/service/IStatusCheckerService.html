1   package gov.cdc.izgateway.service;
2   
3   import gov.cdc.izgateway.model.IDestination;
4   import gov.cdc.izgateway.model.IEndpointStatus;
5   import gov.cdc.izgateway.soap.fault.Fault;
6   
7   /**
8    * The StatusCheckerService is responsible for checking and updating the status of a destination.
9    * A destination that fails status checks or has repeated failures on normal communications is marked
10   * as having thrown a circuit breaker.  Destinations which have thrown a circuit breaker cannot be sent
11   * messages except by those systems in the ADMIN Role. 
12   * 
13   * @author Audacious Inquiry
14   *
15   */
16  public interface IStatusCheckerService {
17  	/**
18  	 * Tells the service to periodically check the connectivity of the specified destination and to
19  	 * reset the circuit breaker when it comes back online.
20  	 * 
21  	 * @param dest	The destination whose circuit breaker was thrown.
22  	 */
23  	void lookForReset(IDestination dest);
24  	
25  	/**
26  	 * Update the status of the endpoint
27  	 * @param s	The endpoint status record.
28  	 * @param dest 
29  	 * @param reason The reason for updating status.
30  	 */
31  	void updateStatus(IEndpointStatus s, IDestination dest, Fault reason);
32  
33  	/**
34  	 * Determine if an endpoint is exempt from status checking (some internal destinations are
35  	 * not connectable by design [for testing purposes], and others are always running so long 
36  	 * as the IZ Gateway application is running).  Other destinations may not support the 
37  	 * connectivity test (historically, Florida fit into that category but no longer does).
38  	 * 
39  	 * @param destId	The destination to check
40  	 * @return	true if the destination is one of the exempted destinations.
41  	 */
42  	boolean isExempt(String destId);
43  
44  	/**
45  	 * Report that a circuit breaker has been reset for a given endpoint.
46  	 * @param status The endpoint status
47  	 */
48  	void logCircuitBreakerReset(IEndpointStatus status);
49  
50  	/**
51  	 * Report that a circuit breaker has been thrown for a given endpoint.
52  	 * @param status The endpoint status
53  	 * @param why The reason for the status
54  	 */
55  	void logCircuitBreakerThrown(IEndpointStatus status, Throwable why);
56  
57  }
