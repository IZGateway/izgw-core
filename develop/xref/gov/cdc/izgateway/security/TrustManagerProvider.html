26  /**
27   * This class provides the server trust manager for SSL connections.
28   * When {@code server.ssl.bundle} names a Spring Boot {@code SslBundle}, the trust store is
29   * read from that bundle and kept current via {@link SslBundles#addBundleUpdateHandler}, so a
30   * trust-store rotation (e.g. via {@link ServerSslBundleReloader}) is reflected here too, not
31   * just in the embedded Tomcat connector. Otherwise, falls back to loading the legacy
32   * {@code server.ssl.trust-store}/{@code -type}/{@code -provider} properties directly, once, for
33   * consumers that have not migrated to a named bundle -- that path has no reload, by design (no
34   * bundle means nothing to register an update handler against).
35   */
36  @Slf4j
37  @Component
38  public class TrustManagerProvider {
39  
40      private final SslBundles sslBundles;
41  
42      @Value("${server.ssl.bundle:}")
43      private String sslBundleName;
44  
45      @Value("${server.ssl.trust-store:}")
46      private String trustStorePath;
47  
48      @Value("${server.ssl.trust-store-password:}")
49      private String trustStorePassword;
50  
51      @Value("${server.ssl.trust-store-type:}")
52      private String trustStoreType;
53  
54      @Value("${server.ssl.trust-store-provider:}")
55      private String trustStoreProvider;
56  
57      @Getter
58      private X509TrustManager serverTrustManager;
59  
60      @Getter
61      private KeyStore trustStore;
62  
63      /**
64       * Create the trust manager provider.
65       * @param sslBundles The application's registered SSL bundles, used to resolve
66       * {@code server.ssl.bundle} when a named bundle is configured.
67       */
68      public TrustManagerProvider(SslBundles sslBundles) {
69          this.sslBundles = sslBundles;
70      }
71  
72      @PostConstruct
73      private void initializeTrustManager() {
74          applyTrustStore(StringUtils.hasText(sslBundleName) ? loadFromBundle() : loadFromLegacyProperties(),
75                  StringUtils.hasText(sslBundleName) ? "SSL bundle " + sslBundleName : trustStorePath);
76          if (StringUtils.hasText(sslBundleName)) {
77              sslBundles.addBundleUpdateHandler(sslBundleName, updatedBundle -> {
78                  applyTrustStore(updatedBundle.getStores().getTrustStore(), "SSL bundle " + sslBundleName);
79                  log.info("Reloaded trust manager from SSL bundle {} after keystore/truststore change", sslBundleName);
80              });
81          }
82      }
83  
84      /**
85       * Builds a fresh trust manager from the given store and assigns both fields, or throws if
86       * that fails. Called both from {@link #initializeTrustManager()} and from the bundle update
87       * handler registered there, so a trust-store rotation refreshes {@link #serverTrustManager}
88       * the same way the initial load does.
89       */
90      private void applyTrustStore(KeyStore newTrustStore, String source) {
91          try {
92              TrustManagerFactory tmf = TrustManagerFactory.getInstance("PKIX", "BCJSSE");
93              tmf.init(newTrustStore);
94  
95              for (TrustManager tm : tmf.getTrustManagers()) {
96                  if (tm instanceof X509TrustManager t) {
97                      this.trustStore = newTrustStore;
98                      this.serverTrustManager = t;
99                      return;
100                 }
101             }
102         } catch (GeneralSecurityException e) {
103             log.error(Markers2.append(e), "Cannot create Trust Manager from {}: {}", source, e.getMessage());
104             throw new ServiceConfigurationError(e.getMessage(), e);
105         }
106         throw new ServiceConfigurationError("No ServerTrustManager could be found in " + source);
107     }
108 
109     private KeyStore loadFromBundle() {
110         try {
111             return sslBundles.getBundle(sslBundleName).getStores().getTrustStore();
112         } catch (NoSuchSslBundleException e) {
113             log.error(Markers2.append(e), "Cannot create Trust Manager from SSL bundle {}: {}", sslBundleName, e.getMessage());
114             throw new ServiceConfigurationError(e.getMessage(), e);
115         }
116     }
117 
118     private KeyStore loadFromLegacyProperties() {
119         try {
120             KeyStore ks = KeyStore.getInstance(trustStoreType, trustStoreProvider);
121             try (FileInputStream fis = new FileInputStream(trustStorePath)) {
122                 ks.load(fis, trustStorePassword.toCharArray());
123             }
124             return ks;
125         } catch (GeneralSecurityException | IOException e) {
126             log.error(Markers2.append(e), "Cannot create Trust Manager from {}: {}", trustStorePath, e.getMessage());
127             throw new ServiceConfigurationError(e.getMessage(), e);
128         }
129     }
130 
131     /**
132      * Finds the issuer certificate for the given leaf certificate by matching its issuer DN
133      * against the subject DN of each entry in the trust store.
134      * Returns null if no match is found — callers should skip OCSP in that case.
135      */
136     public X509Certificate findIssuerCert(X509Certificate cert) {
137         String issuerDN = cert.getIssuerX500Principal().getName();
138         try {
139             Enumeration<String> aliases = trustStore.aliases();
140             while (aliases.hasMoreElements()) {
141                 Certificate candidate = trustStore.getCertificate(aliases.nextElement());
142                 if (candidate instanceof X509Certificate x509
143                         && x509.getSubjectX500Principal().getName().equals(issuerDN)) {
144                     return x509;
145                 }
146             }
147         } catch (KeyStoreException e) {
148             log.warn("Error looking up issuer cert for {}", issuerDN, e);
149         }
150         return null;
151     }
