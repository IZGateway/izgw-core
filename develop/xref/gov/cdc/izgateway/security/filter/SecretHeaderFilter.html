16  import java.util.List;

18  /**
19   * SecretHeaderFilter is a servlet filter that checks for a specific header in incoming requests.
20   * This is used to ensure that only requests from internal services (like ALB or WAF) are allowed to pass through.
21   * It is disabled by default, but can be enabled via the settings in the constructor.
22   * If the filter is enabled but the key or value is missing, an IllegalStateException is thrown.
23   */
24  @Slf4j
25  @Component
26  @Order()
27  public class SecretHeaderFilter implements Filter {
28      private final boolean headerFilterEnabled;
29      private final String headerFilterKey;
30      private final String headerFilterValue;
31      private final List<String> bypassPaths;
32  
33      public SecretHeaderFilter(
34              @Value("${hub.security.secret-header-filter.enabled:false}") boolean headerFilterEnabled,
35              @Value("${hub.security.secret-header-filter.key:}") String headerFilterKey,
36              @Value("${hub.security.secret-header-filter.value:}") String headerFilterValue,
37              @Value("${hub.security.secret-header-filter.bypass-paths:/rest/health}") List<String> bypassPaths
38      ) {
39          this.headerFilterEnabled = headerFilterEnabled;
40          this.headerFilterKey = headerFilterKey;
41          this.headerFilterValue = headerFilterValue;
42          this.bypassPaths = bypassPaths;
43  
44          if (this.headerFilterEnabled) {
45              if (StringUtils.isEmpty(this.headerFilterKey) || StringUtils.isEmpty(this.headerFilterValue)) {
46                  throw new IllegalStateException("Secret header filter is enabled, but the header key or value is not set.");
47              }
48          } else {
49              log.warn("Secret header filter not enabled. Requests will not be filtered.");
50          }
51      }
52  
53      @Override
54      public void doFilter(ServletRequest servletRequest, ServletResponse servletResponse, FilterChain filterChain) throws IOException, ServletException {
55          if (bypassThisFilter(servletRequest)) {
56              filterChain.doFilter(servletRequest, servletResponse);
57              return;
58          }
59  
60          // Check if the request is from an internal service (ALB, WAF) by checking the header
61          HttpServletRequest request = (HttpServletRequest) servletRequest;
62          HttpServletResponse response = (HttpServletResponse) servletResponse;
63          String headerValue = request.getHeader(headerFilterKey);
64  
65          if (headerValue == null || !headerValue.equals(headerFilterValue)) {
66              log.error(Markers2.append(RequestContext.getSourceInfo()), "Request does not contain the secret header, rejecting request.");
67              response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
68              return;
69          }
70  
71          // Continue the filter chain
72          filterChain.doFilter(servletRequest, servletResponse);
73      }
74  
75      private boolean bypassThisFilter(ServletRequest servletRequest) {
76          if (!headerFilterEnabled || AccessControlValve.isLocalHost(servletRequest.getRemoteAddr())) {
77              return true;
78          }
79  
80          String requestURI = ((HttpServletRequest) servletRequest).getRequestURI();
81          return bypassPaths.stream().anyMatch(requestURI::equalsIgnoreCase);
82      }
83  }
