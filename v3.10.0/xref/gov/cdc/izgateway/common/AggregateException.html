1   package gov.cdc.izgateway.common;
2   
3   import java.util.List;
4   
5   /**
6    * A runtime exception that aggregates multiple exceptions thrown during a bulk operation.
7    * <p>
8    * Use this when a loop must continue processing all items regardless of per-item failures,
9    * collecting each failure and throwing a single combined exception at the end so no errors
10   * are silently discarded.
11   * </p>
12   *
13   * <pre>
14   * List&lt;RuntimeException&gt; errors = new ArrayList&lt;&gt;();
15   * for (Item item : items) {
16   *     try {
17   *         process(item);
18   *     } catch (RuntimeException e) {
19   *         errors.add(e);
20   *     }
21   * }
22   * AggregateException.throwIfAny(errors, "processing items");
23   * </pre>
24   */
25  @SuppressWarnings("serial")
26  public class AggregateException extends RuntimeException {
27  
28      /** The individual exceptions collected during the bulk operation. */
29      private final List<RuntimeException> causes;
30  
31      /**
32       * Creates an {@code AggregateException} from a list of collected exceptions.
33       *
34       * @param message a description of the bulk operation that produced the errors
35       * @param causes  the individual exceptions; must not be {@code null} or empty
36       */
37      public AggregateException(String message, List<RuntimeException> causes) {
38          super(buildMessage(message, causes == null ? java.util.List.of() : causes),
39                (causes == null || causes.isEmpty()) ? null : causes.get(0));
40          this.causes = java.util.List.copyOf(causes == null ? java.util.List.of() : causes);
41          for (int i = 1; i < this.causes.size(); i++) {
42              addSuppressed(this.causes.get(i));
43          }
44      }
45  
46      /**
47       * Returns the unmodifiable list of individual exceptions collected during the operation.
48       *
49       * @return the collected exceptions; never {@code null}, never empty
50       */
51      public List<RuntimeException> getCauses() {
52          return causes;
53      }
54  
55      /**
56       * Throws an {@code AggregateException} if {@code errors} is non-empty; otherwise returns
57       * silently. Intended as a convenient end-of-loop guard.
58       *
59       * @param errors      the exceptions collected during a bulk operation
60       * @param operationDescription a short label used in the exception message (e.g. {@code "resetting circuit breakers"})
61       * @throws AggregateException if {@code errors} is non-empty
62       */
63      public static void throwIfAny(List<RuntimeException> errors, String operationDescription) {
64          if (errors != null && !errors.isEmpty()) {
65              throw new AggregateException(operationDescription, errors);
66          }
67      }
68  
69      private static String buildMessage(String operation, List<RuntimeException> causes) {
70          StringBuilder sb = new StringBuilder();
71          sb.append(causes.size())
72            .append(" error(s) occurred while ")
73            .append(operation)
74            .append(':');
75          for (int i = 0; i < causes.size(); i++) {
76              sb.append('\n')
77                .append(i + 1)
78                .append(". ")
79                .append(causes.get(i).getMessage());
80          }
81          return sb.toString();
82      }
83  }
