22  @Slf4j
23  public abstract class AbstractJwtPrincipalProvider implements JwtPrincipalProvider {
24  
25      @Value("${jwt.roles-claim}")
26      private String rolesClaim;
27  
28      @Value("${jwt.scopes-claim}")
29      private String scopesClaim;
30  
31      protected final GroupToRoleMapper groupToRoleMapper;
32      protected final ScopeToRoleMapper scopeToRoleMapper;
33      protected final JwtTokenExtractor jwtTokenExtractor;
34  
35      protected AbstractJwtPrincipalProvider(
36              @Nullable GroupToRoleMapper groupToRoleMapper,
37              @Nullable ScopeToRoleMapper scopeToRoleMapper,
38              JwtTokenExtractor jwtTokenExtractor) {
39          this.groupToRoleMapper = groupToRoleMapper;
40          this.scopeToRoleMapper = scopeToRoleMapper;
41          this.jwtTokenExtractor = jwtTokenExtractor;
42      }
43  
44      protected abstract NimbusJwtDecoder createDecoder();
45  
46      protected abstract boolean validConfiguration();
47  
48      protected DelegatingOAuth2TokenValidator<Jwt> createValidator() {
49          JwtTimestampValidator timestampValidator = new JwtTimestampValidator();
50  
51          // Make sure exp is in token, we are requiring it
52          OAuth2TokenValidator<Jwt> expPresenceValidator = token -> {
53              if (!token.getClaims().containsKey("exp")) {
54                  return OAuth2TokenValidatorResult.failure(
55                          new OAuth2Error("token_missing_claim",
56                                  "token must contain exp claim", null)
57                  );
58              }
59              return OAuth2TokenValidatorResult.success();
60          };
61  
62          // Parse claims directly for a couple scenarios
63          OAuth2TokenValidator<Jwt> directClaimsVerification = token -> {
64              try {
65                  SignedJWT signedJWT = (SignedJWT) JWTParser.parse(token.getTokenValue());
66                  Map<String, Object> claimsMap = signedJWT.getJWTClaimsSet().getClaims();
67  
68                  // Make sure that JWT has iat or nbf, both can't be left out
69                  boolean hasIat = claimsMap.containsKey("iat");
70                  boolean hasNbf = claimsMap.containsKey("nbf");
71                  if (!hasIat && !hasNbf) {
72                      return OAuth2TokenValidatorResult.failure(
73                              new OAuth2Error("token_missing_claim",
74                                      "token must contain either iat or nbf claim", null)
75                      );
76                  }
77  
78                  // Check to see if iat is in the future
79                  if (hasIat) {
80                      Instant iat = token.getIssuedAt();
81                      if ((iat != null) && iat.isAfter(Instant.now())) {
82                          return OAuth2TokenValidatorResult.failure(
83                                  new OAuth2Error("invalid_claim",
84                                          "iat cannot be in the future", null)
85                          );
86                      }
87                  }
88              } catch (ParseException e) {
89                  return OAuth2TokenValidatorResult.failure(
90                          new OAuth2Error("invalid_token",
91                                  "failed to parse claims from JWT", null)
92                  );
93              }
94              return OAuth2TokenValidatorResult.success();
95          };
96  
97          return new DelegatingOAuth2TokenValidator<>(
98                  timestampValidator,
99                  expPresenceValidator,
100                 directClaimsVerification
101         );
102     }
103 
104     @Override
105     public IzgPrincipal createPrincipalFromJwt(HttpServletRequest request) {
106         if (!validConfiguration()) {
107             return null;
108         }
109 
110         try {
111             String token = jwtTokenExtractor.extractToken(request);
112 
113             NimbusJwtDecoder jwtDecoder = createDecoder();
114             jwtDecoder.setJwtValidator(createValidator());
115 
116             Jwt jwt = jwtDecoder.decode(token);
117 
118             log.debug("JWT claims for current request: {}", jwt.getClaims());
119 
120             return new JWTPrincipal(jwt,
121                     rolesClaim,
122                     scopesClaim,
123                     groupToRoleMapper,
124                     scopeToRoleMapper
125             );
126 
127         } catch (MissingJwtTokenException e) {
128         	// Logged when originally thrown, no need to log again
129             return null;
130         } catch (Exception e) {
131             log.warn("Issue processing JWT token: {}", e.getMessage(), e);
132             return null;
133         }
134     }
135 }
