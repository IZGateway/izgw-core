1   package gov.cdc.izgateway.security;
2   
3   import java.io.File;
4   import java.io.FileInputStream;
5   import java.io.FileNotFoundException;
6   import java.io.IOException;
7   import java.security.GeneralSecurityException;
8   import java.security.KeyStore;
9   import java.util.ServiceConfigurationError;
10  
11  import com.fasterxml.jackson.annotation.JsonIgnore;
12  
13  import gov.cdc.izgateway.logging.markers.Markers2;
14  import lombok.Getter;
15  import lombok.extern.slf4j.Slf4j;
16  
17  @Slf4j 
18  class KeyStoreLoader {
19  	@Getter
20  	private final String file;
21  	@Getter
22  	@JsonIgnore // should this beast ever attempt to be converted to Json, hide the password.
23  	private final String password;
24  	@Getter
25  	private final String provider;
26  	@Getter
27  	private final String type;
28  
29  	@Getter
30  	private long lastUpdated = 0;
31  	@Getter
32  	private int reloadCount = 0;
33  	
34  	@Getter
35  	@JsonIgnore
36  	private KeyStore store;
37  	
38  	KeyStoreLoader(String file, String password, String provider, String type) {
39  		try {
40  			if (file == null) {
41  				throw new NullPointerException("file cannot be null");
42  			}
43  			if (password == null) {
44  				throw new NullPointerException("password cannot be null");
45  			}
46  			if (provider == null) {
47  				throw new NullPointerException("provider cannot be null");
48  			}
49  			if (type == null) {
50  				throw new NullPointerException("type cannot be null");
51  			}
52  			this.file = file;
53  			this.password = password;
54  			this.provider = provider;
55  			this.type = type;
56  			File f = new File(this.file);
57  			if (!f.exists()) {
58  				throw new IllegalArgumentException("File " + file + " does not exist");
59  			}
60  		} catch (Exception ex) {
61  			// Unbury the root cause in logs
62  			log.error("Error initializing key or trust store: {}", ex.getMessage());
63  			throw ex;
64  		}
65  	}
66  	
67  	long getLastModified() {
68  		File f = new File(file);
69  		if (!f.exists()) {
70  			return 0;
71  		}
72  		return f.lastModified();
73  	}
74  	
75  	void updated() {
76  		++reloadCount;
77  		this.lastUpdated = getLastModified();
78  	}
79  	
80  	KeyStore load() {
81  		return load(true);
82  	}
83  	
84  	KeyStore load(boolean reload) {
85  		reload = reload || isOutOfDate();
86  		if (!reload && store != null) {
87  			return store;
88  		}
89  		try {
90  			KeyStore ks = KeyStore.getInstance(getType().toUpperCase(), getProvider());
91  			ks.load(new FileInputStream(new File(getFile())), getPassword().toCharArray());
92  			updated();
93  			store = ks;
94  			return store;
95  		} catch (GeneralSecurityException e) {
96  			log.error(Markers2.append(e), "Cannot create Key Store : {}", e.getMessage());
97  			throw new ServiceConfigurationError(e.getMessage(), e);
98  		} catch (FileNotFoundException e) {
99  			log.error(Markers2.append(e), "Cannot open Key Store file: {}", getFile());
100 			throw new ServiceConfigurationError(e.getMessage(), e);
101 		} catch (IOException e) {
102 			log.error(Markers2.append(e), "Cannot read Key Store file: {}", getFile());
103 			throw new ServiceConfigurationError(e.getMessage(), e);
104 		}
105 	}
106 	
107 	boolean isOutOfDate() {
108 		return getLastModified() > lastUpdated;
109 	}
110 }
