1   package gov.cdc.izgateway.security;
2   
3   import java.math.BigInteger;
4   import java.nio.charset.StandardCharsets;
5   import java.util.Date;
6   import java.util.HashSet;
7   
8   public class UnauthenticatedPrincipal extends IzgPrincipal {
9   
10      public UnauthenticatedPrincipal() {
11          this.name = "Unauthenticated";
12          this.organization = "Public";
13          this.validFrom = new Date();
14          this.validTo = new Date(Long.MAX_VALUE);
15          this.serialNumber = "0";
16          this.issuer = "None";
17          this.audience = null;
18          this.scopes = new HashSet<>();
19          this.roles = new HashSet<>();
20      }
21  
22      /**
23       * {@inheritDoc}
24       * <p>
25       * Always false: this principal stands in for a caller that presented no usable
26       * credential, or whose credential failed authentication.
27       */
28      @Override
29      public boolean isAuthenticated() {
30          return false;
31      }
32  
33      @Override
34      public String getSerialNumberHex() {
35          return String.format("%040x", new BigInteger(1, serialNumber.getBytes(StandardCharsets.UTF_8)));
36      }
37  }
