3   import com.fasterxml.jackson.core.JsonProcessingException;
4   import com.fasterxml.jackson.databind.ObjectMapper;
5   
6   /**
7    * Quick and dirty converter of POJOs to JSON Strings.
8    */
9   public class JsonUtils {
10  	public static String toString(Object o) {
11  	    // Create a json object from the data elements, and convert that to a string.
12  	    ObjectMapper om = new ObjectMapper();
13  	    try {
14  	        return om.writeValueAsString(o);
15  	    } catch (JsonProcessingException e) {
16  	        return "Error converting " + o.getClass().getSimpleName() + " data to JSON: " + e.getMessage();
17  	    }
18  	}
19  }
