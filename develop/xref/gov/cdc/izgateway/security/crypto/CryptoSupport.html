21  /**
22   * Cryptographic support for encryption and decryption of sensitive data.
23   * 
24   * @author Audacious Inquiry
25   */
26  @Slf4j
27  public class CryptoSupport {
28  	/** Length in bytes of the Initialization Vector */
29  	private static final int IV_LENGTH = 16;
30  	/** Length in bytes of the Authentication Tag */
31  	private static final int TAG_LENGTH = 16;
32  	/** The Cipher Algorithm to Use */
33      private static final String CIPHER_ALGORITHM = "AES/GCM/NoPadding";
34      /** A secure random number generator */
35      private static final SecureRandom secureRandom = getSecureRandom();
36  
37      private static KeyProvider keyProvider = null;
38  
39      private CryptoSupport() {
40          // Prevent instantiation

42      /**
43       * Sets the key provider for dependency injection.

51      static void setKeyProvider(KeyProvider provider) {
52          keyProvider = provider;
53      }

55      static KeyProvider getKeyProvider() {
56      	if (keyProvider == null) {
57      		// Initialize default provider
58          	keyProvider = new AwsSecretsManagerKeyProvider();
59  		}
60  		return keyProvider;
61  	}

63      /**
64       * Encrypts the given plain text using AES-GCM with a random IV.
65       * 
66       * @param plainText	the text to encrypt
67       * @param keyBytes the AES encryption key (must be 32 bytes for AES-256)
68       * @return	the encrypted text, base64-encoded and prefixed with "=="
69       * @throws CryptoException	if an error occurs during encryption
70       */
71      public static String encrypt(String plainText, byte[] keyBytes) throws CryptoException {
72          if (plainText == null || plainText.isEmpty() || plainText.startsWith("==")) {
73              return plainText;
74          }
75  
76          try {
77              byte[] iv = new byte[IV_LENGTH];
78              secureRandom.nextBytes(iv);
79  
80              SecretKeySpec key = new SecretKeySpec(keyBytes, "AES");
81              Cipher cipher = Cipher.getInstance(CIPHER_ALGORITHM, "BCFIPS");
82              cipher.init(Cipher.ENCRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH*8, iv));
83  
84              byte[] input = plainText.getBytes(StandardCharsets.UTF_8);
85              byte[] encrypted = cipher.doFinal(input, 0, input.length);
86  
87              // Prepend IV to the ciphertext
88              byte[] result = Arrays.concatenate(iv, encrypted);
89  
90              return "==" + Base64.toBase64String(result);
91  
92          } catch(Exception e) {
93          	throw new CryptoException("Failed to encrypt.", e);
94          }
95      }
96  
97      /**
98       * Decrypts encrypted text using key rotation fallback strategy.
99       * <p>
100      * This method attempts to decrypt the input using multiple keys to support
101      * key rotation scenarios. It first tries all keys from the provider's history,
102      * then attempts to load a fresh key if decryption fails.
103      * </p>
104      * <p>
105      * <strong>Decryption Strategy:</strong>
106      * <ol>
107      *   <li>Try decryption with each key from the key history</li>
108      *   <li>If all history keys fail, load the current key from the provider</li>
109      *   <li>Add the current key to history if not already present</li>
110      *   <li>Attempt final decryption with the current key</li>
111      * </ol>
112      * </p>
113      *
114      * @param encryptedText the encrypted text to decrypt (must be prefixed with "==")
115      * @return the decrypted plain text
116      * @throws CryptoException if decryption fails with all available keys
117      */
118     public static String decrypt(String encryptedText) throws CryptoException {
119         if (encryptedText == null || !encryptedText.startsWith("==")) {
120             return encryptedText;
121         }
122         KeyProvider keyProvider = getKeyProvider();
123         for (byte[] key : keyProvider.getAllKeys()) {
124             try {
125                 return decrypt(encryptedText, key);
126             } catch (CryptoException e) {
127                 // Log and continue to next key
128                 log.debug("Decryption failed with a key from history, trying next if available.", e);
129             }
130         }
131 
132         try {
133             byte[] keyBytes = keyProvider.loadKey();
134             if (!keyProvider.keyExists(keyBytes)) {
135                 keyProvider.addKeyToHistory(keyBytes);
136                 return decrypt(encryptedText, keyBytes);
137             } else {
138                 throw new CryptoException("Decryption failed with all available keys.");
139             }
140         } catch (CryptoException e) {
141             throw new CryptoException("Failed to decrypt with all available keys.", e);
142         }
143     }

145     /**
146      * Decrypts the given encrypted text using AES-GCM.
147      * @param encryptedText	the text to decrypt, base64-encoded and prefixed with "=="
148      * @param keyBytes the AES decryption key (must be 32 bytes for AES-256)
149      * @return	the decrypted plain text
150      * @throws CryptoException	if an error occurs during decryption
151      */
152     private static String decrypt(String encryptedText, byte[] keyBytes) throws CryptoException {
153 
154         try {
155             byte[] data = Base64.decode(encryptedText.substring(2));
156             byte[] iv = Arrays.copyOfRange(data, 0, IV_LENGTH);
157             byte[] encrypted = Arrays.copyOfRange(data, IV_LENGTH, data.length);
158 
159             SecretKeySpec key = new SecretKeySpec(keyBytes, "AES");
160 
161             Cipher cipher = Cipher.getInstance(CIPHER_ALGORITHM, "BCFIPS");
162             cipher.init(Cipher.DECRYPT_MODE, key, new GCMParameterSpec(TAG_LENGTH*8, iv));
163             byte[] decrypted = cipher.doFinal(encrypted, 0, encrypted.length);
164             return new String(decrypted, StandardCharsets.UTF_8);
165         } catch (Exception e) {
166             throw new CryptoException("Failed to decrypt.", e);
167         }
168     }

170     /**
171      * Get a SecureRandom instance using a FIPS 140-2 approved DRBG
172      * @return The SecureRandom instance
173      */
174     public static SecureRandom getSecureRandom() {
175 		/*
176          * According to NIST Special Publication 800-90A, a Nonce is
177          * A time-varying value that has at most a negligible chance of
178          * repeating, e.g., a random value that is generated anew for each
179          * use, a timestamp, a sequence number, or some combination of
180          * these.
181          *
182          * The nonce is combined with the entropy input to create the initial
183          * DRBG seed.
184          */
185         byte[] nonce = ByteBuffer.allocate(8).putLong(System.nanoTime()).array();
186         EntropySourceProvider entSource = new BasicEntropySourceProvider(new SecureRandom(), true);
187         FipsDRBG.Builder drgbBldr = FipsDRBG.SHA512
188                 .fromEntropySource(entSource).setSecurityStrength(256)
189                 .setEntropyBitsRequired(256);
190         return drgbBldr.build(nonce, true);
191     }

194      * Initialize the BC-FIPS Module as the JCA/JCE provider.
195      */
196     public static void initialize() {
197 		CryptoServicesRegistrar.setSecureRandom(getSecureRandom());
198 		Security.insertProviderAt(new BouncyCastleFipsProvider(), 1);
199 		Security.insertProviderAt(new BouncyCastleJsseProvider(), 2);
