13  import com.fasterxml.jackson.databind.JsonNode;

16  /**
17   * Verifies the public view of a health report discloses nothing beyond the liveness answer.
18   *
19   * @see Health#publicView()
20   */
21  class PublicHealthTests {
22  	private static final ObjectMapper MAPPER = new ObjectMapper();
23  
24  	/** The only keys an unauthenticated caller may receive. Changing this changes what we disclose. */
25  	private static final Set<String> ALLOWED_KEYS = Set.of("healthy", "statusAt");
26  
27  	private static Health populatedHealth() {
28  		Health h = new Health();
29  		h.setHealthy(true);
30  		h.setLastChangeReason("Database izgateway-prod.example.internal came back");
31  		h.setBuildName("3.8.0-1234");
32  		h.setServerName("phiz-web-ws-7.prod.internal");
33  		h.setDatabase("jdbc:mysql://izgateway-prod.cluster-abc123.us-east-1.rds.amazonaws.com:3306/izgw");
34  		h.setIngressDnsAddress(new String[] { "3.232.82.52" });
35  		h.setEgressDnsAddress("54.87.148.103");
36  		h.setAccessControlAction("warn");
37  		h.setLastException(new IllegalStateException("connection refused to 10.0.3.44:3306"));
38  		h.incrementRequestVolume();
39  		h.incrementSuccessVolume();
40  		return h;
41  	}
42  
43  	/**
44  	 * The guard that matters: the serialized public view carries exactly the allowed keys.
45  	 * A new field added to Health does not reach an unauthenticated caller, and if someone
46  	 * adds one to PublicHealth instead, this fails rather than leaking quietly.
47  	 */
48  	@Test
49  	void publicViewSerializesOnlyTheAllowedKeys() throws Exception {
50  		JsonNode json = MAPPER.valueToTree(populatedHealth().publicView());
51  
52  		Set<String> actual = new TreeSet<>();
53  		json.fieldNames().forEachRemaining(actual::add);
54  
55  		assertEquals(new TreeSet<>(ALLOWED_KEYS), actual,
56  				"The public health view must disclose exactly " + ALLOWED_KEYS);
57  	}
58  
59  	/** None of the infrastructure detail appears anywhere in the public payload, under any key. */
60  	@Test
61  	void publicViewDisclosesNoInfrastructureDetail() throws Exception {
62  		String json = MAPPER.writeValueAsString(populatedHealth().publicView());
63  
64  		for (String secret : new String[] {
65  				"phiz-web-ws-7.prod.internal",
66  				"izgateway-prod.cluster-abc123.us-east-1.rds.amazonaws.com",
67  				"3.232.82.52",
68  				"54.87.148.103",
69  				"3.8.0-1234",
70  				"warn",
71  				"Database izgateway-prod.example.internal came back",
72  				"connection refused",
73  				"10.0.3.44" }) {
74  			assertFalse(json.contains(secret), "Public health view disclosed: " + secret);
75  		}
76  	}
77  
78  	/**
79  	 * The exception message is the easiest of these to miss: {@code Health}'s {@code lastException}
80  	 * field is {@code @JsonIgnore}d, which makes it look withheld, but the hand-written
81  	 * {@link Health#getLastException()} is {@code @JsonProperty} and puts the message on the wire.
82  	 * Pinned against the full report as well so that the premise behind this test stays true.
83  	 */
84  	@Test
85  	void theFullReportIsWhatDisclosesTheExceptionMessage() throws Exception {
86  		Health h = populatedHealth();
87  
88  		assertTrue(MAPPER.writeValueAsString(h).contains("connection refused"),
89  				"If the full report stops carrying the exception message, this allowlist entry is moot");
90  		assertFalse(MAPPER.writeValueAsString(h.publicView()).contains("connection refused"));
91  	}
92  
93  	/** The full report is unchanged for authenticated callers; publicView() does not mutate it. */
94  	@Test
95  	void publicViewLeavesTheFullReportIntact() throws Exception {
96  		Health h = populatedHealth();
97  		String before = MAPPER.writeValueAsString(h);
98  
99  		h.publicView();
100 
101 		assertEquals(before, MAPPER.writeValueAsString(h));
102 		assertTrue(before.contains("phiz-web-ws-7.prod.internal"),
103 				"The full report should still carry the detail an authenticated caller expects");
104 	}
105 
106 	/** The liveness answer itself survives, since that is what the endpoint is for. */
107 	@Test
108 	void publicViewCarriesTheLivenessAnswer() {
109 		Health unhealthy = new Health();
110 		unhealthy.setHealthy(false);
111 
112 		assertTrue(populatedHealth().publicView().isHealthy());
113 		assertFalse(unhealthy.publicView().isHealthy());
114 	}
115 
116 	/** The timestamp is copied, so a caller cannot reach back into the live report through it. */
117 	@Test
118 	void statusAtIsDefensivelyCopied() {
119 		Health h = populatedHealth();
120 		PublicHealth view = h.publicView();
121 
122 		assertNotSame(h.getStatusAt(), view.getStatusAt());
123 		assertEquals(h.getStatusAt(), view.getStatusAt());
124 		assertNotSame(view.getStatusAt(), view.getStatusAt());
125 	}
