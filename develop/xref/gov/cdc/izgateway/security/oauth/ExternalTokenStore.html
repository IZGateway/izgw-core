58      public static ExternalTokenStore getTokenStore(URL url, String username, String password) { 
59      	ExternalTokenStore t = stores.computeIfAbsent(url.toString(), u -> new ExternalTokenStore(url, username, password));
60      	// If the username or password changed since we created it, update them.
61      	if (!t.username.equals(username)) {
62      		t.username = username;
63      	}
64      	if (!t.password.equals(password)) {
65      		t.password = password;
66      	}
67      	return t;
68      }

75      	ExternalTokenStore t = stores.get(url);

86      @SuppressWarnings("serial")
87      public static class OAuthReportedHttpException extends HttpStatusException { // NOSONAR, depth of class hierarchy OK
88          private final String body;
89          private final String msg;
90          private OAuthReportedHttpException(int responseCode, String body) {
91              super(responseCode);
92          	msg = String.format("OAuth Response: %d%n", getStatusCode());
93              this.body = body == null ? "" : body;
94          }
95          
96          public String getErrorBody() {
97              return body;
98          }
99          
100         @Override
101         public String getMessage() {
102         	return msg + body;
103         }
104     }

106     private URL getURL(String endpoint) throws MalformedURLException {
107         if (isUsingQueryParameters()) {
108         	endpoint = String.format("%s?username=%s&password=%s", 
109         			endpoint,
110         			URLEncoder.encode(username, StandardCharsets.UTF_8),
111         			URLEncoder.encode(password, StandardCharsets.UTF_8)
112         	);
113         }
114         try {
115             return url.toURI().resolve(endpoint).toURL();
116         } catch (URISyntaxException | IllegalArgumentException e) {
117             throw new MalformedURLException(e.getMessage());
118         }
119     	
120     }

122     private AccessToken getNewToken() throws IOException {
123     	return requestAccessToken(this::writeAccessTokenRequest, getURL("oauth"), null);
124     }
125     
126     private AccessToken refreshToken(AccessToken token) throws IOException {
127         // If our refresh period expired, we need to get a brand new token.
128         if (token == null || !token.isRefreshable()) {
129             return getNewToken();
130         }
131     	return requestAccessToken(this::writeRefreshTokenRequest, getURL("oauth/refresh"), token);
132     }

134     private void writeAccessTokenRequest(HttpsURLConnection con, AccessToken notUsed) throws IOException {
135         // Write the request content
136     	// I believe we have to at least get the OutputStream or no Content-Length header will be written.
137     	OutputStream os = con.getOutputStream();
138         Writer w = new OutputStreamWriter(os, StandardCharsets.UTF_8);
139     	if (!isUsingQueryParameters()) {
140     	    w.write("username=");
141             w.write(URLEncoder.encode(username, StandardCharsets.UTF_8));
142             w.write("&password=");
143             w.write(URLEncoder.encode(password, StandardCharsets.UTF_8));
144             w.write("&grant_type=password");
145     	}
146     	w.flush();
147     }
148 
149     private void writeRefreshTokenRequest(HttpsURLConnection con, AccessToken token) throws IOException {
150     	OutputStream os = con.getOutputStream();
151         Writer w = new OutputStreamWriter(os, StandardCharsets.UTF_8);
152     	if (!isUsingQueryParameters()) {
153             w.write("refresh_token=");
154             w.write(URLEncoder.encode(token.getRefreshToken(), StandardCharsets.UTF_8));
155             w.write("&grant_type=refresh_token");
156     	}
157     	w.flush();
158     }

160     private AccessToken requestAccessToken(RequestWriter requestWriter, URL base, AccessToken token) throws IOException {
161         HttpsURLConnection con = (HttpsURLConnection) tlsSupport.getSNIEnabledConnection(base);
162         prepareConnectionForFormPost(con);
163         requestWriter.write(con, token);
164         int responseCode = con.getResponseCode(); 
165         if (responseCode == HttpServletResponse.SC_OK || responseCode == HttpServletResponse.SC_CREATED) {
166             // Read the response back into a token.
167             return save(readToken(con));
168         }
169         throw new OAuthReportedHttpException(responseCode, readError(con));
170     }

172 	private String readError(HttpsURLConnection con) {
173         try (InputStream is = con.getErrorStream()) {
174         	if (is == null) {
175         		return null;
176         	}
177           return IOUtils.toString(is, StandardCharsets.UTF_8);
178         } catch (IOException e) {
179             return null;
180         }
181     }

183     /**
184      * Get an existing token, or renew a previously existing token.
185      * @return An unexpired token
186      * @throws IOException  If an IO Error occured while getting a token. 
187      */
188     public String getToken() throws IOException  {
189     	int retries = 0;
190     	IOException lastEx = null;
191     	
192     	do {
193     		try {
194 		        if (accessToken == null) {
195 		        	accessToken = getNewToken();
196 		        	log.info(Markers2.append(OAUTH_ENDPOINT, url.toString()), "Got new access token");
197 		        } else if (!accessToken.isValid()) {
198 		        	refreshOnFailure();
199 		        }
200 		        if (accessToken != null) {
201 		        	return accessToken.getAccessToken();
202 		        }
203     		} catch (OAuthReportedHttpException ex) {
204     			log.error(Markers2.append(ex, OAUTH_ENDPOINT, url.toString()), 
205     					"Authorization failed for user {} to {}", username, url);
206     			lastEx = new IOException("OAuth Exception: " + ex.getMessage(), ex);
207     		} catch (IOException ex) {
208     			log.error(Markers2.append(ex, OAUTH_ENDPOINT, url.toString()), 
209     					"Exception on authorization for user {} to {}", username, url);
210     			lastEx = ex;
211     		} catch (Exception ex) {
212     			log.error(Markers2.append(ex, OAUTH_ENDPOINT, url.toString()), 
213     					"Exception on authorization for user {} to {}", username, url);
214     			lastEx = new IOException("Unexpected exception: " + ex.getMessage(), ex);
215     		}
216     	} while (retries++ < numRetries);
217     	
218 		accessToken = null;
219 
220 		throw lastEx;
221     }
222 
223 	private void refreshOnFailure() throws IOException {
224 		try {
225 			accessToken = refreshToken(accessToken);
226 			log.info(Markers2.append(OAUTH_ENDPOINT, url.toString()), "Refreshed access token");
227 		} catch (OAuthReportedHttpException oex) {
228 			// On a refresh failure, try to get a brand new token instead.
229 			log.warn(Markers2.append(oex, OAUTH_ENDPOINT, url.toString()),
230 					"Authorization refresh failed for user {} to {}", username, url);
231 			accessToken = getNewToken();
232 			log.info(Markers2.append(OAUTH_ENDPOINT, url.toString()), 
233 					"Got new access token after failed refresh");
234 		}
235 	}

237     private void prepareConnectionForFormPost(HttpsURLConnection con) throws java.net.ProtocolException {
238         // Set up our trust store
239     	SSLSocketFactory factory = con.getSSLSocketFactory();
240     	if (isDebugging()) {
241     		factory = new CapturingSSLSocketFactory(factory);
242     	}
243     	con.setSSLSocketFactory(factory);
244         con.setUseCaches(false);
245         // Write this out using application/x-www-form-urlencoded
246         con.setRequestProperty(HttpHeaders.CONTENT_TYPE, MediaType.APPLICATION_FORM_URLENCODED.toString());
247         if (isUsingQueryParameters()) {
248         	// Set Content-Length to 0 for query parameters.
249         	con.setRequestProperty(HttpHeaders.CONTENT_LENGTH, "0");
250         }
251         // Set up POST request for read/write
252         con.setDoInput(true);
253         // No writing to this request if using query parameters.
254         con.setDoOutput(true);
255         con.setRequestMethod(HttpMethod.POST.toString());
256     }

258     private AccessToken readToken(HttpsURLConnection con) throws IOException {
259         AccessToken token = null;
260         try (InputStream is = con.getInputStream()) {
261         	String value = IOUtils.toString(is, StandardCharsets.UTF_8);
262             ObjectMapper mapper = new ObjectMapper();
263             mapper.disable(
264                 DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES,
265                 DeserializationFeature.FAIL_ON_IGNORED_PROPERTIES
266             );
267             token = mapper.readValue(new StringReader(value), AccessToken.class);
268         }
269         return token;
270     }
