104 		private static final long serialVersionUID = 1L;
105 		@Getter
106 		private final String exchange;
107 		SoapParseException(Throwable th, String exchange) {
108 			super(getMessage(th), 
109 				th instanceof XMLStreamException xex ? xex.getLocation() : XMLStreamLocation2.NOT_AVAILABLE, th);
110 			this.exchange = exchange;
111 		}
112 		/**
113 		 * Remove extra text in the produced message.
114 		 * @param th	The original exception.
115 		 * @return	Just the message without location and Message: prefix
116 		 */
117 		private static String getMessage(Throwable th) {
118 			if (th instanceof XMLStreamException) {
119 				String msg = StringUtils.substringAfter(th.getMessage(), "Message: ");
120 				if (StringUtils.isBlank(msg)) {
121 					msg = th.getMessage();
122 				}
123 				return msg;
124 			}
125 			return th.getMessage();
126 		}

167 	/**
168 	 * Translate the SoapMessages for IZ Gateway Hub, CDC-2011 and 2014 WDSLs into
169 	 * business objects containing the relevant information.
170 	 * 
171 	 * @return A SoapMessage of the appropriate type representing the message.
172 	 * @throws SecurityFault
173 	 * @throws XMLStreamException
174 	 */
175 	public SoapMessage read() throws SecurityFault, XMLStreamException {
176 		try {
177 			boolean filter = AppProperties.isProduction();
178 			int eventType;
179 			while ((eventType = reader.next()) != XMLStreamConstants.END_DOCUMENT) {
180 				boolean skip = false;
181 				switch (eventType) {
182 				case XMLStreamConstants.START_ELEMENT:
183 					skip = parseElement(filter);
184 					QName element = reader.getName();
185 					stack.push(element);
186 					lastElement = toNameString(element);
187 					break;
188 				case XMLStreamConstants.END_ELEMENT:
189 					stack.pop();
190 					lastElement = toNameString(stack.peekFirst());
191 					break;
192 				case XMLStreamConstants.CDATA, XMLStreamConstants.CHARACTERS:
193 					validateText(reader.getText());
194 					break;
195 				default:
196 					break;
197 				}
198 
199 				// If we are saving output, and shouldn't skip handling of this element
200 				// content.
201 				if (!skip) {
202 					copyEvent();
203 				}
204 			}
205 		} catch (XMLStreamException ex) {  // NOSONAR Exception handling is OK
206 			log.error(Markers2.append(ex), "XMLStreamException parsing SOAP Message");
207 			throw new SoapParseException(ex, req == null ? null : req.getClass().getSimpleName());
208 		} catch (BadRequestException ex) {	// NOSONAR Exception handling is OK
209 			log.error(Markers2.append(ex), "Invalid XML input in SOAP Message");
210 			throw new SoapParseException(ex, req == null ? null : req.getClass().getSimpleName());
211 		} catch (SecurityFault sf) {
212 			throw sf;
213 		} catch (Exception ex) {	// NOSONAR Exception handling is OK
214 			log.error(Markers2.append(ex), "Unexpected exception parsing SOAP Message");
215 			throw new SoapParseException(ex, req == null ? null : req.getClass().getSimpleName());
216 		}
217 		req.updateAction(req.isHubMessage());
218 		return req;
219 	}

221 	private void copyEvent() {
222 		if (writer == null) {
223 			return;
224 		}
225 		try {
226 			// Copy the event, ignoring any whitespace nodes.
227 			XmlUtils.copyEvent(reader, writer, true);
228 		} catch (XMLStreamException ignoreThis) {
229 			// Got an exception on the writer, we will ignore it
230 			// and stop processing writer events.
231 			writer = null;
232 		}
233 	}

421 	private static boolean hasCData(int flags) {
422 		return (flags & (1 << XMLStreamConstants.CDATA)) != 0;
423 	}

425 	private <T extends SoapMessage> T updateSoapMessageType(SoapMessage req, Class<T> clazz, String schema) {
426 		T newReq = null;
427 		if (req.getClass() == clazz) {
428 			newReq = clazz.cast(req);
429 		} else if (req.getClass() == SoapMessage.class) {
430 			try {
431 				newReq = clazz.getConstructor(SoapMessage.class, String.class, boolean.class).newInstance(req,
432 						req.getSchema(), true);
433 			} catch (Exception e) {
434 				throw new IllegalArgumentException(
435 						"Class " + clazz.getName() + " does not implement an appropriate constructor.");
436 			}
437 		} else {
438 			throw new BadRequestException("Value not allowed");
439 		}
440 
441 		newReq.setSchema(schema);
442 		return newReq;
443 	}

445 	private String validateText(String elementText) throws SecurityFault {
446 		if (isHtmlMessage) {
447 			return elementText;
448 		}
449 		if (!StringUtils.isEmpty(elementText) && Strings.CI.contains(elementText, TEXT_VALUE_PATTERN)) {
450 			throw SecurityFault.sourceAttack(
451 					"Illegal text value in " + type + " inside: <" + lastElement + "> element", endpoint);
452 		}
453 		return elementText;
454 	}

456 	private void verifyElementName() throws SecurityFault {
457 		String localName = reader.getLocalName();
458 		// Check for an HTML response
459 		if (documentElementName == null) {
460 			documentElementName = localName;
461 			if (localName.equalsIgnoreCase("html")) {
462 				isHtmlMessage = true;
463 			}
464 		}
465 		// If this is an HTML response, don't check for tags or attributes
466 		if (isHtmlMessage) {
467 			return;
468 		}
469 		// Check element name for illegal values.
470 		if (Strings.CI.contains(localName, TAG_NAME_PATTERN)) {
471 			throw SecurityFault.sourceAttack(
472 					"Illegal element name <" + localName + "> found in the " + type + " at: <" + lastElement,
473 					endpoint);
474 		}
475 		verifyAttributes();
476 	}
477 
478 	private void verifyAttributes() throws SecurityFault {
479 		if (isHtmlMessage) {
480 			return;
481 		}
482 		// Check attribute values
483 		int count = reader.getAttributeCount();
484 		while (count-- > 0) {
485 			String text = reader.getAttributeValue(count);
486 			if (Strings.CI.contains(text, TEXT_VALUE_PATTERN)) {
487 				throw SecurityFault.sourceAttack("Illegal attribute value in " + type + " at: <" + lastElement
488 						+ " " + toNameString(reader.getAttributeName(count)) + "=", endpoint);
489 			}
490 		}
491 	}

506 	private int verifyAndSet(Object holder, Supplier<String> value, Consumer<String> setter)
507 			throws XMLStreamException, SecurityFault {
508 		verifyAttributes();
509 		return verifyAndSet(holder, value, setter, false);
510 	}

512 	private int verifyAndSet(Object holder, Supplier<String> value, Consumer<String> setter, boolean filter)
513 			throws XMLStreamException, SecurityFault {
514 		if (holder != null) {
515 			Pair<String, Integer> v = getElementText(filter);
516 			verifyAndSet(value, setter, v.getLeft());
517 			return v.getRight();
518 		} else {
519 			throw new BadRequestException("Misplaced " + reader.getLocalName() + " element");
520 		}
521 	}
522 	
523 	private String verifyAndSet(Supplier<String> value, Consumer<String> setter, String t) throws SecurityFault {
524 		if (value != null) {
525 			verifyEmpty(value.get());
526 		}
527 		String text = validateText(t);
528 		setter.accept(text);
529 		return text;
530 	}

532 	private Pair<String, Integer> getElementText(boolean filter) throws XMLStreamException {
533 		String text = null;
534 		try {
535 			if (writer != null) {
536 				XmlUtils.copyStartElement(reader, writer);
537 			}
538 		} catch (XMLStreamException ignore) {
539 			// If the write failed, ignore it and write no more.
540 			writer = null;
541 		}
542 		// Don't catch on read XMLStreamException.
543 		Pair<String, Integer> parsedText = getElementText(); 
544 		
545 		try {
546 			if (writer == null) {
547 				return parsedText;
548 			}
549 			if (hasCData(parsedText.getRight())) {
550 				writer.writeCData(filter ? SoapMessageWriter.HIDDEN : text);
551 			} else {
552 				writer.writeCharacters(filter ? SoapMessageWriter.HIDDEN : text);
553 			}
554 			writer.writeEndElement();
555 		} catch (XMLStreamException ignore) {
556 			// If the write failed, ignore it and write no more.
557 			writer = null;
558 		}
559 		return parsedText;
560 	}

562 	private Pair<String, Integer> getElementText() throws XMLStreamException {
563 
564         if (reader.getEventType() != XMLStreamConstants.START_ELEMENT) {
565             throw new XMLStreamException(
566                     "parser must be on START_ELEMENT to read next text", reader.getLocation());
567         }
568         int eventType = reader.next();
569         StringBuilder content = new StringBuilder();
570         int flags = 0;
571         while (eventType != XMLStreamConstants.END_ELEMENT) {
572         	flags |= 1 << eventType;
573             if (eventType == XMLStreamConstants.CHARACTERS
574                     || eventType == XMLStreamConstants.CDATA
575                     || eventType == XMLStreamConstants.SPACE
576                     || eventType == XMLStreamConstants.ENTITY_REFERENCE) {
577                 content.append(reader.getText());
578             } else if (eventType == XMLStreamConstants.PROCESSING_INSTRUCTION
579                     || eventType == XMLStreamConstants.COMMENT) {
580                 // skipping
581             } else if (eventType == XMLStreamConstants.END_DOCUMENT) {
582                 throw new XMLStreamException(
583                         "unexpected end of document when reading element text content");
584             } else if (eventType == XMLStreamConstants.START_ELEMENT) {
585                 throw new XMLStreamException("elementGetText() function expects text "
586                         + "only elment but START_ELEMENT was encountered.", reader.getLocation());
587             } else {
588                 throw new XMLStreamException(
589                         "Unexpected event type " + eventType, reader.getLocation());
590             }
591             eventType = reader.next();
592         }
593         return Pair.of(content.toString(), flags);
594     }

596 	private static String toNameString(QName name) {
597 		if (name == null) {
598 			return null;
599 		}
600 		String prefix = name.getPrefix();
601 		if (StringUtils.isEmpty(prefix)) {
602 			return name.getLocalPart();
603 		}
604 		return name.getPrefix() + ":" + name.getLocalPart();
605 	}
