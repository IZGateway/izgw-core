16  /**
17   * The LogController provides access to in memory logging data on a server.
18   * This is used for integration testing to verify log content is as expected
19   * when sending messages.
20   */
21  
22  // TODO: Presently, blacklisted users are allowed to access the logs request, b/c blacklisting only
23  // applies to the SOAP Stack.  Once we apply it to the full HTTP stack, we will have to provide
24  // SECURE mechanism for clearing the blacklisted state of the testing user.  It cannot be said
25  // to have been applied to the full stack until this loophole is resolved.
26  public class LogControllerBase {
27  
28  	protected LogControllerBase() {
29  	}
30  
31  	// TODO: Presently, blacklisted users are allowed to access the logs request, b/c blacklisting only
32  	// applies to the SOAP Stack.  Once we apply it to the full HTTP stack, we will have to provide
33  	// SECURE mechanism to clearing the state.
34  	protected List<LogEvent> getLogs(String search) {
35  		MemoryAppender logData = MemoryAppender.getInstance("memory");
36  
37  		List<ILoggingEvent> events;
38  		if (logData == null) {
39  			events = Collections.emptyList();
40  		} else if (StringUtils.isBlank(search)) {
41  			events = logData.getLoggedEvents();
42  		} else {
43  			events = logData.search(search);
44  		}
45  
46  		return new ListConverter<>(events, LogEvent::new);
47  	}
48  
49  	/**
50  	 * Delete the logs
51  	 * @param servletReq	The servlet request
52  	 * @param clear		The clear parameter
53  	 * @throws SecurityFault	If the user is not authorized
54  	 */
55  	public void deleteLogs(HttpServletRequest servletReq, String clear) throws SecurityFault {
56          if (!RequestContext.getRoles().contains(Roles.ADMIN) && !RequestContext.getRoles().contains(Roles.OPERATIONS)) {
57              throw SecurityFault
58                      .generalSecurity("Delete Log Attempt By Role",
59                              RequestContext.getRoles().toString(), null);
60          }
61  
62  		MemoryAppender logData = MemoryAppender.getInstance("memory");
63  		if (logData != null) {
64  			logData.reset();
65  		}
66  	}
67  
68  }
