149 	 * Log the start content for sending a SOAP Message
150 	 * @param soapMessage	The message being sent
151 	 */
152 	public void logStartOfRequest(SoapMessage soapMessage) {
153 		TransactionData tData = RequestContext.getTransactionData();
154 		String message;
155 		if (soapMessage != null) {
156 			soapMessage.updateAction(isHubWsdl());
157 			if (soapMessage instanceof SubmitSingleMessageRequest s) {
158 				tData.setMessageType(MessageType.SUBMIT_SINGLE_MESSAGE);
159 				message = s.getHl7Message();
160 				tData.getSource().setType(getSourceType(message));
161 				tData.setRequestHL7Message(message);
162 			} else if (soapMessage instanceof ConnectivityTestRequest c) {
163 				tData.setMessageType(MessageType.CONNECTIVITY_TEST);
164 				message = c.getEchoBack();
165 				tData.setRequestEchoBack(message);
166 			} else {
167 				tData.setMessageType(MessageType.OTHER);
168 			}
169 
170 			if (soapMessage instanceof Request) {
171 				MessageInfo messageInfo = new MessageInfo(soapMessage, EndpointType.SERVER, Direction.INBOUND, AppProperties.isProduction());
172 				tData.setMessageId(soapMessage.getWsaHeaders().getMessageID());
173 				tData.setReplyTo(soapMessage.getWsaHeaders().getRelatesTo());
174 				tData.setResponseReceived(false);
175 				tData.getServerRequest().setWs_request_message(messageInfo);
176 				messageInfo.setHttpHeaders(RequestContext.getHttpHeaders());
177 			}
178 			if (soapMessage instanceof HasFacilityID hfid) {
179 				tData.getSource().setFacilityId(hfid.getFacilityID());
180 			}
181 		}
182 		tData.setServerMode(mode);
183 		tData.setServiceType(serviceType);
184 	}

186 	private String getSourceType(String message) {
187 		HL7Message m = new HL7Message(message);
188 		String[] mshVals = {
189 				m.getField(HL7Message.SENDING_APPLICATION),
190 				m.getField(HL7Message.SENDING_FACILITY)
191 		};
192 		if (mshService == null) {
193 			return "unknown";
194 		}
195 		return mshService.getSourceType(mshVals);

198 	protected <T> ResponseEntity<T> logEndOfRequest(Work<ResponseEntity<T>> doWork) throws Fault {
199 		Fault error = null;
200 		try {
201 			ResponseEntity<T> result = doWork.call();
202 			
203 			if (result != null && result.getBody() instanceof SoapMessage body) {
204 				catchAndKillNonTestMessages();
205 				logResponseMessage(body);
206 			}
207 			return result;
208 		} catch (Fault f) {
209 			error = f;
210 			logFault(error);
211 			throw f;
212 		} catch (Throwable e) { // NOSONAR Catch any Throwable is intentional
213 			error = wrapExceptionAsFault(e);
214 			logFault(error);
215 			throw e;
216 		}
217 	}

237 	/**
238 	 * Log data from any fault.
239 	 * @param fault	The fault
240 	 * @return	The fault
241 	 */
242 	public static Fault logFault(Fault fault) {
243 		if (lastLoggedFault.get() == fault) {
244 			// Already logged this fault, avoid duplicate logging
245 			return fault;
246 		}
247 		lastLoggedFault.set(fault);
248 		if (fault instanceof HasDestinationUri dcf) {
249 			log.error(Markers2.append(fault, "destinationUrl", dcf.getDestinationUri()), "{} occured during processing: {}", fault.getClass().getSimpleName(), getCauseMessage(fault));
250 		} else {
251 			log.error(Markers2.append(fault), "{} occured during processing: {}", fault.getClass().getSimpleName(), getCauseMessage(fault));
252 		}
253 		TransactionData tData = RequestContext.getTransactionData();
254 		if (tData == null) {
255 			// This should never happen, protect against it and log an error.
256 			tData = RequestContext.init();
257 			log.error("LogFault called without TransactionData present");
258 		}
259 		tData.setProcessError(fault);
260 		return fault;
261 	}

315 	@ApiResponse(responseCode = "200", description = "The WSDL or requested schema", content = @Content(mediaType = MediaType.APPLICATION_XML_VALUE, schema = @Schema(implementation = IMessageHeader.Map.class)))
316 	@ApiResponse(responseCode = "404", description = "The requested schema was not found", content = @Content)
317 	@ApiResponse(responseCode = "500", description = "A Fault occurred while processing", content = @Content)
318 	@GetMapping(produces =

327 	public ResponseEntity<?> getWsdlOrXsd(  // NOSONAR Intentional use of ?
328 											@Schema(description="Get the WSDL")
329 											@RequestParam(name="WSDL", required=false)
330 											String wsdl,
331 											@Schema(description="Alternate request to get the WSDL")
332 											@RequestParam(name="wsdl", required=false)
333 											String wsdl2,
334 											@Schema(description="The Schema to get")
335 											@RequestParam(name="xsd", required=false)
336 											String xsd,
337 											@Schema(description="Throws the fault specified in the header parameter")
338 											@RequestHeader(value="X-IIS-Hub-Dev-Action", required=false)
339 											String devAction
340 	) {
341 		Fault fault = null;
342 		logStartOfRequest(null);
343 		try {
344 			if (!StringUtils.isEmpty(devAction)) {
345 				throwSimulatedFault(devAction, null);
346 			}
347 			wsdl = ObjectUtils.getIfNull(wsdl, wsdl2);
348 			if (wsdl == null && StringUtils.isEmpty(xsd)) {
349 				throw new UnsupportedOperationFault("At least one of the WSDL or xsd parameters must be present", null);
350 			} else if (xsd != null && !XSD_FILES.contains(xsd)) {
351 				throw new UnsupportedOperationFault("Schema " + xsd + " unknown", null);
352 			}
353 			
354 			String resourceName = wsdl != null ? "/soap/wsdl/" + getWsdl() : ("/soap/schema/" + xsd);
355 			return logEndOfRequest(() -> getResource(resourceName));
356 		} catch (Fault ex) {
357 			fault = ex;
358 		} catch (Throwable ex) {
359 			fault = wrapExceptionAsFault(ex);
360 		}
361 		return handleFault(fault);
362 	}

383 	public ResponseEntity<?> submitSoapRequest( // NOSONAR ? is intentional here
384 		@RequestBody SoapMessage soapMessage,
385 		@Schema(description="Throws the fault specified in the header parameter")
386 		@RequestHeader(value="X-IIS-Hub-Dev-Action", required=false)
387 		String devAction
388 	) throws SecurityFault {
389 		logStartOfRequest(soapMessage);
390 		
391 		Fault fault;
392 		try {
393 			catchAndKillNonTestMessages();
394 			String destinationId = getDestinationId(soapMessage);
395 			if (!StringUtils.isEmpty(devAction)) {
396 				throwSimulatedFault(devAction, destinationId);
397 			}
398 			
399 			if (soapMessage instanceof SubmitSingleMessageRequest s) {
400 				return logEndOfRequest(() -> submitSingleMessage(s, destinationId));
401 			} else if (soapMessage instanceof ConnectivityTestRequest c) {
402 				return logEndOfRequest(() -> connectivityTest(c, destinationId));
403 			} else {
404 				throw new UnexpectedExceptionFault("Schema Error", "The content did not reflect the expected type. " + JsonUtils.toString(soapMessage), null, RetryStrategy.CORRECT_MESSAGE, null);
405 			}
406 		} catch (Fault ex) {
407 			fault = ex;
408 		} catch (Throwable ex) {
409 			fault = wrapExceptionAsFault(ex);
410 		}
411 		return handleFault(fault);
412 	}

414 	private void catchAndKillNonTestMessages() throws SecurityFault {
415 		if (!isCatchAndKillEnabled()) {
416 			return;
417 		}
418 		if (RequestContext.getTransactionData().isProd()) {
419 			return;
420 		}
421 		if (RequestContext.getTransactionData().isKnownTestMessage() ) {
422 			return;
423 		}
424 		// Message does not match known test cases.
425 		throw SecurityFault.generalSecurity("Unknown Test Case", null, null);
426 	}

438 	private void throwSimulatedFault(String devAction, String destinationId) throws Fault {
439 
440 		IDestination destination = null;
441 		try {
442 			if (destinationId == null) {
443 				destination = getDestinationService().getExample(destinationId);
444 			} else {
445 				destination = getDestination(destinationId);
446 			}
447 		} catch (UnknownDestinationFault f) {
448 			if (UnknownDestinationFault.FAULT_NAME.equals(devAction)) {
449 				throw f;  // Good enough!
450 			}
451 			destination = getDestinationService().getExample(destinationId);
452 		}
453 
454 		switch (devAction) {
455 			case DestinationConnectionFault.FAULT_NAME:
456 				throw DestinationConnectionFault.devAction(destination);
457 			case HubClientFault.FAULT_NAME:
458 				throw HubClientFault.devAction(destination);
459 			case MessageTooLargeFault.FAULT_NAME:
460 				throw MessageTooLargeFault.devAction();
461 			case UnexpectedExceptionFault.FAULT_NAME:
462 				throw UnexpectedExceptionFault.devAction();
463 			case UnknownDestinationFault.FAULT_NAME:
464 				throw UnknownDestinationFault.devAction(destinationId);
465 			case UnsupportedOperationFault.FAULT_NAME:
466 			default:
467 				throw UnsupportedOperationFault.devAction();
468 		}
469 	}

475 	private String getDestinationId(SoapMessage soapMessage) {
476 		String destinationId = null;
477 		if (!isUsingHubHeaders()) {
478 			soapMessage.getHubHeader().clear();  // Ignore any hub header values.
479 		} else {
480 			destinationId = soapMessage.getHubHeader().getDestinationId();
481 			if (!StringUtils.isEmpty(destinationId)) {
482 				return destinationId;
483 			}
484 		}
485 		// If there is not hub request header, consider the wsa:To parameter
486 		String toDest = soapMessage.getWsaHeaders().getTo();
487 		// Ensure wsa:To meets the destination id pattern and return what we were given
488 		if (StringUtils.isEmpty(toDest) || !toDest.matches(IDestination.ID_PATTERN)
489 		) {
490 			return destinationId;
491 		}
492 		return toDest;
493 	}

524 	protected void checkMessage(SoapMessage soapMessage) throws MessageTooLargeFault, UnexpectedExceptionFault, SecurityFault {
525 		checkMessageSize(soapMessage, MessageTooLargeFault.Direction.REQUEST);
526 
527 		// A "feature" of the new marshalling structure is that systems don't care WHICH version of the schema you
528 		// use, 2011, 2014, or Hub.  So, we check for a mismatch from the expected values.
529 		if (!isValidMessage(soapMessage)) {
530 			throw new UnexpectedExceptionFault("Schema Error", "The input does not match the requirements of the " + getMessageNamespace() + " schema.", null, RetryStrategy.CORRECT_MESSAGE, null);
531 		}
532 		if (soapMessage instanceof HasCredentials cred) {
533 			checkCredentials(cred);
534 		}
535 	}

567 		String result = resourceCache.get(path);
568 		if (result == null) {
569 			try {
570 				result = IOUtils.resourceToString(path, StandardCharsets.UTF_8);
571 				result = result.replace("{{SERVER.NAME}}", serverName);
572 				// Need to substitute dev.phiz-project.org in WSDL with correct hostname

576 				throw new UnsupportedOperationFault("Resource " + path + " unknown", ex);
577 			}
578 			resourceCache.put(path, result);
579 		}
580 		return new ResponseEntity<>(result, HttpStatus.OK);

591 	@ExceptionHandler(SoapConversionException.class)
592 	protected ResponseEntity<FaultMessage> handleBadXML(SoapConversionException ex) {
593 		Fault f = null;
594 		
595 		logStartOfRequest(null);
596 		if (ex.getCause() instanceof SoapParseException) {
597 			f = new UnexpectedExceptionFault("Syntax Error", null, ex, RetryStrategy.CORRECT_MESSAGE, "An exception occurred parsing the SOAP Message");
598 		} else if (ex.getCause() instanceof SecurityFault sf) {
599 			f = sf;
600 		} else {
601 			f = new UnexpectedExceptionFault("Syntax Error", null, ex, RetryStrategy.CORRECT_MESSAGE, "An exception occurred processing the SOAP Message");
602 		}
603 		return handleFault(f);
604 	}
