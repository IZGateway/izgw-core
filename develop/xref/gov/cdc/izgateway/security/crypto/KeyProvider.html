3   import java.util.List;
4   
5   /**
6    * Interface for managing cryptographic keys and key history.
7    * <p>
8    * This provider handles the loading and management of encryption keys,
9    * maintaining a history of keys to support key rotation scenarios where
10   * data encrypted with older keys may still need to be decrypted.
11   * </p>
12   *
13   * @author CDC IZ Gateway Team
14   * @since 1.0
15   */
16  interface KeyProvider {
17  
18      /**
19       * Loads the current active encryption key.
20       * <p>
21       * This method returns the primary key that should be used for new
22       * encryption operations. The implementation should ensure the key
23       * is valid and properly formatted for the intended cryptographic use.
24       * </p>
25       *
26       * @return the current active key as a byte array
27       * @throws CryptoException if the key cannot be loaded or is invalid
28       */
29      byte[] loadKey() throws CryptoException;
30  
31      /**
32       * Retrieves all available keys including the current key and historical keys.
33       * <p>
34       * This method returns a list of all keys that can be used for decryption,
35       * supporting scenarios where data was encrypted with older keys that are
36       * still valid for decryption but no longer used for new encryption.
37       * </p>
38       *
39       * @return an immutable list of all available keys as byte arrays,
40       *         with the current key typically being the first element
41       */
42      List<byte[]> getAllKeys();
43  
44      /**
45       * Checks if a specific key exists.
46       * <p>
47       * This method can be used to verify if a particular key is known
48       * to the provider before attempting to use it for cryptographic operations.
49       * </p>
50       *
51       * @param keyBytes the key to check for existence
52       * @return {@code true} if the key exists in the provider, {@code false} otherwise
53       * @throws IllegalArgumentException if keyBytes is null
54       */
55      boolean keyExists(byte[] keyBytes);
56  
57      /**
58       * Adds a key to the historical key storage.
59       * <p>
60       * This method is typically called during key rotation to preserve
61       * older keys for decryption purposes. The key will be available
62       * through {@link #getAllKeys()} but should not be used for new
63       * encryption operations.
64       * </p>
65       *
66       * @param keyBytes the key to add to the historical key store
67       * @throws IllegalArgumentException if keyBytes is null or invalid
68       */
69      void addKeyToHistory(byte[] keyBytes);
70  }
