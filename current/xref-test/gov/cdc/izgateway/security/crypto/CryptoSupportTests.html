8   class CryptoSupportTests {
9   
10      private static final String TEST_PLAINTEXT = "Hello, World!";
11      private TestKeyProvider testKeyProvider;
12  
13      @BeforeEach
14      void setUp() {
15          CryptoSupport.initialize();
16  
17          // Inject test key provider
18          testKeyProvider = new TestKeyProvider();
19          CryptoSupport.setKeyProvider(testKeyProvider);
20      }
21  
22      @Test
23      void testEncryptDecrypt() throws CryptoException {
24          byte[] key = testKeyProvider.loadKey();
25          String encrypted = CryptoSupport.encrypt(TEST_PLAINTEXT, key);
26          String decrypted = CryptoSupport.decrypt(encrypted);
27  
28          assertEquals(TEST_PLAINTEXT, decrypted, "Decrypted text should match original");
29      }
30  
31      @Test
32      void testEncryptDecryptWithKeyRotation() throws CryptoException {
33          // Encrypt with initial key
34          byte[] originalKey = testKeyProvider.loadKey();
35          String encrypted = CryptoSupport.encrypt(TEST_PLAINTEXT, originalKey);
36          String decrypted = CryptoSupport.decrypt(encrypted);
37          assertEquals(TEST_PLAINTEXT, decrypted, "Initial decryption should work");
38  
39          // Rotate the key
40          TestKeyProvider.rotateKey();
41          byte[] newKey = testKeyProvider.loadKey();
42          assertFalse(java.util.Arrays.equals(originalKey, newKey), "New key should differ from original");
43  
44          // Encrypt with new key
45          String encryptedWithNewKey = CryptoSupport.encrypt(TEST_PLAINTEXT, newKey);
46          String decryptedWithNewKey = CryptoSupport.decrypt(encryptedWithNewKey);
47          assertEquals(TEST_PLAINTEXT, decryptedWithNewKey, "Decryption with new key should work");
48  
49          // Verify old encrypted data can still be decrypted (key history)
50          String decryptedOld = CryptoSupport.decrypt(encrypted);
51          assertEquals(TEST_PLAINTEXT, decryptedOld, "Old encrypted data should still be decryptable");
52      }
53  
54      @Test
55      void testEncryptWithEmptyInput() throws CryptoException {
56          byte[] key = testKeyProvider.loadKey();
57          String encrypted = CryptoSupport.encrypt("", key);
58          String decrypted = CryptoSupport.decrypt(encrypted);
59          assertEquals("", decrypted, "Empty string should encrypt and decrypt correctly");
60      }
