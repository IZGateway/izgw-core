1   package gov.cdc.izgateway.security;
2   
3   import static org.junit.jupiter.api.Assertions.assertEquals;
4   
5   import org.junit.jupiter.api.BeforeEach;
6   import org.junit.jupiter.api.Test;
7   import org.springframework.web.bind.annotation.RequestMethod;
8   
9   import java.util.List;
10  
11  /**
12   * Pins the roles the registry resolves for {@link AccessController} endpoints (IGDD-3499).
13   * <p>
14   * The deny list and ADS file type endpoints are routine operational tasks and must stay
15   * reachable by OPERATIONS in addition to ADMIN. Every other endpoint on this controller is
16   * unchanged and must remain ADMIN-only, since that is the class-level default.
17   */
18  class AccessControlRegistryRoleTests {
19      private static final List<String> ADMIN_AND_OPERATIONS = List.of(Roles.ADMIN, Roles.OPERATIONS);
20      private static final List<String> ADMIN_ONLY = List.of(Roles.ADMIN);
21  
22      private AccessControlRegistry registry;
23  
24      @BeforeEach
25      void setUp() {
26          registry = new AccessControlRegistry();
27          registry.register(AccessController.class, "/rest");
28      }
29  
30      @Test
31      void denyListEndpointsAllowAdminAndOperations() {
32          assertEquals(ADMIN_AND_OPERATIONS, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/denylist"));
33          assertEquals(ADMIN_AND_OPERATIONS, registry.getAllowedRoles(RequestMethod.POST, "/rest/access/denylist"));
34          assertEquals(ADMIN_AND_OPERATIONS, registry.getAllowedRoles(RequestMethod.DELETE, "/rest/access/denylist"));
35      }
36  
37      @Test
38      void adsFileTypeEndpointAllowsAdminAndOperations() {
39          assertEquals(ADMIN_AND_OPERATIONS, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/filetypes"));
40      }
41  
42      @Test
43      void otherAccessControlEndpointsRemainAdminOnly() {
44          assertEquals(ADMIN_ONLY, registry.getAllowedRoles(RequestMethod.GET, "/rest/access"));
45          assertEquals(ADMIN_ONLY, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/endpoints"));
46          assertEquals(ADMIN_ONLY, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/groups"));
47          assertEquals(ADMIN_ONLY, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/groups/*"));
48          assertEquals(ADMIN_ONLY, registry.getAllowedRoles(RequestMethod.GET, "/rest/access/users"));
49      }
50  }
