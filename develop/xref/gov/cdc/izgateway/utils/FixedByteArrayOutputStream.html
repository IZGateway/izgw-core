1   package gov.cdc.izgateway.utils;

3   import java.io.ByteArrayOutputStream;
4   import java.io.FilterOutputStream;
5   import java.io.IOException;
6   import java.nio.charset.StandardCharsets;
7   
8   /**
9    * A memory based output stream that can only hold so much
10   * before it fails with an IOException.
11   *  
12   */
13  public class FixedByteArrayOutputStream  extends FilterOutputStream {
14  	private final int maxSize;
15  	private final ByteArrayOutputStream bos;
16  	private int currentSize = 0;
17  	public static final int DEFAULT_SIZE = 8192;
18  	
19  	public FixedByteArrayOutputStream(int maxSize) {
20  		super(new ByteArrayOutputStream());
21  		bos = (ByteArrayOutputStream) out;
22  		this.maxSize = maxSize;
23  	}
24  	
25  	public FixedByteArrayOutputStream() {
26  		this(DEFAULT_SIZE);
27  	}
28  	
29  	public class SizeExceededException extends IOException {
30  		private static final long serialVersionUID = 1L;
31  
32  		private SizeExceededException() {
33  			super(String.format("Maximum size of %d exceeded", maxSize));
34  		}
35  		
36  		public int getSize() {
37  			return currentSize;
38  		}
39  	}
40  
41  	@Override
42  	public void write(int b) throws IOException {
43  		if (++currentSize > maxSize) {
44  			throw new SizeExceededException();
45  		}
46  		super.write(b);
47  	}
48  
49  	@Override
50  	public void write(byte[] b, int off, int len) throws IOException {
51  		if ((currentSize += len) > maxSize) {
52  			throw new SizeExceededException();
53  		}
54  		super.write(b, off, len);
55  	}
56  
57  	@Override
58  	public void flush() throws IOException {
59  		super.flush();
60  	}
61  
62  	@Override
63  	public void close() throws IOException {
64  		super.close();
65  	}
66  	
67  	public byte[] toByteArray() {
68  		return bos.toByteArray();
69  	}
70  	
71  	public String toString() {
72  		return bos.toString(StandardCharsets.UTF_8);
73  	}
74  	
75  	public void reset() {
76  		currentSize = 0;
77  		bos.reset();
78  	}
79  
80  }
