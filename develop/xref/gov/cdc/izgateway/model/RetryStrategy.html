1   package gov.cdc.izgateway.model;
2   
3   import org.springframework.http.HttpStatus;
4   
5   /**
6    * Represents the retry strategy for handling message delivery failures in IZ Gateway.
7    * <p>
8    * Provides different strategies for retrying messages, including normal retry, correcting message,
9    * checking IIS status, and contacting support. Each strategy is associated with a message and HTTP status.
10   * </p>
11   */
12  public enum RetryStrategy {
13      /**
14       * A possible transient error. The normal retry strategy would be to retry ONCE immediately,
15       * then fall back ~ 5 minutes before trying again, then again at 10 minutes.  If after 2 to 3
16       * attempts the message still fails, it should be queued somewhere for a final retry attempt
17       * performed several hours in the future (e.g., a nightly retry job, or a retry queue that attempts
18       * to send messages no more frequently than three times spread out through the day).
19       *
20       * On multiple occurences of the same error to the same destination, move to the CHECK_IIS_STATUS
21       * strategy.
22       */
23      NORMAL("Use normal retry strategy", HttpStatus.SERVICE_UNAVAILABLE),
24      /**
25       * The message content itself is the cause of the error.  It must be corrected, and this usually
26       * involves human intervention.  The message should be queued somewhere for a person to inspect and
27       * correct before it is retried.
28       */
29      CORRECT_MESSAGE("Correct message before retrying again", HttpStatus.BAD_REQUEST),
30      /**
31       * The IIS is not responsive for some reason. This may be due to networking infrastructure (Internet)
32       * failures between the IZ Gateway and the IIS, or it may be related to routine or emergency IIS
33       * maintenance.  Check the IIS Status before attempting a retry.  Some errors (e.g. DNS not found,
34       * expired certificates, or invalid response data) will not disappear without human intervention.
35       */
36      CHECK_IIS_STATUS("Check IIS Status before Retry", HttpStatus.BAD_GATEWAY),
37  
38      /**
39       * Some times it doesn't matter what the sender or the receiver does, the problem won't be fixed until the problem
40       * is addressed in the IZ Gateway application or configuration. For these problems, contact IZ Gateway
41       * support, and retry the message after support has addressed the issue.
42       */
43      CONTACT_SUPPORT("Contact support before retrying", HttpStatus.INTERNAL_SERVER_ERROR);
44      private final String message;
45  	private final HttpStatus status;
46      RetryStrategy(String message, HttpStatus status) {
47          this.message = message;
48          this.status = status;
49      }
50      /**
51       * Gets the retry code for this strategy.
52       * @return the retry code as a string
53       */
54      public String getRetryCode() {
55          return Integer.toString(ordinal());
56      }
57      
58      /**
59       * Gets the HTTP status associated with this retry strategy.
60       * @return the HTTP status
61       */
62      public HttpStatus getStatus() {
63      	return status;
64      }
65      /**
66       * Returns a string representation of the retry strategy and its message.
67       * @return the string representation
68       */
69      @Override
70      public String toString() {
71          return super.toString() + ": " + message;
72      }
73      
74      /**
75       * Determines if this retry strategy allows retrying the message.
76       * @return true if retryable, false otherwise
77       */
78      public boolean isRetryable() {
79      	return this == NORMAL || this == CHECK_IIS_STATUS;
80      }
81  }
