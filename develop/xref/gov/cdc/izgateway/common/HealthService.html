1   package gov.cdc.izgateway.common;
2   
3   import gov.cdc.izgateway.logging.event.Health;
4   import gov.cdc.izgateway.logging.markers.Markers2;
5   import lombok.extern.slf4j.Slf4j;
6   
7   @Slf4j
8   public class HealthService {
9   	private HealthService() {}
10  	private static Health health = new Health();
11  
12     //overloaded setHealthy method  that is invoked when there is an exception
13      public static void setHealthy(Throwable ex) {
14          health.setLastException(ex);
15          setHealthy(false, ex.getMessage());
16      }
17  
18      public static void setHealthy(boolean b, String message) {
19      	health.setHealthy(b);
20      	health.setLastChangeReason(message);
21          log.info(Markers2.append("health", health), "Server health changed to {}", health.isHealthy() ? "healthy" : "not healthy");
22      }
23  
24  	public static Health getHealth() {
25          return health.copy();
26      }
27      
28  	public static void incrementVolumes(boolean hasProcessError) {
29  		health.incrementRequestVolume();
30  		if (!hasProcessError) {
31  			health.incrementSuccessVolume();
32  		}
33  	}
34  	public static void setServerName(String serverName) {
35  		health.setServerName(serverName);
36  	}
37  	public static void setBuildName(String build) {
38  		health.setBuildName(build);
39  	}
40  
41  	public static void setDatabase(String url) {
42  		if (health.getDatabase() == null) {
43  			health.setDatabase(url);
44  		} else {
45  			health.setDatabase(health.getDatabase() + ", " + url);
46  		}
47  	}
48  	
49  	/**
50  	 * Set the egress DNS address for this server
51  	 * @param address	The egress DNS address
52  	 */
53  	public static void setEgressDnsAddress(String address) {
54  		health.setEgressDnsAddress(address);
55  	}
56  	
57  	/**
58  	 * Set the ingress DNS address for this server
59  	 * @param address	The ingress DNS address
60  	 */
61  	public static void setIngressDnsAddress(String[] address) {
62  		health.setIngressDnsAddress(address);
63  	}
64  
65  	public static void setAccessControlAction(String accessControlAction) {
66  		health.setAccessControlAction(accessControlAction);
67  		
68  	}
69  	
70  }
