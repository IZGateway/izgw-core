1   package gov.cdc.izgateway.configuration;
2   
3   import java.util.function.Supplier;
4   
5   import lombok.extern.slf4j.Slf4j;
6   import org.springdoc.core.properties.SwaggerUiConfigProperties;
7   import org.springframework.beans.factory.config.BeanPostProcessor;
8   import org.springframework.context.annotation.Bean;
9   import org.springframework.context.annotation.Configuration;
10  import org.webjars.WebJarVersionLocator;
11  
12  /**
13   * Aligns Springdoc's swagger-ui resource-handler version with the actual
14   * {@code org.webjars:swagger-ui} version present on the classpath.
15   *
16   * <p>Springdoc reads its version from {@code springdoc.swagger-ui.version}
17   * and falls back to a value baked into {@code springdoc.config.properties}
18   * inside the springdoc starter jar. {@code izgw-bom} independently overrides
19   * the {@code org.webjars:swagger-ui} version, so the two drift and the UI
20   * 404s on {@code /swagger/swagger-ui/index.html} after a webjar bump.
21   *
22   * <p>Implemented as a {@link BeanPostProcessor} returned from a static
23   * {@code @Bean} factory so it runs even with
24   * {@code spring.main.lazy-initialization=true} (BPPs are always eagerly
25   * registered) and so the version is mutated during the
26   * {@link SwaggerUiConfigProperties} bean's own initialization — before any
27   * other bean (e.g. Springdoc's resource-handler configurer) reads it.
28   *
29   * <p>Lives in {@code izgw-core} (rather than being duplicated per service,
30   * as it originally was in {@code izgw-transform} under IGDD-2976) so every
31   * consumer that component-scans this package inherits the fix automatically.
32   * See IGDD-3084 and the {@code swagger-ui-version-sync} design doc in
33   * {@code izgw-hub}.
34   */
35  @Slf4j
36  @Configuration(proxyBeanMethods = false)
37  public final class SwaggerUiVersionConfig {
38  
39      static final String SWAGGER_UI_WEBJAR_NAME = "swagger-ui";
40  
41      private SwaggerUiVersionConfig() {
42          // Utility-style @Configuration: only a static @Bean factory and static helpers.
43      }
44  
45      @Bean
46      public static BeanPostProcessor swaggerUiVersionAligner() {
47          return new BeanPostProcessor() {
48              @Override
49              public Object postProcessAfterInitialization(Object bean, String beanName) {
50                  if (bean instanceof SwaggerUiConfigProperties props) {
51                      alignVersionFromWebjar(props);
52                  }
53                  return bean;
54              }
55          };
56      }
57  
58      static void alignVersionFromWebjar(SwaggerUiConfigProperties props) {
59          alignVersion(props, () -> new WebJarVersionLocator().version(SWAGGER_UI_WEBJAR_NAME));
60      }
61  
62      static void alignVersion(SwaggerUiConfigProperties props, Supplier<String> detector) {
63          try {
64              String detected = detector.get();
65              if (detected == null || detected.isBlank()) {
66                  log.warn("Could not detect {} webjar version; leaving existing Springdoc swagger-ui version in place", SWAGGER_UI_WEBJAR_NAME);
67                  return;
68              }
69              props.setVersion(detected);
70              log.info("Detected {} webjar version: {}", SWAGGER_UI_WEBJAR_NAME, detected);
71          } catch (Exception | LinkageError e) {
72              log.warn("Failed to detect {} webjar version; leaving existing Springdoc swagger-ui version in place", SWAGGER_UI_WEBJAR_NAME, e);
73          }
74      }
75  }
