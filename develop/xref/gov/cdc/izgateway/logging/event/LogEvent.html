24  /**
25   * Class for documenting the LogEvent format and interfacing with the ILoggingEvent interface used by the
26   * IZ Gateway logging system. The main purpose of this class is to support documentation in Swagger, rather
27   * than marshalling between ILoggingEvent and the appenders that write the output (as Jackson could do that
28   * by itself very easily). However, it also serves for that purpose as well.
29   */
30  @Slf4j
31  @JsonPropertyOrder(
32  	{"@timestamp", "@version", "logger_name", "thread_name", "level", "level_value", "eventId", "sessionId", "message"}
33  )
34  @Schema(description="An entry in logs. Note: Log entries can have additional structured properties "
35  		+ " (e.g., Health or TransactionData).")
36  @JsonInclude(JsonInclude.Include.NON_NULL)
37  public class LogEvent {
38  	private ILoggingEvent event;
39  	private Map<String, Object> properties;
40  	private final Health health;
41  	private final TransactionData transactionData;
42  	public LogEvent(ILoggingEvent event) {
43  		this.event = event;
44  		properties = new LinkedHashMap<>(event.getMDCPropertyMap());
45  		// These properties are removed because they are explicitly accessed
46  		// through named getters on this interface; leaving them in the map
47  		// causes @JsonAnyGetter to serialize them a second time.
48  		properties.remove("eventId");
49  		properties.remove("sessionId");
50  		properties.remove("commonName");
51  		getMarkers(event.getMarkerList(), properties);
52  		// Cache typed values before removing from map to keep explicit getters functional.
53  		Object td = properties.remove("transactionData");
54  		this.transactionData = td instanceof TransactionData ? (TransactionData) td : null;
55  		Object h = properties.remove("health");
56  		this.health = h instanceof Health ? (Health) h : null;
57  	}

143 	public String toString() {
144 		if (event.getMarkerList() != null) {
145 			return event.toString() + " " + event.getMarkerList().toString();
146 		}
147 		return event.toString();
148 	}

150 	private Map<String, Object> getMarkers(Iterable<Marker> markers, Map<String, Object> map) {
151 		if (markers == null) {
152 			return map;
153 		}
154 		for (Marker marker: markers) {
155 			if (marker instanceof LogstashMarker l) {
156 				if (marker instanceof ObjectAppendingMarker o) {
157 					Object value = null;
158 					try {
159 						Method m = ObjectAppendingMarker.class.getDeclaredMethod("getFieldValue");
160 						m.setAccessible(true); // NOSONAR Intentional call to setAccessible
161 						value = m.invoke(o);
162 					} catch (NoSuchMethodException | SecurityException | IllegalAccessException | IllegalArgumentException | InvocationTargetException e) {
163 						log.error("No such method available", e);
164 					}
165 					map.put(o.getFieldName(), value);
166 				}
167 				// LogstashMarker is Iterable<Marker> over its child markers. Iterating a marker
168 				// with no children does nothing, so no check is needed. (Marker.hasReferences()
169 				// and hasChildren() are deprecated in SLF4J 2.x with no replacement.)
170 				getMarkers(l, map);
171 			}
172 		}
173 		return map;
174 	}
