84  	public void afterPropertiesSet() throws IOException {
85  		/* Set JDK Default protocols and cipher suites for client and server connections. */
86  		String p = config.getProtocols().replace("+",",");
87  		System.setProperty("jdk.tls.client.protocols", p);
88  		System.setProperty("jdk.tls.server.protocols", p);
89  		System.setProperty("https.protocols", p);
90  		System.setProperty("jdk.tls.client.cipherSuites", config.getCipherSuites());
91  		System.setProperty("jdk.tls.server.cipherSuites", config.getCipherSuites());
92  		System.setProperty("https.cipherSuites", config.getCipherSuites());
93  
94  		sslContext = getSSLContext();
95  		SSLContext.setDefault(sslContext);
96  		
97  		this.addSslTrustChangedListener(this::getSSLContext);
98  		
99  		tse.scheduleAtFixedRate(
100 			this::updateTrust, 					// Check for trust update needed
101 			config.getMonitoringPeriod(), 		// Initially wait for one monitoring period
102 			config.getMonitoringPeriod(), 		// But then monitoring at usual rate
103 			TimeUnit.SECONDS);					// specified in sections
104 	}

124 	private static void customizeConnector(Connector connector) {
125 		if (!"https".equals(connector.getScheme())) {
126 			return;
127 		}
128 		ProtocolHandler handler = connector.getProtocolHandler();
129 		if (handler instanceof AbstractHttp11Protocol<?> jsse) {
130 			jsse.setSslImplementationName(SSLImplementation.class.getName());
131 		}

146 	/**
147 	 * @return The SSL Context to use for client connections.
148 	 */
149 	public SSLContext getSSLContext() {
150 		boolean reload = checkForUpdates();
151 		if (sslContext != null && !reload) {
152 			return sslContext;
153 		}
154 
155 		try {
156 			KeyStore trustStore = loadTrustStore(reload);
157 			TrustManager[] tm = { getTrustManager(trustStore) };
158 			// Force to TSLv1.2 or higher
159 			sslContext = SSLContext.getInstance("TLSv1.2", BCJSSE);
160 			sslContext.init(getKeyManagerFactory(reload).getKeyManagers(), tm, null);
161 		}  catch (GeneralSecurityException e) {
162 			log.error(Markers2.append(e), "Cannot create SSL Context: {}", e.getMessage());
163 			throw new ServiceConfigurationError(e.getMessage(), e);
164 		}
165 
166 		// Set the default SSL context to use for clients to enable database connectors to use it.
167 		SSLContext.setDefault(sslContext);
168 		HttpsURLConnection.setDefaultHostnameVerifier((a,b) -> true);  // NOSONAR IZGW only uses certificate verification
169 
170 		return sslContext;
171 	}

192 		String cn = checkCertificate(keystore, config.getClientHostname());

199 	private X509ExtendedTrustManager getTrustManager(KeyStore keystore) {
200 		TrustManagerFactory trustMgrFact;
201 		try {
202 			trustMgrFact = TrustManagerFactory.getInstance("PKIX", BCJSSE);
203 			trustMgrFact.init(keystore);
204 		} catch (GeneralSecurityException e) {
205 			log.error(Markers2.append(e), "Cannot create Trust Manager from {}: {}", e.getMessage());
206 			throw new ServiceConfigurationError(e.getMessage(), e);
207 		}
208 		return new RevocationTrustManager((X509ExtendedTrustManager) trustMgrFact.getTrustManagers()[0]);
209 	}

211 	/**
212 	 * Check the private key and certificate for the server.
213 	 * @param keystore The keystore to check
214 	 * @param commonName the Certificate Common Name to find a key for, or null if using the first one.
215 	 * @throws CertificateException
216 	 * @throws KeyStoreException
217 	 */
218 	private String checkCertificate(KeyStore keystore, String commonName) {
219 		try {
220 			Enumeration<String> e = keystore.aliases();
221 			while (e.hasMoreElements()) {
222 				String alias = e.nextElement();
223 
224 				X509Certificate c = (X509Certificate) keystore.getCertificate(alias);
225 				if (StringUtils.isNotEmpty(commonName) && Strings.CS.equals(X500Utils.getCommonName(c), commonName) ||
226 					StringUtils.isEmpty(commonName)
227 				) {
228 					String cn = X500Utils.getCommonName(c);
229 					if (!keystore.isKeyEntry(alias)) {
230 						if (StringUtils.isEmpty(commonName)) {
231 							continue;
232 						}
233 						log.error("No Private Key found for {}", cn);
234 						throw new ServiceConfigurationError("No Private Key found for " + cn);
235 					}
236 					
237 					log.info("Valid Certificate found for {}", cn);
238                     certificate = c;
239                     certificate.checkValidity();
240 					return cn;
241 				} 
242 			}
243 		} catch (KeyStoreException e1) {
244 			log.error(Markers2.append(e1), "Cannot enumerate certificates");
245 			throw new ServiceConfigurationError(e1.getMessage(), e1);
246 		} catch (CertificateExpiredException e1) {
247 			log.error("Certificate for {} has expired.", commonName);
248 			throw new ServiceConfigurationError(e1.getMessage(), e1);
249 		} catch (CertificateNotYetValidException e1) {
250 			log.error("Certificate for {} not yet valid.", commonName);
251 			throw new ServiceConfigurationError(e1.getMessage(), e1);
252 		}
253 		log.error("No certificate found for {}", commonName);
254 		throw new ServiceConfigurationError("No certificate found for " + commonName);
255 	}

257 	private KeyManagerFactory getKeyManagerFactory(boolean reload) {
258 		reload = reload || config.getClientKeyStore().isOutOfDate();
259 		if (keyMgrFact != null && !reload) {
260 			return keyMgrFact;
261 		}
262 		try {
263 			keyMgrFact = KeyManagerFactory.getInstance("PKIX", BCJSSE);
264 			keyMgrFact.init(loadKeyStore(reload), config.getClientKeyStore().getPassword().toCharArray());
265 			return keyMgrFact;
266 		} catch (GeneralSecurityException e) {
267 			log.error(Markers2.append(e), "Cannot create Server Key Manager: {}", e.getMessage());
268 			throw new ServiceConfigurationError(e.getMessage(), e);
269 		}
270 	}

280 	/**
281 	 * Check if the trust material has changed, and if so, update the trust managers
282 	 */
283 	public void updateTrust() {
284 		if (checkForUpdates()) {
285 			for (Runnable trustChangedListener: trustChangedListeners) {
286 				try {
287 					trustChangedListener.run();
288 				} catch (Exception ex) {
289 					log.error("Error updating trust in %s", trustChangedListener);
290 				}
291 			}
292 			log.info("Reloaded key trust information");
293 		}
294 	}

296 	boolean checkForUpdates() {
297 		return config.getClientKeyStore().isOutOfDate() || config.getClientTrustStore().isOutOfDate();
298 	}

302 	 * This method obtains an SNI enabled Server Socket Factory
303 	 * @param location	The URL to obtain a connection for
304 	 * @return	An HttpURLConnection that supports SNI if the URL starts with HTTPS or has no URL Scheme.
305 	 * @throws IOException
306 	 */
307 	public HttpURLConnection getSNIEnabledConnection(URL location) throws IOException {
308 		String protocol = location.getProtocol();
309 		if (protocol == null || !protocol.startsWith("https")) {
310 			// Not HTTPS
311 			return (HttpURLConnection) location.openConnection();
312 		}
313 		
314 		SSLSocketFactory factory = fiddle ? new CapturingSSLSocketFactory(getSSLContext().getSocketFactory()) : getSSLContext().getSocketFactory();
315 		URLConnectionUtil util = new URLConnectionUtil(factory);
316 		HttpsURLConnection conx = (HttpsURLConnection) util.openConnection(location); 
317 		conx.setHostnameVerifier(ClientTlsSupport::verifyHostname);
318 		return conx;
319 	}
