3   import java.io.FilterOutputStream;
4   import java.io.IOException;
5   import java.io.OutputStream;
6   import java.nio.charset.StandardCharsets;
7   import java.util.Arrays;
8   
9   /**
10   * This at present unused class is a FilterOutputStream that preserves the first N bytes of the output
11   * for later reporting/diagnostics, et cetera.
12   */
13  public class PreservingOutputStream extends FilterOutputStream {
14  	byte[] buffer;
15  	String string = null;
16  	int length = 0;
17  	public PreservingOutputStream(OutputStream out, int bufferSize) {
18  		super(out);
19  		if (bufferSize < 1) {
20  			throw new IllegalArgumentException("Buffer size mist be > 0");
21  		}
22  		buffer = new byte[bufferSize];
23  	}
24  	
25  	@Override
26  	public void write(int b) throws IOException {
27  		if (remaining() > 0) {
28  			buffer[length++] = (byte)b;
29  		}
30  		super.write(b);
31  		string = null;
32  	}
33  	
34  	@Override
35  	public void write(byte[] b, int off, int len) throws IOException {
36  		if (remaining() > 0) {
37  			int toCopy = Math.max(remaining(), len);
38  			System.arraycopy(buffer, length, b, off, toCopy);
39  			length += toCopy;
40  		}
41  		super.write(b, off, len);
42  		string = null;
43  	}
44  	
45  	public byte[] getBytes() {
46  		if (length == buffer.length) { 
47  			return buffer;
48  		}
49  		return Arrays.copyOf(buffer, length);
50  	}
51  	
52  	private int remaining() {
53  		return buffer.length - length;
54  	}
55  	
56  	@Override
57  	public String toString() {
58  		if (string == null) {
59  			string = new String(getBytes(), StandardCharsets.UTF_8);
60  		}
61  		return string;
62  	}
63  }
