5   /**
6    * Test implementation of the KeyProvider interface for unit testing and development.
7    * <p>
8    * This provider generates deterministic test keys that can be rotated to simulate
9    * key rotation scenarios in a controlled testing environment. It cycles through
10   * three different keys based on an internal call counter, allowing tests to
11   * verify encryption/decryption behavior across multiple key versions.
12   * </p>
13   * <p>
14   * <strong>Warning:</strong> This provider uses predictable, hardcoded keys and
15   * should NEVER be used in production environments. It is intended solely for
16   * unit testing and development purposes.
17   * </p>
18   *
19   * @author CDC IZ Gateway Team
20   * @since 1.0
21   */
22  class TestKeyProvider extends KeyProviderBase implements KeyProvider {
23      private static int callCount = 0;
24      private static final String BASE_KEY = "a1b2c3d4e5f6789012345678901234567890abcdefabcdef123456789012345";
25  
26      static void rotateKey() {
27          // This method can be used to reset the call count if needed
28          callCount++;
29      }
30  
31      @Override
32      public byte[] loadKey() throws CryptoException {
33          int suffix = (callCount % 3);
34  
35          String key = BASE_KEY + suffix;
36  
37          return Hex.decode(key);
38      }
39  
40  }
