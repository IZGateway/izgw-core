3   import java.util.concurrent.ConcurrentHashMap;
4   
5   public class SimpleTokenStore {
6   
7       private final ConcurrentHashMap<String, AccessToken> tokens = new ConcurrentHashMap<>();
8   
9       public AccessToken createToken() {
10          return save(new AccessToken());
11      }
12  
13      protected AccessToken save(AccessToken token) {
14          tokens.put(token.getAccessToken(), token);
15          tokens.put(token.getRefreshToken(), token);
16          return token;
17      }
18      /**
19       * Remove all saved tokens in a token store.
20       * Used to erase any memory of previously retrieved tokens for debugging purposes.
21       */
22      public void clearTokens() {
23      	tokens.clear();
24      }
25      public AccessToken getAccess(String accessToken2) throws OAuthException {
26          AccessToken tok = tokens.get(accessToken2);
27          if (tok == null) {
28              throw new OAuthException("invalid_grant", "Invalid access token");
29          } else if (!tok.getAccessToken().equals(accessToken2)) {
30              throw new OAuthException("invalid_grant", "Not an access token");
31          }
32          if (tok.getAccessToken().equals(accessToken2) && tok.isValid()) {
33              return tok;
34          }
35          throw new OAuthException("invalid_grant", "Access token has expired");
36      }
37  
38      public AccessToken getRefresh(String refreshToken2) throws OAuthException {
39          AccessToken tok = tokens.get(refreshToken2);
40          if (tok == null) {
41              throw new OAuthException("invalid_grant", "Invalid refresh token");
42          } else if (!tok.getRefreshToken().equals(refreshToken2)) {
43              throw new OAuthException("invalid_grant", "Not a refresh token");
44          }
45          if (tok.getRefreshToken().equals(refreshToken2) && tok.isRefreshable()) {
46              return tok;
47          }
48          throw new OAuthException("invalid_grant", "Refresh token has expired");
49      }

51  }
