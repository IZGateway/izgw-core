1   package gov.cdc.izgateway.soap.message;
2   
3   import java.util.Arrays;
4   import java.util.List;
5   
6   import com.fasterxml.jackson.annotation.JsonIgnore;
7   
8   import gov.cdc.izgateway.common.HasDestinationId;
9   import gov.cdc.izgateway.common.HasDestinationUri;
10  import gov.cdc.izgateway.soap.fault.DestinationConnectionFault;
11  import gov.cdc.izgateway.soap.fault.Fault;
12  import gov.cdc.izgateway.soap.fault.HubClientFault;
13  import gov.cdc.izgateway.soap.fault.MessageTooLargeFault;
14  import gov.cdc.izgateway.soap.fault.UnknownDestinationFault;
15  import lombok.Data;
16  import lombok.EqualsAndHashCode;
17  
18  /** Interface for translation of faults to SOAP Messages */
19  @Data
20  @EqualsAndHashCode(callSuper=true)
21  public class FaultMessage extends SoapMessage {
22  	private static final long serialVersionUID = 1L;
23  	protected static final List<String> HUB_FAULTS = Arrays.asList(
24  			HubClientFault.class.getSimpleName(), 
25  			DestinationConnectionFault.class.getSimpleName(), 
26  			UnknownDestinationFault.class.getSimpleName()
27  		);
28  	public FaultMessage(SoapMessage that, String schema, boolean isUpgradeOrSchemaChange) {
29  		super(that, schema, isUpgradeOrSchemaChange);
30  	}
31  	
32  	public FaultMessage(String schema) {
33  		setSchema(schema);
34  	}
35  	
36  	public FaultMessage(Fault fault, String schema) {
37  		setSchema(schema);
38  		this.fault = fault;
39  		// Handle cases common to all faults
40  		this.code = fault.getCode();
41  		this.detail = fault.getDetail();
42  		this.diagnostics = fault.getDiagnostics();
43  		this.faultName = fault.getFaultName();
44  		this.retry = fault.getRetry().name();
45  		this.summary = fault.getSummary();
46  		this.reason = fault.getMessage();
47  		
48  		if (fault instanceof HasDestinationUri furi) {
49  			this.destinationId = furi.getDestinationId();
50  			this.destinationUri = furi.getDestinationUri();
51  		} else if (fault instanceof HasDestinationId fid) {
52  			this.destinationId = fid.getDestinationId();
53  		} 
54  		if (fault instanceof MessageTooLargeFault fmtl) {
55  			this.size = Long.toString(fmtl.getSize());
56  			this.maxSize = Long.toString(fmtl.getMaxSize());
57  		} 
58  		if (fault instanceof HubClientFault fhcf) {
59  			this.original = fhcf.getOriginalBody();
60  			this.statusCode = Integer.toString(fhcf.getStatusCode());
61  		}
62  	}
63  	
64  	// All Faults
65  	@JsonIgnore
66  	private Throwable fault;
67  	private String faultName = "Fault";
68  	private String code;
69  	private String eventId;
70  	private String retry;
71  	private String summary;
72  	private String detail;
73  	private String diagnostics;
74  	private String reason;
75  	
76  	// MessageTooLargeFault
77  	private String size;
78  	private String maxSize;
79  	
80  	// DestinationConnectionFault, UnknownDestinationFault and HubClientFault
81  	private String destinationId;
82  	private String destinationUri;
83  	
84  	// Only for HubClientFault
85  	private String original;
86  	private String statusCode;
87  	
88  	public boolean isHubFault() {
89  		return HUB_FAULTS.contains(getFaultName());
90  	}
91  }
