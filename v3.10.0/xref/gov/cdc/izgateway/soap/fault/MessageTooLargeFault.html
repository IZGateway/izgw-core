1   package gov.cdc.izgateway.soap.fault;
2   
3   import gov.cdc.izgateway.model.RetryStrategy;
4   import lombok.Getter;
5   
6   public class MessageTooLargeFault extends Fault {
7   	private static final long serialVersionUID = 1L;
8       public enum Direction {
9           REQUEST, RESPONSE
10      }
11      public static final String FAULT_NAME = "MessageTooLargeFault";
12      
13      @Getter
14      private final long size;
15      @Getter
16      private final long maxSize; 
17  
18      private static final MessageSupport[] MESSAGE_TEMPLATES = {
19          new MessageSupport(FAULT_NAME, "30", "Request Message Too Large", null,
20              "The Request is too large to process.",
21              RetryStrategy.CORRECT_MESSAGE),
22          new MessageSupport(FAULT_NAME, "31", "Response Message Too Large", null,
23              "The Response is too large to process.",
24              RetryStrategy.CONTACT_SUPPORT)
25      };
26      
27      static {
28      	for (MessageSupport m: MESSAGE_TEMPLATES) {
29      		MessageSupport.registerMessageSupport(m);
30      	}
31      }
32  
33      public MessageTooLargeFault(Direction direction, long maxSize, long actualSize) {
34          super(MESSAGE_TEMPLATES[direction.ordinal()].setDetail(String.format("%d exceeds %d", actualSize, maxSize)));
35          this.size = actualSize;
36          this.maxSize = maxSize;
37      }
38      
39      public static MessageTooLargeFault devAction() {
40      	return new MessageTooLargeFault(Direction.REQUEST, 0, 1);
41      }
42  }
