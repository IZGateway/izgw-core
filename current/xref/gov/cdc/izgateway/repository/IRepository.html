1   package gov.cdc.izgateway.repository;
2   
3   import java.util.Collection;
4   import java.util.List;
5   import java.util.ServiceConfigurationError;
6   
7   /** 
8    * Marker interface for repositories (e.g., DynamoDb, JPA or other database technology)
9    * @author Audacious Inquiry
10   * @param <T> The type of item the repository stores
11   *
12   */
13  public interface IRepository<T> {
14  	/**
15  	 * Get all entities in the repository.
16  	 * @return a List of entities in the repository.
17  	 */
18  	List<T> findAll();
19  
20  	/**
21  	 * Get all entities in the repository for the current environment.
22  	 * @return	a List of entities in the repository for the current environment.
23  	 */
24  	List<T> findAllForEnvironment();
25  	
26  	/**
27  	 * 
28  	 * @param h	The entity to store
29  	 * @return	The stored entity (may be of a different class if from different repositories).
30  	 */
31  	T store(T h);
32  	
33  	/**
34  	 * Copy a list of data from one place to another.
35  	 * @param list	The list to copy
36  	 */
37  	default void migrate(Collection<? extends T> list) {
38  		try {
39  			for (T e: list) {
40  				store(e);
41  			}
42  		} catch (Exception e) {
43  			throw new ServiceConfigurationError("Failed to migrate " + this.getClass().getSimpleName(), e);
44  		}
45  	}
46  	/**
47  	 * Create a new Entity supported by this repository.
48  	 * @return a new entity supported by this repository.
49  	 */
50  	public T createEntity();
51  	
52  	/**
53  	 * Delete an entity from the repository.
54  	 * @param r	The entity to delete
55  	 */
56  	public void delete(T r);
57  }
