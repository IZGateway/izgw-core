40  	 * A destination id.
41  	 * A composite of the destination endpoint identifier, and the environment id (a.k.a., destination type). 
42  	 * @author Audacious Inquiry
43  	 */
44  	@Schema(properties= {
45  		@StringToClassMapItem(key="ak", value=AbstractDestination.class),
46  		@StringToClassMapItem(key="dev", value=AbstractDestination.class),
47  		@StringToClassMapItem(key="dex", value=AbstractDestination.class),

54  	@Data
55  	@AllArgsConstructor
56  	@NoArgsConstructor
57  	public static class DestinationId implements Serializable, IDestinationId {
58  		@Column(name="dest_id")
59  		private String destId;
60  		@Column(name="dest_type")
61  		private int destType;
62  
63  		/**
64  		 * Constructor for a destination id
65  		 * @param destId
66  		 */
67  		public DestinationId(String destId) {
68  			this.destId = destId;
69  			this.destType = SystemUtils.getDestType();
70  		}
71  		/**
72  		 * Create a new destination identifier from an existing one. 
73  		 * @param id
74  		 */
75  		public DestinationId(IDestinationId id) {
76  			this.destId = id.getDestId();
77  			this.destType = id.getDestType();
78  		}
79  		
80  		@Override
81  		public DestinationId copy() {
82  			return new DestinationId(destId, destType);
83  		}
84  
85  		@Override
86  		public void setDestType(String destType) {
87  			int result = SystemUtils.getDestTypeId(destType);
88  			if (result < 1) {
89  				throw new IllegalArgumentException("Destination Type must come from " + SystemUtils.getDestTypes());
90  			}
91  			this.destType = result;
92  		}
93  
94  		@Override
95  		public void setDestType(int destType) {
96  			this.destType = destType;
97  		}
98  		
99  		@Override
100 		public String toString() {
101 			return String.format("%d#%s", destType, destId);
102 		}
103 	}

251 	@Override
252 	@JsonIgnore
253 	public boolean isUnderMaintenance() {
254 		Date now = new Date();
255 		if (!StringUtils.isEmpty(maintReason)) {
256 			return (maintStart == null || now.after(maintStart)) && (maintEnd == null || now.before(maintEnd));
257 		}
258 		return false;
259 	}

261 	@Override
262 	@JsonIgnore
263 	public String getMaintenanceDetail() {
264 		String detail = String.format("Destination %s in %s under maintenance from %tc until ", getDestId(),
265 				SystemUtils.getDestTypeAsString(), getMaintStart());
266 		if (getMaintEnd() != null) {
267 			detail += String.format("%tc", getMaintEnd());
268 		} else {
269 			detail += " further notice.";
270 		}
271 		return detail;
272 	}
