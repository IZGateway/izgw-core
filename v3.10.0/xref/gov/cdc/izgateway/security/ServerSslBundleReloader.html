8   import org.springframework.boot.ssl.SslBundle;

21  import jakarta.annotation.PostConstruct;
22  import lombok.extern.slf4j.Slf4j;

24  /**
25   * Rebuilds and pushes the embedded HTTPS listener's named {@code SslBundle} when its backing
26   * keystore/truststore file changes on disk, by polling file modification time rather than relying
27   * on Spring Boot's own {@code reload-on-update} (inotify-based {@code WatchService}).
28   * <p>
29   * Inotify does not reliably detect a change written by a different client to a network
30   * filesystem -- confirmed for this application's actual deployment, where the keystore lives on
31   * an AWS EFS (NFSv4.1) volume (IGDD-3582). Polling file mtime is the same, already-proven-correct
32   * mechanism this library already uses for the outbound client TLS path
33   * ({@link ClientTlsSupport}/{@code KeyStoreLoader}).
34   * <p>
35   * This class only provides the check-and-reload operation; deciding how often (or whether) to run
36   * it is left to the consuming application -- see {@code ServerSslBundleReloadScheduler} in
37   * izgw-hub -- since polling cadence is a deployment policy decision, not something a shared
38   * library should dictate to every consumer.
39   */
40  @Component
41  @Slf4j
42  public class ServerSslBundleReloader {
43  
44  	private final SslBundleRegistry sslBundleRegistry;
45  	private final SslBundles sslBundles;
46  	private final Environment environment;
47  	private final ResourceLoader resourceLoader;
48  
49  	private volatile long lastSeenModified = 0;
50  
51  	/**
52  	 * Create the reloader.
53  	 * @param sslBundleRegistry The application's SSL bundle registry, used to push a rebuilt
54  	 * bundle. Spring Boot registers one bean implementing both this and {@link SslBundles}.
55  	 * @param sslBundles The application's SSL bundles, used to look up the current bundle (to
56  	 * preserve its key/options/protocol across a reload).
57  	 * @param environment The application's environment, used to resolve {@code server.ssl.bundle}
58  	 * and the named bundle's {@code spring.ssl.bundle.jks.*} properties.
59  	 * @param resourceLoader The resource loader used to resolve keystore/truststore locations,
60  	 * consistent with how Spring Boot's own bundle registrar resolves them.
61  	 */
62  	public ServerSslBundleReloader(@Autowired SslBundleRegistry sslBundleRegistry, @Autowired SslBundles sslBundles,
63  			@Autowired Environment environment, @Autowired ResourceLoader resourceLoader) {
64  		this.sslBundleRegistry = sslBundleRegistry;
65  		this.sslBundles = sslBundles;
66  		this.environment = environment;
67  		// Wraps the webapp-context loader so bare paths (no "file:" prefix, what production
68  		// uses) resolve the same way Spring Boot's own bundle registrar resolves them.
69  		this.resourceLoader = ApplicationResourceLoader.get(resourceLoader, true);
70  	}
71  
72  	/**
73  	 * Captures the bundle's keystore/truststore modification time at startup, so the first
74  	 * scheduled {@link #reloadIfChanged()} call only reacts to a change made after the bundle's
75  	 * own initial load -- not to the initial load itself.
76  	 */
77  	@PostConstruct
78  	void primeLastSeenModified() {
79  		lastSeenModified = currentModified();
80  	}
81  
82  	/**
83  	 * Checks whether the configured bundle's keystore/truststore files have changed since the
84  	 * last call (or startup), and if so, rebuilds the bundle from disk and pushes it into the
85  	 * registry. No-ops when {@code server.ssl.bundle} is not configured -- legacy, non-bundle SSL
86  	 * configuration has no reload path, by design.
87  	 */
88  	public synchronized void reloadIfChanged() {
89  		String bundleName = bundleName();
90  		if (bundleName == null) {
91  			return;
92  		}
93  		long modified = currentModified();
94  		if (modified <= 0 || modified <= lastSeenModified) {
95  			return;
96  		}
97  		try {
98  			SslBundle current = sslBundles.getBundle(bundleName);
99  			String prefix = propertyPrefix(bundleName);
100 			JksSslStoreDetails keyStoreDetails = new JksSslStoreDetails(
101 					environment.getProperty(prefix + ".keystore.type"),
102 					environment.getProperty(prefix + ".keystore.provider"),
103 					environment.getProperty(prefix + ".keystore.location"),
104 					environment.getProperty(prefix + ".keystore.password"));
105 			JksSslStoreDetails trustStoreDetails = new JksSslStoreDetails(
106 					environment.getProperty(prefix + ".truststore.type"),
107 					environment.getProperty(prefix + ".truststore.provider"),
108 					environment.getProperty(prefix + ".truststore.location"),
109 					environment.getProperty(prefix + ".truststore.password"));
110 			JksSslStoreBundle stores = new JksSslStoreBundle(keyStoreDetails, trustStoreDetails, resourceLoader);
111 			SslBundle updated = SslBundle.of(stores, current.getKey(), current.getOptions(), current.getProtocol());
112 			sslBundleRegistry.updateBundle(bundleName, updated);
113 			lastSeenModified = modified;
114 			log.info("Reloaded SSL bundle {} after keystore/truststore change", bundleName);
115 		} catch (Exception | ServiceConfigurationError ex) {
116 			log.error(Markers2.append(ex), "Failed to reload SSL bundle {}: {}", bundleName, ex.getMessage());
117 		}
118 	}
119 
120 	private String bundleName() {
121 		String name = environment.getProperty("server.ssl.bundle");
122 		return StringUtils.hasText(name) ? name : null;
123 	}
124 
125 	private String propertyPrefix(String bundleName) {
126 		return "spring.ssl.bundle.jks." + bundleName;
127 	}
128 
129 	private long currentModified() {
130 		String bundleName = bundleName();
131 		if (bundleName == null) {
132 			return 0;
133 		}
134 		String prefix = propertyPrefix(bundleName);
135 		return Math.max(lastModified(environment.getProperty(prefix + ".keystore.location")),
136 				lastModified(environment.getProperty(prefix + ".truststore.location")));
137 	}
138 
139 	private long lastModified(String location) {
140 		if (!StringUtils.hasText(location)) {
141 			return 0;
142 		}
143 		try {
144 			Resource resource = resourceLoader.getResource(location);
145 			File file = resource.getFile();
146 			return file.exists() ? file.lastModified() : 0;
147 		} catch (Exception ex) {
148 			return 0;
149 		}
150 	}
151 }
