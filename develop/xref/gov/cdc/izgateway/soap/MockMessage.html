224 	/**
225 	 * Constructor for simple mocks that return a fixed message. These fixed
226 	 * messages are to test IZ Gateway behavior that is driven by message
227 	 * content, mostly with regard to logging.
228 	 *
229 	 * @param message
230 	 *            The message to return.
231 	 */
232 	MockMessage(String message) {
233 		SubmitSingleMessageResponse t = new SubmitSingleMessageResponse(
234 				message);
235 		expected = t;
236 		body = m -> new ResponseEntity<SubmitSingleMessageResponse>(t,
237 				HttpStatus.OK);
238 	}

244 	private static String repeatString(MockMessage response, int size) {
245 		String value = ((SubmitSingleMessageResponse) response.getExpected())
246 				.getHl7Message();
247 		StringBuilder newMessage = new StringBuilder();
248 		do {
249 			newMessage.append(value);
250 		} while (size >= newMessage.length());
251 		return newMessage.toString();
252 	}

312 	/**
313 	 * Constructor for mocks that return SOAP Faults of various types.
314 	 *
315 	 * @param t
316 	 *            The SOAP Fault that should appear.
317 	 * @param message
318 	 *            The message explaining the reason for failure.
319 	 * @param mutant
320 	 *            Mutated form to use (make invalid according to schema)
321 	 */
322 	MockMessage(Class<? extends Throwable> t, String message) {
323 		Throwable throwable;
324 		try {
325 			if (SecurityFault.class.isAssignableFrom(t)) {
326 				throwable = SecurityFault.generalSecurity(message, null,
327 						new Exception("Mock Security Fault"));
328 			} else if (UnexpectedExceptionFault.class.isAssignableFrom(t)) {
329 				throwable = new UnexpectedExceptionFault(message,
330 						new Exception("Mock Unexpected Exception Fault"),
331 						"Fault generated by " + this.name());
332 			} else if (MessageTooLargeFault.class.isAssignableFrom(t)) {
333 				throwable = new MessageTooLargeFault(Direction.REQUEST, 1, 0);
334 			} else {
335 				throwable = t.getConstructor(String.class).newInstance(message);
336 			}
337 		} catch (InstantiationException | IllegalAccessException
338 				| IllegalArgumentException | InvocationTargetException
339 				| NoSuchMethodException | SecurityException e) {
340 			throw new MockFault(e);
341 		}
342 		this.body = m -> {
343 			throw new WrappedFault(throwable);
344 		};
345 		this.expected = throwable;
346 	}

348 	/**
349 	 * Get the message response this mock is intended to return.
350 	 *
351 	 * @param message
352 	 *            The SOAP message object, used again to return a non-XML body.
353 	 * @return The mocked response
354 	 * @throws Fault To mock a generic fault
355 	 * @throws SecurityFault
356 	 *             To mock a security fault.
357 	 * @throws MessageTooLargeFault
358 	 *             To mock a MessageTooLargeFault
359 	 * @throws UnsupportedOperationFault
360 	 *             To mock an unsupported operation
361 	 */
362 	public ResponseEntity<?> getMessage(SoapMessage message) // NOSONAR ? is
363 																// intentional
364 			throws Fault {
365 		try {
366 			return body.apply(message);
367 		} catch (WrappedFault ex) {
368 			Throwable cause = ex.getCause();
369 			if (cause instanceof MessageTooLargeFault mtlf) {
370 				throw mtlf;
371 			} else if (cause instanceof SecurityFault secf) {
372 				throw secf;
373 			} else if (cause instanceof UnsupportedOperationFault uof) {
374 				throw uof;
375 			} else if (cause instanceof Fault f) {
376 				throw f;
377 			} else {
378 				throw ex;
379 			}
380 		}
381 	}

414 	 * The function call that the timeout mock uses to sleep beyond IZ Gateway's
415 	 * patience.
416 	 * 
417 	 * @param resp
418 	 *            The HttpServletResponse, not used.
419 	 * @param msg
420 	 *            The Message, also not used.
421 	 * @return An empty response.
422 	 */
423 	private static ResponseEntity<?> forceTimeout(SoapMessage msg) {
424 		try {
425 			Thread.sleep(62000);
426 		} catch (InterruptedException e) {
427 			Thread.currentThread().interrupt();
428 		}
429 		SubmitSingleMessageResponse sr = new SubmitSingleMessageResponse("");
430 		sr.setSchema(msg.getSchema());
431 		return new ResponseEntity<>(sr, HttpStatus.OK);
432 	}

434 	private static ResponseEntity<?> simulateFault(SoapMessage msg)
435 			throws Fault {
436 		throw UnknownDestinationFault
437 				.missingDestination("There is no destination in the request");
438 	}
439 	/**
440 	 * This method fails on odd requests (first, third, et cetera) And succeeds
441 	 * on even requests. If running with a solo client that has sticky sessions,
442 	 * this enables the client to verify retry attempts.
443 	 * 
444 	 * @param resp
445 	 *            The response
446 	 * @param msg
447 	 *            The message
448 	 * @return A response that fails on odd attempts and succeeds on even ones.
449 	 * @throws Fault
450 	 */
451 	private static ResponseEntity<?> retryableFailure(SoapMessage msg)
452 			throws Fault {
453 		// Succeed on even requests (second, fourth, et cetera).
454 		if (incrementRequestCount()) {
455 			return TC_04.body.apply(msg);
456 		}
457 		// Fail on odd requests (first, third, et cetera)
458 		return TC_21D.body.apply(msg);
459 	}
460 
461 	private static boolean incrementRequestCount() {
462 		return ++retryableRequestCount % 2 == 0;
463 	}
