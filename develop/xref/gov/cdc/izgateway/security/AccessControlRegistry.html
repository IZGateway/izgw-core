24  /**
25   * This component acts as a registry of methods that have @RolesAllowed annotations
26   * to support Access controls.
27   * 
28   */
29  @Slf4j
30  @Component
31  public class AccessControlRegistry implements IAccessControlRegistry {
32  	private Map<String, List<String>> allowedMap = new LinkedHashMap<>();
33  	private Map<String, List<String>> specialMap = new LinkedHashMap<>();
34  	private static final String[] ALL_METHODS = { "*" };
35  	
36  	/**
37  	 * Get the access control map
38  	 * @return The access control map
39  	 */
40  	public Map<String, List<String>> getApiEndpoints() {
41  		Map<String, List<String>> result = new TreeMap<>();
42  		
43  		for (Map.Entry<String, List<String>> e: allowedMap.entrySet()) {
44  			for (String group : e.getValue()) {
45  				List<String> users = result.computeIfAbsent(group, k -> new ArrayList<>());
46  				users.add(e.getKey());
47  			}
48  		}
49  		for (Map.Entry<String, List<String>> e: specialMap.entrySet()) {
50  			for (String group : e.getValue()) {
51  				List<String> users = result.computeIfAbsent(group, k -> new ArrayList<>());
52  				users.add(e.getKey());
53  			}
54  		}
55  		return result;
56  	}

59  	/**
60  	 * Register a controller in this registry. Each method and path associated with this controller
61  	 * or methods within this controller that have a RolesAllowed annotation will allow access
62  	 * to paths specified by the Controller or it's methods.  Methods are registered first so that
63  	 * their annotations will take precedence over controller annotations.  Controller annotations 
64  	 * serve as defaults.
65  	 * 
66  	 * @param controller	The @RestController object to register
67  	 */
68  	public void register(Object controller) {
69  		String[] prefixes = getControllerPrefix(controller.getClass());
70  		if (prefixes.length == 0) {
71  			String[] noPrefixes = { "" };
72  			prefixes = noPrefixes;
73  		}
74  		for (String prefix: prefixes) {
75  			register(controller.getClass(), prefix);
76  		}
77  	}
78  	
79  	private String[] getControllerPrefix(Class<?> controller) {
80  		RequestMapping m = controller.getAnnotation(RequestMapping.class);
81  		if (m == null) {
82  			return new String[0];
83  		}
84  		String[] mapping = m.value();
85  		if (mapping.length == 0) {
86  			return new String[0];
87  		}
88  		return mapping;
89  	}

91  	/**
92  	 * Register a controller class in this registry. Does the bulk of the work for above method.
93  	 * This method exists to enable testing without instantiating the class (possibly expensive). 
94  
95  	 * @param controllerClass The class of the @RestController object to register
96  	 * @param prefix	The prefix under which this controller is installed.
97  	 */
98  	public void register(Class<?> controllerClass, String prefix) {
99  		RolesAllowed defaultRoles = controllerClass.getAnnotation(RolesAllowed.class);
100 		RolesAllowed roles = null;
101 		MergedAnnotation<RequestMapping> mapping = null;
102 		
103 		Method[] methods = controllerClass.getMethods();
104 		for (Method method: methods ) {
105 			MergedAnnotations merged = MergedAnnotations.from(method, MergedAnnotations.SearchStrategy.INHERITED_ANNOTATIONS);
106 			mapping = merged.get(RequestMapping.class);
107 			if (mapping.equals(MergedAnnotation.missing())) {
108 				continue;
109 			}
110 			
111 			roles = ObjectUtils.getIfNull(method.getAnnotation(RolesAllowed.class), defaultRoles);
112 			if (roles == null) {
113 				continue;
114 			}
115 
116 			registerPathsAndRoles(prefix, roles, mapping, true);
117 		}
118 	}

121 	/**
122 	 * Register the paths and roles found on the class or one of its methods.
123 	 * @param prefix	Prefix for the controller.
124 	 * @param roles		Roles associated with the method or class.
125 	 * @param mapping	Request Mappings associated with the method or class.
126 	 */
127 	private void registerPathsAndRoles(String prefix, RolesAllowed roles, MergedAnnotation<RequestMapping> mapping, boolean add) {
128 		if (!StringUtils.isEmpty(prefix) && !prefix.endsWith("/")) {
129 			prefix += "/";
130 		}
131 		if (mapping == null) {
132 			return;
133 		}
134 		
135 		String[] paths = mapping.getStringArray("value");
136 		if (paths.length == 0) {
137 			String[] emptyPaths = { "" };
138 			paths = emptyPaths; 
139 		}
140 		
141 		String[] allowed = roles != null ? roles.value() : new String[0];
142 		if (allowed == null || allowed.length == 0) {
143 			return;
144 		}
145 		
146 		RequestMethod[] methods = mapping.getValue("method", RequestMethod[].class).orElse(null);
147 
148 		if (methods != null) {
149 			registerPathsAndRoles(prefix, methods, allowed, paths, add);
150 		} else {
151 			registerPathsAndRoles(prefix, ALL_METHODS, allowed, paths, add);
152 		}
153 	}

156 	 * Dynamically add an access control for a path.
157 	 * @param methods The methods that apply to the path
158 	 * @param path	The path to add the access control for.
159 	 * @param roles The roles to add the access control for.
160 	 */
161 	
162 	public void register(RequestMethod[] methods, String path, String ... roles  ) {
163 		String[] paths = { path };
164 		registerPathsAndRoles(null, methods, roles, paths, true);
165 	}

168 	 * Dynamically remove an access control for a path.
169 	 * @param methods The methods used with the path.
170 	 * @param path	The path to add the access control for.
171 	 * @param roles The roles to add the access control for.
172 	 */
173 	public void unregister(RequestMethod[] methods, String path, String ... roles  ) {
174 		String[] paths = { path };
175 		registerPathsAndRoles(null, methods, roles, paths, false);
176 	}

179 	private void registerPathsAndRoles(String prefix, Object[] methods, String[] allowed, String[] paths, boolean add) {
180 		if (methods == null || methods.length == 0) {
181 			methods = new String[] { "*" };
182 		}
183 		
184 		for (String path: paths) {
185 			if (prefix != null) {
186 				path = prefix + path;
187 			}
188 			if (path.endsWith("/")) {
189 				path = path.substring(0, path.length()-1);
190 			}
191 			boolean hasHead = false;
192 			boolean hasGet = false;
193 			for (Object method: methods) {
194 				String m = method.toString();
195 				updatePathAndMethod(allowed, add, method, path);
196 				hasGet = hasGet || m.equals("GET");
197 				hasHead = hasHead || m.equals("HEAD");
198 			}
199 			// Auto register HEAD if GET is registered.
200 			if (hasGet && !hasHead) {
201 				updatePathAndMethod(allowed, add, "HEAD", path);
202 			}
203 		}
204 	}
205 
206 	private void updatePathAndMethod(String[] allowed, boolean add, Object method, String path) {
207 		String key = getPathParts(method.toString(), path);
208 		if (add) {
209 			if (key.contains("*")) {
210 				specialMap.put(key, Arrays.asList(allowed));
211 			} else {
212 				allowedMap.put(key, Arrays.asList(allowed));
213 			}
214 		} else {
215 			if (key.contains("*")) {
216 				specialMap.remove(key);
217 			} else {
218 				allowedMap.remove(key);
219 			}
220 		}
221 	}

223 	/**
224 	 * Break a path down into its parts.  Replaces path variables in the form of {variable} with *.
225 	 * Returns the list of parts, with any path variables replaced by *.
226 	 * 
227 	 * @param method	The method.
228 	 * @param path	The path to break down.
229 	 * @return	The parts of the path.
230 	 */
231 	private String getPathParts(String method, String path) {
232 		int end = path.length();
233 		if (path.endsWith("/")) {
234 			end--;
235 		}
236 		int start = 0;
237 		if (path.startsWith("/")) {
238 			start++;
239 		}
240 		path = path.substring(start, end);
241 		
242 		path = method + " /" + path.replaceAll("/+", "/");
243 		String[] parts = path.split("/");
244 		for (int i = 0; i < parts.length; i++) {
245 			if (parts[i].startsWith("{")) {
246 				parts[i] = "*";
247 			}
248 		}
249 		return StringUtils.join(parts, "/");
250 	}

252 	/**
253 	 * Get the allowed roles for the specified path.
254 	 * @param path	The path to check.
255 	 * @return	The allowed roles.
256 	 */
257 	public List<String> getAllowedRoles(RequestMethod method, String path) {
258 		String key = getPathParts(method.toString(), path);
259 		List<String> value = allowedMap.get(key);
260 		if (value != null) {
261 			return value;
262 		}
263 		for (Map.Entry<String, List<String>> allowed: specialMap.entrySet()) {
264 			if (pathMatches(key, allowed.getKey())) {
265 				return allowed.getValue();
266 			}
267 		}
268 		log.debug("NO path matching {} on {}", method, path);
269 		return Collections.emptyList();
270 	}

272 	/**
273 	 * Compare two paths.
274 	 * @param testParts	The path to test.  Should NOT have * in it.
275 	 * @param pathParts	The path to test against.  * in it will match any value.
276 	 * @return true if the path matches.
277 	 */
278 	private boolean pathMatches(String testKey, String pathKey) {
279 		// Simplest case, evaluate first.
280 		if (testKey.equals(pathKey)) {
281 			return true;
282 		}
283 		String[] testParts = testKey.split("/");
284 		String[] pathParts = pathKey.split("/");
285 		// Fix space after method
286 		testParts[0] = testParts[0].trim();
287 		pathParts[0] = pathParts[0].trim();
288 		int i;
289 		String pathPart = null;
290 		for (i = 0; i < testParts.length && i < pathParts.length; i++) {
291 			pathPart = pathParts[i];
292 			String testPart = testParts[i];
293 			if (!testPart.equals(pathPart) && !"*".equals(pathPart)) {
294 				break;
295 			}
296 		}
297 		// Consumed all of both
298 		if (i == testParts.length && i == pathParts.length) {
299 			return true;
300 		}
301 		// Last part of pathPart didn't match, but it is **
302 		return i == (pathParts.length-1) && "**".equals(pathPart);
303 	}
