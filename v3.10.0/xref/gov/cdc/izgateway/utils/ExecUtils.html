14  	public static boolean waitFor(long maxDelay, BooleanSupplier t) {
15  		long stopTime = System.currentTimeMillis() + maxDelay;
16  		do {
17  			try {
18  				Thread.sleep(100);
19  			} catch (InterruptedException e) {
20  				Thread.currentThread().interrupt();
21  			}
22  		} while (System.currentTimeMillis() < stopTime && !Boolean.TRUE.equals(t.getAsBoolean()));
23  		return Boolean.TRUE.equals(t.getAsBoolean());
24  	}

27  	public static <T> boolean execAll(List<T> l, Consumer<T> consumer, int wait, TimeUnit units) {
28  		ExecutorService exec = Executors.newFixedThreadPool(Math.max(4, wait));
29  		l.forEach(consumer::accept);
30  		exec.shutdown();
31  		boolean allComplete = false;
32  		try {
33  			allComplete = exec.awaitTermination(wait, units);
34  		} catch (InterruptedException e) {
35  			Thread.currentThread().interrupt();
36  		}
37  		return allComplete;
38  	}
