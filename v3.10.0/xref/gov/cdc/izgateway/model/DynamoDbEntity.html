10  /**
11   * This class is used to mark DynamoDbBean entities which use a single table design for storing data.  
12   * Each entity must implement the primaryId() method, and may override the sortKey method.
13   * 
14   * It assumes that all entities are uniquely identified by their simple class name.
15   *  
16   * @author Audacious Inquiry
17   *
18   */
19  @DynamoDbBean
20  public interface DynamoDbEntity extends DbAudit {
21  	/** The attribute to use for entity type */
22  	public static final String ENTITY_TYPE = "entityType";
23  	/** The attribute to use for the sort key */
24  	public static final String SORT_KEY = "sortKey";
25  
26  	/**
27  	 * Compute the partition key for this object.
28  	 * The partition key is the simple type name of the object + the entity specific sort key.
29  	 * @return The partition key for this object.
30  	 */
31  	@DynamoDbPartitionKey
32  	@DynamoDbAttribute(ENTITY_TYPE)
33  	@JsonIgnore
34  	default String getEntityType() {
35  		return getClass().getSimpleName();
36  	}
37  	
38  	/**
39  	 * Compute the sort key for this object.
40  	 * The sort key is derived from the primary id for the object.
41  	 * @return The sort key for this object.
42  	 */
43  	@DynamoDbSortKey
44  	@DynamoDbAttribute(SORT_KEY)
45  	@JsonIgnore
46  	default String getSortKey() {
47  		return getPrimaryId();
48  	}
49  	
50  	/**
51  	 * Report the primary identifier for the object.
52  	 * This will be used as the sort key.
53  	 * @return The primary id for this object.
54  	 */
55  	@JsonIgnore
56  	public String getPrimaryId();
57  	
58  	/**
59  	 * Phantom setter for entityType
60  	 * 
61  	 * The AWS DynamoDb SDK ignores bean properties that don't have both a reader 
62  	 * and a writer, failing to understand that some properties could be derived and therefore 
63  	 * NOT writable. Thus, this is ignored. This method exists simply to make DynamoDb work as expected
64  	 * for persistence.
65  	 * @param value The value to ignore
66  	 */
67  	default void setEntityType(String value) {
68  		// Do nothing
69  	}
70  	
71  	/**
72  	 * Phantom setter for sortKey
73  	 * 
74  	 * The AWS DynamoDb SDK ignores bean properties that don't have both a reader 
75  	 * and a writer, failing to understand that some properties could be derived and therefore 
76  	 * NOT writable. Thus, this is ignored. This method exists simply to make DynamoDb work as expected
77  	 * for persistence.
78  	 * @param value The value to ignore
79  	 */
80  	default void setSortKey(String value) {
81  		// Do nothing
82  	}
83  }
