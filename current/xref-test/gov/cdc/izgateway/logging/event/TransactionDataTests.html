23  /**
24   * Verifies the WSDL/schema (HTTP GET) special-casing added to {@link TransactionData}.
25   * A GET on a SOAP endpoint is a WSDL/schema retrieval, not a transaction, so it must
26   * produce a distinct message and be logged at DEBUG rather than INFO.
27   */
28  class TransactionDataTests {
29  
30      private static AppProperties previousInstance;
31  
32      @BeforeAll
33      static void initAppProperties() throws Exception {
34          // TransactionData's constructor reads AppProperties.isProduction(). Instantiating
35          // AppProperties registers itself as the global static singleton, so capture whatever
36          // instance was there first and restore it in @AfterAll to avoid making other tests
37          // (e.g. Spring tests relying on @Value-injected AppProperties) order-dependent.
38          previousInstance = AppProperties.getInstance();
39  
40          // When constructed outside Spring, serverMode is never injected and the effective mode
41          // is incidental. Force a deterministic "prod" mode so these tests do not depend on it.
42          AppProperties props = new AppProperties();
43          Field serverMode = AppProperties.class.getDeclaredField("serverMode");
44          serverMode.setAccessible(true);
45          serverMode.set(props, AppProperties.PROD_MODE_VALUE);
46      }
47  
48      @AfterAll
49      static void restoreAppProperties() throws Exception {
50          Field instance = AppProperties.class.getDeclaredField("instance");
51          instance.setAccessible(true);
52          instance.set(null, previousInstance);
53      }
54  
55      @AfterEach
56      void clearMdc() {
57          MDC.remove(LoggingValveBase.METHOD);
58      }
59  
60      private TransactionData wsdlSource() {
61          TransactionData t = new TransactionData();
62          t.getSource().setCommonName("DocketClosed");
63          t.getSource().setIpAddress("10.1.2.3");
64          return t;
65      }
66  
67      @Test
68      void getMessageForGetProducesWsdlRetrievalMessage() {
69          MDC.put(LoggingValveBase.METHOD, "GET");
70          TransactionData t = wsdlSource();
71          assertEquals("WSDL/schema retrieval from DocketClosed at 10.1.2.3", t.getMessage());
72      }
73  
74      @Test
75      void getMessageForPostProducesTransactionMessage() {
76          MDC.put(LoggingValveBase.METHOD, "POST");
77          TransactionData t = wsdlSource();
78          String message = t.getMessage();
79          assertTrue(message.startsWith("Transaction "),
80                  "POST must keep the standard transaction message, was: " + message);
81      }
82  
83      @Test
84      void getMessageWithNoMethodProducesTransactionMessage() {
85          // No MDC method set (e.g. non-HTTP context) must behave like the pre-existing default.
86          TransactionData t = wsdlSource();
87          assertTrue(t.getMessage().startsWith("Transaction "));
88      }
89  
90      @Test
91      void matchesTestDoesNotThrowOnDelimiterOnlyQpdName() {
92          // QPD-4 (fields[4]) made up only of component separators (as sent by systems that
93          // populate an empty XPN as "^^^^^^" rather than "") must not throw AIOOBE:
94          // String.split() returns a zero-length array for delimiter-only input (verified:
95          // "^^^^^^".split("\\^").length == 0), unlike a truly empty string, which yields [""].
96          String message = "MSH|^~\\&|APP|FAC|APP|FAC|20260101000000||QBP^Q11|1|P|2.5.1\r"
97                  + "QPD|Z34|1|2|^^^^^^|trailing\r";
98          assertTrue(wsdlSource().matchesTest(message));
99      }
100 
101     @Test
102     void matchesTestDoesNotThrowOnDelimiterOnlyPidName() {
103         // PID-5 (fields[5]) made up only of component separators, same AIOOBE guard as above.
104         String message = "MSH|^~\\&|APP|FAC|APP|FAC|20260101000000||RSP^K11|1|P|2.5.1\r"
105                 + "PID|1|2|3|4|^^^^^^|trailing\r";
106         assertTrue(wsdlSource().matchesTest(message));
107     }

109     @Test
110     void logItForGetLogsAtDebug() {
111         assertLogLevel("GET", Level.DEBUG);
112     }

114     @Test
115     void logItForPostLogsAtInfo() {
116         assertLogLevel("POST", Level.INFO);
117     }

119     private void assertLogLevel(String method, Level expected) {
120         Logger logger = (Logger) LoggerFactory.getLogger(TransactionData.class);
121         Level originalLevel = logger.getLevel();
122         ListAppender<ILoggingEvent> appender = new ListAppender<>();
123         appender.start();
124         logger.setLevel(Level.DEBUG); // ensure DEBUG events are not filtered out before capture
125         logger.addAppender(appender);
126         try {
127             MDC.put(LoggingValveBase.METHOD, method);
128             wsdlSource().logIt();
129         } finally {
130             logger.detachAppender(appender);
131             logger.setLevel(originalLevel);
132         }
133         List<ILoggingEvent> events = appender.list;
134         assertEquals(1, events.size(), "expected exactly one log event");
135         assertEquals(expected, events.get(0).getLevel());
136     }
