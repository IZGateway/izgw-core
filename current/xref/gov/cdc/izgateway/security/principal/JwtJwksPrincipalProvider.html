1   package gov.cdc.izgateway.security.principal;
2   
3   import lombok.extern.slf4j.Slf4j;
4   import org.apache.commons.lang3.StringUtils;
5   import org.springframework.beans.factory.annotation.Value;
6   import jakarta.annotation.Nullable;
7   import org.springframework.security.oauth2.jwt.*;
8   import org.springframework.stereotype.Component;
9   
10  @Slf4j
11  @Component
12  public class JwtJwksPrincipalProvider extends AbstractJwtPrincipalProvider {
13      @Value("${jwt.jwk-set-uri:}")
14      private String jwkSetUri;
15  
16      public JwtJwksPrincipalProvider(@Nullable GroupToRoleMapper groupToRoleMapper,
17                                      @Nullable ScopeToRoleMapper scopeToRoleMapper,
18                                      JwtTokenExtractor jwtTokenExtractor) {
19          super(groupToRoleMapper, scopeToRoleMapper, jwtTokenExtractor);
20      }
21  
22      @Override
23      protected NimbusJwtDecoder createDecoder() {
24          return NimbusJwtDecoder
25                  .withJwkSetUri(jwkSetUri)
26                  .build();
27      }
28  
29      @Override
30      protected boolean validConfiguration() {
31          if (StringUtils.isNotBlank(jwkSetUri)) {
32              return true;
33          }
34  
35          log.warn("No JWT set URI configured.  JWT authentication is disabled.");
36          return false;
37      }
38  }
