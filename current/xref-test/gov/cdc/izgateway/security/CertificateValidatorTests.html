18  /**
19   * Regression coverage for IGDD-3582: {@link CertificateValidator} takes a
20   * {@code Supplier<X509TrustManager>} rather than a fixed instance, so it always validates against
21   * whichever trust manager the supplier currently returns instead of one cached at construction.
22   */
23  class CertificateValidatorTests {
24  
25  	@Test
26  	void consultsWhicheverTrustManagerTheSupplierCurrentlyReturns() throws Exception {
27  		X509Certificate cert = mock(X509Certificate.class);
28  		X509TrustManager stale = mock(X509TrustManager.class);
29  		X509TrustManager current = mock(X509TrustManager.class);
30  		doThrow(new CertificateException("untrusted")).when(stale).checkClientTrusted(any(), anyString());
31  
32  		X509TrustManager[] live = { stale };
33  		CertificateValidator validator = new CertificateValidator(() -> live[0]);
34  
35  		assertFalse(validator.isValid(cert));
36  		verify(stale).checkClientTrusted(new X509Certificate[] { cert }, "TLS-client-auth");
37  
38  		live[0] = current;
39  		assertTrue(validator.isValid(cert));
40  		verify(current).checkClientTrusted(new X509Certificate[] { cert }, "TLS-client-auth");
41  	}
42  }
